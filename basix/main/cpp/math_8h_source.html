67template <std::floating_point T>
68void dot_blas(std::span<const T> A, std::array<std::size_t, 2> Ashape,
69 std::span<const T> B, std::array<std::size_t, 2> Bshape,
70 std::span<T> C, T alpha = 1, T beta = 0)
71{
72 static_assert(std::is_same_v<T, float> or std::is_same_v<T, double>);
73
74 assert(Ashape[1] == Bshape[0]);
75 assert(C.size() == Ashape[0] * Bshape[1]);
76
77 int M = Ashape[0];
78 int N = Bshape[1];
79 int K = Ashape[1];
80
81 int lda = K;
82 int ldb = N;
83 int ldc = N;
84 char trans = 'N';
85 if constexpr (std::is_same_v<T, float>)
86 {
87 sgemm_(&trans, &trans, &N, &M, &K, &alpha, const_cast<T*>(B.data()), &ldb,
88 const_cast<T*>(A.data()), &lda, &beta, C.data(), &ldc);
89 }
90 else if constexpr (std::is_same_v<T, double>)
91 {
92 dgemm_(&trans, &trans, &N, &M, &K, &alpha, const_cast<T*>(B.data()), &ldb,
93 const_cast<T*>(A.data()), &lda, &beta, C.data(), &ldc);
94 }
95}

103template <typename U, typename V>
104std::pair<std::vector<typename U::value_type>, std::array<std::size_t, 2>>
105outer(const U& u, const V& v)
106{
107 std::vector<typename U::value_type> result(u.size() * v.size());
108 for (std::size_t i = 0; i < u.size(); ++i)
109 for (std::size_t j = 0; j < v.size(); ++j)
110 result[i * v.size() + j] = u[i] * v[j];
111 return {std::move(result), {u.size(), v.size()}};
112}

118template <typename U, typename V>
119std::array<typename U::value_type, 3> cross(const U& u, const V& v)
120{
121 assert(u.size() == 3);
122 assert(v.size() == 3);
123 return {u[1] * v[2] - u[2] * v[1], u[2] * v[0] - u[0] * v[2],
124 u[0] * v[1] - u[1] * v[0]};
125}

134template <std::floating_point T>
135std::pair<std::vector<T>, std::vector<T>> eigh(std::span<const T> A,
136 std::size_t n)
137{
138 if (A.size() != n * n)
139 throw std::runtime_error("Matrix A must be n x n for eigh.");
140
141 // Copy A
142 std::vector<T> M(A.begin(), A.end());
143
144 // Allocate storage for eigenvalues
145 std::vector<T> w(n, 0);
146
147 int N = n;
148 char jobz = 'V'; // Compute eigenvalues and eigenvectors
149 char uplo = 'L'; // Lower
150 int ldA = n;
151 int lwork = -1;
152 int liwork = -1;
153 int info;
154 std::vector<T> work(1);
155 std::vector<int> iwork(1);
156
157 // Query optimal workspace size
158 if constexpr (std::is_same_v<T, float>)
159 {
160 ssyevd_(&jobz, &uplo, &N, M.data(), &ldA, w.data(), work.data(), &lwork,
161 iwork.data(), &liwork, &info);
162 }
163 else if constexpr (std::is_same_v<T, double>)
164 {
165 dsyevd_(&jobz, &uplo, &N, M.data(), &ldA, w.data(), work.data(), &lwork,
166 iwork.data(), &liwork, &info);
167 }
168
169 if (info != 0)
170 throw std::runtime_error("Could not find workspace size for syevd.");
171
172 // Solve eigen problem
173 work.resize(work[0]);
174 iwork.resize(iwork[0]);
175 lwork = work.size();
176 liwork = iwork.size();
177 if constexpr (std::is_same_v<T, float>)
178 {
179 ssyevd_(&jobz, &uplo, &N, M.data(), &ldA, w.data(), work.data(), &lwork,
180 iwork.data(), &liwork, &info);
181 }
182 else if constexpr (std::is_same_v<T, double>)
183 {
184 dsyevd_(&jobz, &uplo, &N, M.data(), &ldA, w.data(), work.data(), &lwork,
185 iwork.data(), &liwork, &info);
186 }
187 if (info != 0)
188 throw std::runtime_error("Eigenvalue computation did not converge.");
189
190 return {std::move(w), std::move(M)};
191}

197template <std::floating_point T>
198std::vector<T> solve(md::mdspan<const T, md::dextents<std::size_t, 2>> A,
199 md::mdspan<const T, md::dextents<std::size_t, 2>> B)
200{
201 if (A.extent(0) != A.extent(1))
202 throw std::runtime_error("Matrix A must be square to solve A X = B.");
203 if (A.extent(0) != B.extent(0))
204 {
205 throw std::runtime_error("Matrix A and matrix B must have the same "
206 "number of rows to solve A X = B.");
207 }
208
209 // Copy A and B to column-major storage
210 mdex::mdarray<T, md::dextents<std::size_t, 2>, md::layout_left> _A(
211 A.extents()),
212 _B(B.extents());
213 for (std::size_t i = 0; i < A.extent(0); ++i)
214 for (std::size_t j = 0; j < A.extent(1); ++j)
215 _A(i, j) = A(i, j);
216 for (std::size_t i = 0; i < B.extent(0); ++i)
217 for (std::size_t j = 0; j < B.extent(1); ++j)
218 _B(i, j) = B(i, j);
219
220 int N = _A.extent(0);
221 int nrhs = _B.extent(1);
222 int lda = _A.extent(0);
223 int ldb = _B.extent(0);
224
225 // Pivot indices that define the permutation matrix for the LU solver
226 std::vector<int> piv(N);
227 int info;
228 if constexpr (std::is_same_v<T, float>)
229 sgesv_(&N, &nrhs, _A.data(), &lda, piv.data(), _B.data(), &ldb, &info);
230 else if constexpr (std::is_same_v<T, double>)
231 dgesv_(&N, &nrhs, _A.data(), &lda, piv.data(), _B.data(), &ldb, &info);
232 if (info != 0)
233 throw std::runtime_error("Call to dgesv failed: " + std::to_string(info));
234
235 // Copy result to row-major storage
236 std::vector<T> rb(_B.extent(0) * _B.extent(1));
237 md::mdspan<T, md::dextents<std::size_t, 2>> r(rb.data(), _B.extents());
238 for (std::size_t i = 0; i < _B.extent(0); ++i)
239 for (std::size_t j = 0; j < _B.extent(1); ++j)
240 r(i, j) = _B(i, j);
241
242 return rb;
243}

248template <std::floating_point T>
249bool is_singular(md::mdspan<const T, md::dextents<std::size_t, 2>> A)
250{
251 if (A.extent(0) != A.extent(1))
252 {
253 throw std::runtime_error(
254 "Matrix A must be square to test for singularity.");
255 }
256
257 // Copy to column major matrix
258 mdex::mdarray<T, md::dextents<std::size_t, 2>, md::layout_left> _A(
259 A.extents());
260 for (std::size_t i = 0; i < A.extent(0); ++i)
261 for (std::size_t j = 0; j < A.extent(1); ++j)
262 _A(i, j) = A(i, j);
263
264 std::vector<T> B(A.extent(0), 1);
265 int N = _A.extent(0);
266 int nrhs = 1;
267 int lda = _A.extent(0);
268 int ldb = B.size();
269
270 // Pivot indices that define the permutation matrix for the LU solver
271 std::vector<int> piv(N);
272 int info;
273 if constexpr (std::is_same_v<T, float>)
274 sgesv_(&N, &nrhs, _A.data(), &lda, piv.data(), B.data(), &ldb, &info);
275 else if constexpr (std::is_same_v<T, double>)
276 dgesv_(&N, &nrhs, _A.data(), &lda, piv.data(), B.data(), &ldb, &info);
277
278 if (info < 0)
279 {
280 throw std::runtime_error("dgesv failed due to invalid value: "
281 + std::to_string(info));
282 }
283 else if (info > 0)
284 return true;
285 else
286 return false;
287}

294template <std::floating_point T>
295std::vector<std::size_t>
296transpose_lu(std::pair<std::vector<T>, std::array<std::size_t, 2>>& A)
297{
298 std::size_t dim = A.second[0];
299 if (dim != A.second[1])
300 throw std::runtime_error("Matrix A must be square for transpose_lu.");
301 int N = dim;
302 int info;
303 std::vector<int> lu_perm(dim);
304
305 // Comput LU decomposition of M
306 if constexpr (std::is_same_v<T, float>)
307 sgetrf_(&N, &N, A.first.data(), &N, lu_perm.data(), &info);
308 else if constexpr (std::is_same_v<T, double>)
309 dgetrf_(&N, &N, A.first.data(), &N, lu_perm.data(), &info);
310
311 if (info != 0)
312 {
313 throw std::runtime_error("LU decomposition failed: "
314 + std::to_string(info));
315 }
316
317 std::vector<std::size_t> perm(dim);
318 for (std::size_t i = 0; i < dim; ++i)
319 perm[i] = static_cast<std::size_t>(lu_perm[i] - 1);
320
321 return perm;
322}

331template <typename U, typename V, typename W>
332void dot(const U& A, const V& B, W&& C,
333 typename std::decay_t<U>::value_type alpha = 1,
334 typename std::decay_t<U>::value_type beta = 0)
335{
336 using T = typename std::decay_t<U>::value_type;
337
338 if (A.extent(1) != B.extent(0))
339 {
340 throw std::runtime_error("Matrix dimension mismatch: A.extent(1) != "
341 "B.extent(0) in dot(A, B, C).");
342 }
343 if (C.extent(0) != A.extent(0) or C.extent(1) != B.extent(1))
344 {
345 throw std::runtime_error("Output matrix C has the wrong shape in "
346 "dot(A, B, C).");
347 }
348 if (A.extent(0) * B.extent(1) * A.extent(1) < 256)
349 {
350 for (std::size_t i = 0; i < A.extent(0); ++i)
351 {
352 for (std::size_t j = 0; j < B.extent(1); ++j)
353 {
354 T C0 = C(i, j);
355 C(i, j) = 0;
356 T& _C = C(i, j);
357 for (std::size_t k = 0; k < A.extent(1); ++k)
358 _C += A(i, k) * B(k, j);
359 _C = alpha * _C + beta * C0;
360 }
361 }
362 }
363 else
364 {
365 static_assert(std::is_same_v<typename std::decay_t<U>::layout_type,
366 md::layout_right>);
367 static_assert(std::is_same_v<typename std::decay_t<V>::layout_type,
368 md::layout_right>);
369 static_assert(std::is_same_v<typename std::decay_t<W>::layout_type,
370 md::layout_right>);
371 static_assert(std::is_same_v<typename std::decay_t<V>::value_type, T>);
372 static_assert(std::is_same_v<typename std::decay_t<W>::value_type, T>);
373 impl::dot_blas<T>(
374 std::span(A.data_handle(), A.size()), {A.extent(0), A.extent(1)},
375 std::span(B.data_handle(), B.size()), {B.extent(0), B.extent(1)},
376 std::span(C.data_handle(), C.size()), alpha, beta);
377 }
378}

397template <std::floating_point T>
398void orthogonalise(md::mdspan<T, md::dextents<std::size_t, 2>> wcoeffs,
399 std::size_t start = 0)
400{
401 using mdspan2_t = md::mdspan<T, md::dextents<std::size_t, 2>>;
402 using cmdspan2_t = md::mdspan<const T, md::dextents<std::size_t, 2>>;
403
404 const std::size_t ndofs = wcoeffs.extent(0);
405 const std::size_t psize = wcoeffs.extent(1);
406
407 // Classical Gram-Schmidt: after normalising row i, its projection onto
408 // every later row is subtracted. Rather than looping over each row j,
409 // the projection coefficients for all remaining rows are computed as a
410 // single matrix-vector product, and the subtraction as a single rank-1
411 // update, both via BLAS (math::dot).
412 std::vector<T> a_vec;
413 std::vector<T> update;
414 for (std::size_t i = start; i < ndofs; ++i)
415 {
416 T norm = 0;
417 for (std::size_t k = 0; k < psize; ++k)
418 norm += wcoeffs(i, k) * wcoeffs(i, k);
419
420 norm = std::sqrt(norm);
421 if (norm < 2 * std::numeric_limits<T>::epsilon())
422 {
423 throw std::runtime_error("Cannot orthogonalise the rows of a matrix "
424 "with incomplete row rank");
425 }
426
427 for (std::size_t k = 0; k < psize; ++k)
428 wcoeffs(i, k) /= norm;
429
430 const std::size_t nrem = ndofs - i - 1;
431 if (nrem == 0)
432 continue;
433
434 // Row i viewed as a (psize, 1) column and a (1, psize) row (both are
435 // just the same contiguous length-psize data), and the remaining
436 // rows viewed as an (nrem, psize) block.
437 cmdspan2_t row_col(&wcoeffs(i, 0), psize, 1);
438 cmdspan2_t row_row(&wcoeffs(i, 0), 1, psize);
439 cmdspan2_t tail(&wcoeffs(i + 1, 0), nrem, psize);
440
441 // a = tail * row_i
442 a_vec.resize(nrem);
443 mdspan2_t a_view(a_vec.data(), nrem, 1);
444 dot(tail, row_col, a_view);
445
446 // update = a * row_i^T
447 update.resize(nrem * psize);
448 mdspan2_t update_view(update.data(), nrem, psize);
449 dot(cmdspan2_t(a_vec.data(), nrem, 1), row_row, update_view);
450
451 mdspan2_t tail_mut(&wcoeffs(i + 1, 0), nrem, psize);
452 for (std::size_t j = 0; j < nrem; ++j)
453 for (std::size_t k = 0; k < psize; ++k)
454 tail_mut(j, k) -= update_view(j, k);
455 }
456}
