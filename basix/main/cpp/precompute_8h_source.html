136template <typename E>
137void apply_permutation(std::span<const std::size_t> perm, std::span<E> data,
138 std::size_t offset = 0, std::size_t n = 1)
139{
140 for (std::size_t i = 0; i < perm.size(); ++i)
141 for (std::size_t b = 0; b < n; ++b)
142 std::swap(data[n * (offset + i) + b], data[n * (offset + perm[i]) + b]);
143}

164template <typename E>
165void apply_inv_permutation_right(std::span<const std::size_t> perm,
166 std::span<E> data, std::size_t offset = 0,
167 std::size_t n = 1)
168{
169 const std::size_t dim = perm.size();
170 const std::size_t data_size = (data.size() + (dim < n ? n - dim : 0)) / n;
171 for (std::size_t b = 0; b < n; ++b)
172 {
173 for (std::size_t i = 0; i < dim; ++i)
174 {
175 std::swap(data[data_size * b + offset + i],
176 data[data_size * b + offset + perm[i]]);
177 }
178 }
179}

235template <typename T, typename E>
236void apply_matrix(std::span<const std::size_t> v_size_t,
237 md::mdspan<const T, md::dextents<std::size_t, 2>> M,
238 std::span<E> data, std::size_t offset = 0, std::size_t n = 1)
239{
240 using U = typename impl::scalar_value_type_t<E>;
241
242 const std::size_t dim = v_size_t.size();
243 apply_permutation(v_size_t, data, offset, n);
244
245 // data has n contiguous (unit-stride) values per dof. Looping over b
246 // innermost vectorises each elimination step, but for small n (common
247 // in practice, e.g. 1-3) the loop-entry cost per (i, j) pair outweighs
248 // that benefit, so fall back to the original b-outermost ordering.
249 // Threshold chosen from measurements showing the crossover is n=12-16.
250 if (n <= 12)
251 {
252 for (std::size_t b = 0; b < n; ++b)
253 {
254 for (std::size_t i = 0; i < dim; ++i)
255 {
256 for (std::size_t j = i + 1; j < dim; ++j)
257 {
258 data[n * (offset + i) + b]
259 += static_cast<U>(M(i, j)) * data[n * (offset + j) + b];
260 }
261 }
262
263 for (std::size_t i = 1; i <= dim; ++i)
264 {
265 data[n * (offset + dim - i) + b] *= static_cast<U>(M(dim - i, dim - i));
266 for (std::size_t j = 0; j < dim - i; ++j)
267 {
268 data[n * (offset + dim - i) + b]
269 += static_cast<U>(M(dim - i, j)) * data[n * (offset + j) + b];
270 }
271 }
272 }
273 }
274 else
275 {
276 for (std::size_t i = 0; i < dim; ++i)
277 {
278 for (std::size_t j = i + 1; j < dim; ++j)
279 {
280 const U Mij = static_cast<U>(M(i, j));
281 for (std::size_t b = 0; b < n; ++b)
282 {
283 data[n * (offset + i) + b] += Mij * data[n * (offset + j) + b];
284 }
285 }
286 }
287
288 for (std::size_t i = 1; i <= dim; ++i)
289 {
290 const U Mdiag = static_cast<U>(M(dim - i, dim - i));
291 for (std::size_t b = 0; b < n; ++b)
292 data[n * (offset + dim - i) + b] *= Mdiag;
293
294 for (std::size_t j = 0; j < dim - i; ++j)
295 {
296 const U Mij = static_cast<U>(M(dim - i, j));
297 for (std::size_t b = 0; b < n; ++b)
298 {
299 data[n * (offset + dim - i) + b]
300 += Mij * data[n * (offset + j) + b];
301 }
302 }
303 }
304 }
305}

315template <typename T, typename E>
316void apply_tranpose_matrix_right(
317 std::span<const std::size_t> v_size_t,
318 md::mdspan<const T, md::dextents<std::size_t, 2>> M, std::span<E> data,
319 std::size_t offset = 0, std::size_t n = 1)
320{
321 using U = typename impl::scalar_value_type_t<E>;
322
323 const std::size_t dim = v_size_t.size();
324 const std::size_t data_size = (data.size() + (dim < n ? n - dim : 0)) / n;
325 apply_inv_permutation_right(v_size_t, data, offset, n);
326 for (std::size_t b = 0; b < n; ++b)
327 {
328 for (std::size_t i = 0; i < dim; ++i)
329 {
330 for (std::size_t j = i + 1; j < dim; ++j)
331 {
332 data[data_size * b + offset + i]
333 += static_cast<U>(M(i, j)) * data[data_size * b + offset + j];
334 }
335 }
336 for (std::size_t i = 1; i <= dim; ++i)
337 {
338 data[data_size * b + offset + dim - i]
339 *= static_cast<U>(M(dim - i, dim - i));
340 for (std::size_t j = 0; j < dim - i; ++j)
341 {
342 data[data_size * b + offset + dim - i]
343 += static_cast<U>(M(dim - i, j)) * data[data_size * b + offset + j];
344 }
345 }
346 }
347}
