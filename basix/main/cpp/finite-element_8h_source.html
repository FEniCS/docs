46to_mdspan(std::array<std::vector<mdarray_t<T, 2>>, 4>& x)
47{
48 std::array<std::vector<mdspan_t<const T, 2>>, 4> x1;

50 for (std::size_t j = 0; j < x[i].size(); ++j)

634 template <typename O, typename P, typename Q, typename R>
635 std::function<void(O&, const P&, const Q&, F, const R&)> map_fn() const
636 {
637 switch (_map_type)
638 {
639 case maps::type::identity:
640 return [](O& u, const P& U, const Q&, F, const R&)
641 {
642 assert(U.extent(0) == u.extent(0));
643 assert(U.extent(1) == u.extent(1));
644 for (std::size_t i = 0; i < U.extent(0); ++i)
645 for (std::size_t j = 0; j < U.extent(1); ++j)
646 u(i, j) = U(i, j);
647 };
648 case maps::type::covariantPiola:
649 return [](O& u, const P& U, const Q& J, F detJ, const R& K)
650 { maps::covariant_piola(u, U, J, detJ, K); };
651 case maps::type::contravariantPiola:
652 return [](O& u, const P& U, const Q& J, F detJ, const R& K)
653 { maps::contravariant_piola(u, U, J, detJ, K); };
654 case maps::type::doubleCovariantPiola:
655 return [](O& u, const P& U, const Q& J, F detJ, const R& K)
656 { maps::double_covariant_piola(u, U, J, detJ, K); };
657 case maps::type::doubleContravariantPiola:
658 return [](O& u, const P& U, const Q& J, F detJ, const R& K)
659 { maps::double_contravariant_piola(u, U, J, detJ, K); };
660 default:
661 throw std::runtime_error("Map not implemented");
662 }
663 }

868 void permute_subentity_closure(std::span<std::int32_t> d,
869 std::uint32_t cell_info,
870 cell::type entity_type, int entity_index) const
871 {
872 const int entity_dim = cell::topological_dimension(entity_type);
873
874 int face_start = _cell_tdim == 3 ? 3 * _edofs[2].size() : 0;
875
876 std::uint32_t entity_info = 0;
877 switch (entity_dim)
878 {
879 case 0:
880 return;
881 case 1:
882 entity_info = cell_info >> (face_start + entity_index) & 1;
883 break;
884 case 2:
885 entity_info = cell_info >> (3 * entity_index) & 7;
886 break;
887 default:
888 throw std::runtime_error("Unsupported cell dimension");
889 }
890 permute_subentity_closure(d, entity_info, entity_type);
891 }

946 void permute_subentity_closure(std::span<std::int32_t> d,
947 std::uint32_t entity_info,
948 cell::type entity_type) const
949 {
950 if (!_dof_transformations_are_permutations)
951 {
952 throw std::runtime_error(
953 "The DOF transformations for this element are not permutations");
954 }
955
956 const int entity_dim = cell::topological_dimension(entity_type);
957
958 if (entity_dim == 0)
959 return;
960
961 auto& perm = _subentity_closure_perm.at(entity_type);
962 if (entity_dim == 1)
963 {
964 if (entity_info & 1)
965 {
966 precompute::apply_permutation(perm[0], d);
967 }
968 }
969 else if (entity_dim == 2)
970 {
971 // Rotate a face
972 for (std::uint32_t r = 0; r < (entity_info >> 1 & 3); ++r)
973 {
974 precompute::apply_permutation(perm[0], d);
975 }
976
977 // Reflect a face (post rotate)
978 if (entity_info & 1)
979 {
980 precompute::apply_permutation(perm[1], d);
981 }
982 }
983 else
984 {
985 throw std::runtime_error(
986 "Invalid dimension for permute_subentity_closure");
987 }
988 }

1731template <std::floating_point F>
1732template <typename T, bool post>
1733void FiniteElement<F>::permute_data(
1734 std::span<T> data, int block_size, std::uint32_t cell_info,
1735 const std::map<cell::type, std::vector<std::vector<std::size_t>>>& eperm)
1736 const
1737{
1738 if (_cell_tdim >= 2)
1739 {
1740 // This assumes 3 bits are used per face. This will need updating if 3D
1741 // cells with faces with more than 4 sides are implemented
1742 int face_start = _cell_tdim == 3 ? 3 * _edofs[2].size() : 0;
1743
1744 // Permute DOFs on edges
1745 {
1746 auto& trans = eperm.at(cell::type::interval)[0];
1747 for (std::size_t e = 0; e < _edofs[1].size(); ++e)
1748 {
1749 // Reverse an edge
1750 if (cell_info >> (face_start + e) & 1)
1751 {
1752 precompute::apply_permutation_mapped(trans, data, _edofs[1][e],
1753 block_size);
1754 }
1755 }
1756 }
1757
1758 if (_cell_tdim == 3)
1759 {
1760 // Permute DOFs on faces
1761 for (std::size_t f = 0; f < _edofs[2].size(); ++f)
1762 {
1763 auto& trans = eperm.at(_cell_subentity_types[2][f]);
1764
1765 // Reflect a face (pre rotate)
1766 if (!post and cell_info >> (3 * f) & 1)
1767 {
1768 precompute::apply_permutation_mapped(trans[1], data, _edofs[2][f],
1769 block_size);
1770 }
1771
1772 // Rotate a face
1773 for (std::uint32_t r = 0; r < (cell_info >> (3 * f + 1) & 3); ++r)
1774 {
1775 precompute::apply_permutation_mapped(trans[0], data, _edofs[2][f],
1776 block_size);
1777 }
1778
1779 // Reflect a face (post rotate)
1780 if (post and cell_info >> (3 * f) & 1)
1781 {
1782 precompute::apply_permutation_mapped(trans[1], data, _edofs[2][f],
1783 block_size);
1784 }
1785 }
1786 }
1787 }
1788}

1790template <std::floating_point F>
1791template <typename T, bool post, typename OP>
1792void FiniteElement<F>::transform_data(
1793 std::span<T> data, int block_size, std::uint32_t cell_info,
1794 const std::map<cell::type, trans_data_t>& etrans, OP op) const
1795{
1796 if (_cell_tdim >= 2)
1797 {
1798 // This assumes 3 bits are used per face. This will need updating if
1799 // 3D cells with faces with more than 4 sides are implemented
1800 int face_start = _cell_tdim == 3 ? 3 * _edofs[2].size() : 0;
1801 int dofstart = 0;
1802 for (auto& edofs0 : _edofs[0])
1803 dofstart += edofs0.size();
1804
1805 // Transform DOFs on edges
1806 {
1807 auto& [v_size_t, matrix] = etrans.at(cell::type::interval)[0];
1808 for (std::size_t e = 0; e < _edofs[1].size(); ++e)
1809 {
1810 // Reverse an edge
1811 if (cell_info >> (face_start + e) & 1)
1812 {
1813 op(std::span(v_size_t),
1814 mdspan_t<const F, 2>(matrix.first.data(), matrix.second), data,
1815 dofstart, block_size);
1816 }
1817 dofstart += _edofs[1][e].size();
1818 }
1819 }
1820
1821 if (_cell_tdim == 3)
1822 {
1823 // Permute DOFs on faces
1824 for (std::size_t f = 0; f < _edofs[2].size(); ++f)
1825 {
1826 auto& trans = etrans.at(_cell_subentity_types[2][f]);
1827
1828 // Reflect a face (pre rotation)
1829 if (!post and cell_info >> (3 * f) & 1)
1830 {
1831 const auto& m = trans[1];
1832 const auto& v_size_t = std::get<0>(m);
1833 const auto& matrix = std::get<1>(m);
1834 op(std::span(v_size_t),
1835 mdspan_t<const F, 2>(matrix.first.data(), matrix.second), data,
1836 dofstart, block_size);
1837 }
1838
1839 // Rotate a face
1840 for (std::uint32_t r = 0; r < (cell_info >> (3 * f + 1) & 3); ++r)
1841 {
1842 const auto& m = trans[0];
1843 const auto& v_size_t = std::get<0>(m);
1844 const auto& matrix = std::get<1>(m);
1845 op(std::span(v_size_t),
1846 mdspan_t<const F, 2>(matrix.first.data(), matrix.second), data,
1847 dofstart, block_size);
1848 }
1849
1850 // Reflect a face (post rotation)
1851 if (post and cell_info >> (3 * f) & 1)
1852 {
1853 const auto& m = trans[1];
1854 const auto& v_size_t = std::get<0>(m);
1855 const auto& matrix = std::get<1>(m);
1856 op(std::span(v_size_t),
1857 mdspan_t<const F, 2>(matrix.first.data(), matrix.second), data,
1858 dofstart, block_size);
1859 }
1860
1861 dofstart += _edofs[2][f].size();
1862 }
1863 }
1864 }
