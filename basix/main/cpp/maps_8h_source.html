58template <typename O, typename P, typename Q, typename R>
59void l2_piola(O&& r, const P& U, const Q& /*J*/, double detJ, const R& /*K*/)
60{
61 assert(U.extent(0) == r.extent(0));
62 assert(U.extent(1) == r.extent(1));
63 for (std::size_t i = 0; i < U.extent(0); ++i)
64 for (std::size_t j = 0; j < U.extent(1); ++j)
65 r(i, j) = U(i, j) / detJ;
66}

75template <typename O, typename P, typename Q, typename R>
76void covariant_piola(O&& r, const P& U, const Q& /*J*/, double /*detJ*/,
77 const R& K)
78{
79 using T = typename std::decay_t<O>::value_type;
80 using Z = typename impl::scalar_value_type_t<T>;
81 for (std::size_t p = 0; p < U.extent(0); ++p)
82 {
83 // r_p = K^T U_p, where p indicates the p-th row
84 for (std::size_t i = 0; i < r.extent(1); ++i)
85 {
86 T acc = 0;
87 for (std::size_t k = 0; k < K.extent(0); ++k)
88 acc += static_cast<Z>(K(k, i)) * U(p, k);
89 r(p, i) = acc;
90 }
91 }
92}

102template <typename O, typename P, typename Q, typename R>
103void contravariant_piola(O&& r, const P& U, const Q& J, double detJ,
104 const R& /*K*/)
105{
106 using T = typename std::decay_t<O>::value_type;
107 using Z = typename impl::scalar_value_type_t<T>;
108 for (std::size_t p = 0; p < U.extent(0); ++p)
109 {
110 for (std::size_t i = 0; i < r.extent(1); ++i)
111 {
112 T acc = 0;
113 for (std::size_t k = 0; k < J.extent(1); ++k)
114 acc += static_cast<Z>(J(i, k)) * U(p, k);
115 r(p, i) = acc;
116 }
117 }
118
119 std::transform(r.data_handle(), r.data_handle() + r.size(), r.data_handle(),
120 [detJ](auto ri) { return ri / static_cast<Z>(detJ); });
121}

130template <typename O, typename P, typename Q, typename R>
131void double_covariant_piola(O&& r, const P& U, const Q& /*J*/, double /*detJ*/,
132 const R& K)
133{
134 using T = typename std::decay_t<O>::value_type;
135 using Z = typename impl::scalar_value_type_t<T>;
136 for (std::size_t p = 0; p < U.extent(0); ++p)
137 {
138 md::mdspan<const T, md::dextents<std::size_t, 2>> _U(
139 U.data_handle() + p * U.extent(1), K.extent(0), K.extent(0));
140 md::mdspan<T, md::dextents<std::size_t, 2>> _r(
141 r.data_handle() + p * r.extent(1), K.extent(1), K.extent(1));
142 // _r = K^T _U K
143 for (std::size_t i = 0; i < _r.extent(0); ++i)
144 {
145 for (std::size_t j = 0; j < _r.extent(1); ++j)
146 {
147 T acc = 0;
148 for (std::size_t k = 0; k < K.extent(0); ++k)
149 for (std::size_t l = 0; l < _U.extent(1); ++l)
150 acc += static_cast<Z>(K(k, i)) * _U(k, l) * static_cast<Z>(K(l, j));
151 _r(i, j) = acc;
152 }
153 }
154 }
155}

165template <typename O, typename P, typename Q, typename R>
166void double_contravariant_piola(O&& r, const P& U, const Q& J, double detJ,
167 const R& /*K*/)
168{
169 using T = typename std::decay_t<O>::value_type;
170 using Z = typename impl::scalar_value_type_t<T>;
171 for (std::size_t p = 0; p < U.extent(0); ++p)
172 {
173 md::mdspan<const T, md::dextents<std::size_t, 2>> _U(
174 U.data_handle() + p * U.extent(1), J.extent(1), J.extent(1));
175 md::mdspan<T, md::dextents<std::size_t, 2>> _r(
176 r.data_handle() + p * r.extent(1), J.extent(0), J.extent(0));
177
178 // _r = J U J^T
179 for (std::size_t i = 0; i < _r.extent(0); ++i)
180 {
181 for (std::size_t j = 0; j < _r.extent(1); ++j)
182 {
183 T acc = 0;
184 for (std::size_t k = 0; k < J.extent(1); ++k)
185 for (std::size_t l = 0; l < _U.extent(1); ++l)
186 acc += static_cast<Z>(J(i, k)) * _U(k, l) * static_cast<Z>(J(j, l));
187 _r(i, j) = acc;
188 }
189 }
190 }
191
192 std::transform(r.data_handle(), r.data_handle() + r.size(), r.data_handle(),
193 [detJ](auto ri) { return ri / static_cast<Z>(detJ * detJ); });
194}
