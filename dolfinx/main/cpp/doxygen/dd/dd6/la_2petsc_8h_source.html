280 static auto set_fn(Mat A, InsertMode mode)
281 {
282 return [A, mode, cache = std::vector<PetscInt>()](
283 const common::LocalIndexRange auto& rows,
284 const common::LocalIndexRange auto& cols,
285 std::span<const PetscScalar> vals) mutable -> int
286 {
287 PetscErrorCode ierr;
288#ifdef PETSC_USE_64BIT_INDICES
289 cache.resize(std::ranges::size(rows) + std::ranges::size(cols));
290 std::ranges::copy(rows, cache.begin());
291 std::ranges::copy(cols,
292 std::next(cache.begin(), std::ranges::size(rows)));
293 const PetscInt* _rows = cache.data();
294 const PetscInt* _cols = cache.data() + std::ranges::size(rows);
295 ierr = MatSetValuesLocal(A, std::ranges::size(rows), _rows,
296 std::ranges::size(cols), _cols, vals.data(),
297 mode);
298#else
299 ierr = MatSetValuesLocal(A, std::ranges::size(rows),
300 std::ranges::data(rows), std::ranges::size(cols),
301 std::ranges::data(cols), vals.data(), mode);
302#endif
303
304#ifndef NDEBUG
305 common::petsc::check(ierr, "MatSetValuesLocal");
306#endif
307 return ierr;
308 };
309 }

357 static auto set_block_expand_fn(Mat A, int bs0, int bs1, InsertMode mode)
358 {
359 return [A, bs0, bs1, mode, cache0 = std::vector<PetscInt>(),
360 cache1 = std::vector<PetscInt>()](
361 const common::LocalIndexRange auto& rows,
362 const common::LocalIndexRange auto& cols,
363 std::span<const PetscScalar> vals) mutable -> int
364 {
365 PetscErrorCode ierr;
366 const std::int32_t* _rows = std::ranges::data(rows);
367 const std::int32_t* _cols = std::ranges::data(cols);
368 const std::size_t num_rows = std::ranges::size(rows);
369 const std::size_t num_cols = std::ranges::size(cols);
370 cache0.resize(bs0 * num_rows);
371 cache1.resize(bs1 * num_cols);
372 for (std::size_t i = 0; i < num_rows; ++i)
373 for (int k = 0; k < bs0; ++k)
374 cache0[bs0 * i + k] = bs0 * _rows[i] + k;
375
376 for (std::size_t i = 0; i < num_cols; ++i)
377 for (int k = 0; k < bs1; ++k)
378 cache1[bs1 * i + k] = bs1 * _cols[i] + k;
379
380 ierr = MatSetValuesLocal(A, cache0.size(), cache0.data(), cache1.size(),
381 cache1.data(), vals.data(), mode);
382#ifndef NDEBUG
383 common::petsc::check(ierr, "MatSetValuesLocal");
384#endif
385 return ierr;
386 };
387 }
