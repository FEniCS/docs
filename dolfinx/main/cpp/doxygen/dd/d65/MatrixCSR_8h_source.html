22// Define requirements on sparsity pattern required for MatrixCSR constructor
23// allowing alternative implementations that can provide these essentials.
24template <typename T>
25concept SparsityImplementation = requires(T sp, int i) {
26 { sp.graph() };
27 requires std::forward_iterator<typename decltype(sp.graph().first)::iterator>;
28 requires std::convertible_to<std::int32_t,
29 typename decltype(sp.graph().first)::value_type>;
30 requires std::forward_iterator<
31 typename decltype(sp.graph().second)::iterator>;
32 requires std::convertible_to<
33 std::int64_t, typename decltype(sp.graph().second)::value_type>;
34
35 { sp.block_size(i) } -> std::same_as<int>;
36 {
37 sp.index_map(i)
38 } -> std::same_as<std::shared_ptr<const dolfinx::common::IndexMap>>;
39};

115 template <int BS0 = 1, int BS1 = 1>
116 auto mat_set_values()
117 {
118 if ((BS0 != _bs[0] and BS0 > 1 and _bs[0] > 1)
119 or (BS1 != _bs[1] and BS1 > 1 and _bs[1] > 1))
120 {
121 throw std::runtime_error(
122 "Cannot insert blocks of different size than matrix block size");
123 }
124
125 return [this](std::span<const std::int32_t> rows,
126 std::span<const std::int32_t> cols,
127 std::span<const value_type> data) -> int
128 {
129 this->set<BS0, BS1>(data, rows, cols);
130 return 0;
131 };
132 }

279 template <int BS0, int BS1>
280 void set(std::span<const value_type> x, std::span<const std::int32_t> rows,
281 std::span<const std::int32_t> cols)
282 {
283 check_not_finalized();
284 auto set_fn = [](value_type& y, const value_type& x) { y = x; };
285
286 std::int32_t num_rows
287 = _index_maps[0]->size_local() + _index_maps[0]->num_ghosts();
288 assert(x.size() == rows.size() * cols.size() * BS0 * BS1);
289 if (_bs[0] == BS0 and _bs[1] == BS1)
290 {
291 impl::insert_csr<BS0, BS1>(_data, _cols, _row_ptr, x, rows, cols, set_fn,
292 num_rows);
293 }
294 else if (_bs[0] == 1 and _bs[1] == 1)
295 {
296 // Set blocked data in a regular CSR matrix (_bs[0]=1, _bs[1]=1)
297 // with correct sparsity
298 impl::insert_blocked_csr<BS0, BS1>(_data, _cols, _row_ptr, x, rows, cols,
299 set_fn, num_rows);
300 }
301 else
302 {
303 assert(BS0 == 1 and BS1 == 1);
304 // Set non-blocked data in a blocked CSR matrix (BS0=1, BS1=1)
305 impl::insert_nonblocked_csr(_data, _cols, _row_ptr, x, rows, cols, set_fn,
306 num_rows, _bs[0], _bs[1]);
307 }
308 }

325 template <int BS0 = 1, int BS1 = 1>
326 void add(std::span<const value_type> x, std::span<const std::int32_t> rows,
327 std::span<const std::int32_t> cols)
328 {
329 check_not_finalized();
330 auto add_fn = [](value_type& y, const value_type& x) { y += x; };
331
332 assert(x.size() == rows.size() * cols.size() * BS0 * BS1);
333 if (_bs[0] == BS0 and _bs[1] == BS1)
334 {
335 impl::insert_csr<BS0, BS1>(_data, _cols, _row_ptr, x, rows, cols, add_fn,
336 _row_ptr.size());
337 }
338 else if (_bs[0] == 1 and _bs[1] == 1)
339 {
340 // Add blocked data to a regular CSR matrix (_bs[0]=1, _bs[1]=1)
341 impl::insert_blocked_csr<BS0, BS1>(_data, _cols, _row_ptr, x, rows, cols,
342 add_fn, _row_ptr.size());
343 }
344 else
345 {
346 assert(BS0 == 1 and BS1 == 1);
347 // Add non-blocked data to a blocked CSR matrix (BS0=1, BS1=1)
348 impl::insert_nonblocked_csr(_data, _cols, _row_ptr, x, rows, cols, add_fn,
349 _row_ptr.size(), _bs[0], _bs[1]);
350 }

368 std::vector<value_type> to_dense() const
369 {
370 const std::size_t nrows = num_all_rows();
371 const std::size_t ncols = _index_maps[1]->size_global();
372 std::vector<value_type> A(nrows * ncols * _bs[0] * _bs[1], value_type(0));
373 for (std::size_t r = 0; r < nrows; ++r)
374 {
375 for (std::int32_t j = _row_ptr[r]; j < _row_ptr[r + 1]; ++j)
376 {
377 for (int i0 = 0; i0 < _bs[0]; ++i0)
378 {
379 for (int i1 = 0; i1 < _bs[1]; ++i1)
380 {
381 std::array<std::int32_t, 1> local_col{_cols[j]};
382 std::array<std::int64_t, 1> global_col{0};
383 _index_maps[1]->local_to_global(local_col, global_col);
384 A[(r * _bs[0] + i0) * ncols * _bs[1] + global_col[0] * _bs[1] + i1]
385 = _data[j * _bs[0] * _bs[1] + i0 * _bs[1] + i1];
386 }
387 }
388 }
389 }
390
391 return A;

415 void scatter_rev_begin()
416 {
417 check_not_finalized();
418 const std::int32_t local_size0 = _index_maps[0]->size_local();
419 const std::int32_t num_ghosts0 = _index_maps[0]->num_ghosts();
420 const int bs2 = _bs[0] * _bs[1];
421
422 // For each ghost row, pack and send values to send to neighborhood
423 std::vector<int> insert_pos = _val_send_disp;
424 _ghost_value_data.resize(_val_send_disp.back());
425 for (int i = 0; i < num_ghosts0; ++i)
426 {
427 int rank = _ghost_row_to_rank[i];
428
429 // Get position in send buffer to place data to send to this
430 // neighbour
431 std::int32_t val_pos = insert_pos[rank];
432 std::copy(std::next(_data.data(), _row_ptr[local_size0 + i] * bs2),
433 std::next(_data.data(), _row_ptr[local_size0 + i + 1] * bs2),
434 std::next(_ghost_value_data.begin(), val_pos));
435 insert_pos[rank]
436 += bs2 * (_row_ptr[local_size0 + i + 1] - _row_ptr[local_size0 + i]);
437 }
438
439 _ghost_value_data_in.resize(_val_recv_disp.back());
440
441 // Compute data sizes for send and receive from displacements
442 std::vector<int> val_send_count(_val_send_disp.size() - 1);
443 std::adjacent_difference(std::next(_val_send_disp.begin()),
444 _val_send_disp.end(), val_send_count.begin());
445
446 std::vector<int> val_recv_count(_val_recv_disp.size() - 1);
447 std::adjacent_difference(std::next(_val_recv_disp.begin()),
448 _val_recv_disp.end(), val_recv_count.begin());
449
450 int status = MPI_Ineighbor_alltoallv(
451 _ghost_value_data.data(), val_send_count.data(), _val_send_disp.data(),
452 dolfinx::MPI::mpi_t<value_type>, _ghost_value_data_in.data(),
453 val_recv_count.data(), _val_recv_disp.data(),
454 dolfinx::MPI::mpi_t<value_type>, _comm.comm(), &_request);
455 dolfinx::MPI::check_error(_comm.comm(), status);
456 }
457
463 void scatter_rev_end()
464 {
465 check_not_finalized();
466 int status = MPI_Wait(&_request, MPI_STATUS_IGNORE);
467 dolfinx::MPI::check_error(_comm.comm(), status);
468
469 _ghost_value_data.clear();
470 _ghost_value_data.shrink_to_fit();
471
472 // Add to local rows
473 int bs2 = _bs[0] * _bs[1];
474 assert(_ghost_value_data_in.size() == _unpack_pos.size() * bs2);
475 for (std::size_t i = 0; i < _unpack_pos.size(); ++i)
476 for (int j = 0; j < bs2; ++j)
477 _data[_unpack_pos[i] * bs2 + j] += _ghost_value_data_in[i * bs2 + j];
478
479 _ghost_value_data_in.clear();
480 _ghost_value_data_in.shrink_to_fit();
481
482 // Set ghost row data to zero
483 std::int32_t local_size0 = _index_maps[0]->size_local();
484 std::fill(std::next(_data.begin(), _row_ptr[local_size0] * bs2),
485 _data.end(), 0);
486 }

491 double squared_norm() const
492 {
493 const std::size_t num_owned_rows = _index_maps[0]->size_local();
494 const int bs2 = _bs[0] * _bs[1];
495 assert(num_owned_rows < _row_ptr.size());
496 double norm_sq_local = std::accumulate(
497 _data.cbegin(),
498 std::next(_data.cbegin(), _row_ptr[num_owned_rows] * bs2), double(0),
499 [](auto norm, value_type y) { return norm + std::norm(y); });
500 double norm_sq;
501 MPI_Allreduce(&norm_sq_local, &norm_sq, 1, MPI_DOUBLE, MPI_SUM,
502 _comm.comm());
503 return norm_sq;
504 }

620 void eliminate_zeros(value_type tol = 0)
621 {
622 // Remove any zero entries (blocks, where all entries in the block
623 // are within tolerance of zero) in data, and update the column
624 // indices and row pointers accordingly.
625 const std::size_t bs2 = _bs[0] * _bs[1];
626
627 // True if every entry of the block starting at block index j is
628 // within tolerance of zero, i.e. the whole block can be dropped.
629 auto is_zero_block = [this, bs2, tol](std::int64_t j)
630 {
631 return std::all_of(std::next(_data.begin(), j * bs2),
632 std::next(_data.begin(), (j + 1) * bs2),
633 [tol](value_type x)
634 { return std::abs(x) <= std::abs(tol); });
635 };
636
637 std::int64_t ptr_out = 0;
638 std::vector<std::int64_t> new_row_ptr = {0};
639 std::vector<std::int64_t> new_off_diagonal_offset;
640 new_row_ptr.reserve(_row_ptr.size());
641 new_off_diagonal_offset.reserve(_off_diagonal_offset.size());
642 for (std::size_t i = 0; i < _row_ptr.size() - 1; ++i)
643 {
644 for (std::int64_t j = _row_ptr[i]; j < _off_diagonal_offset[i]; ++j)
645 {
646 if (!is_zero_block(j))
647 {
648 _cols[ptr_out] = _cols[j];
649 std::copy_n(std::next(_data.begin(), j * bs2), bs2,
650 std::next(_data.begin(), ptr_out * bs2));
651 ++ptr_out;
652 }
653 }
654 new_off_diagonal_offset.push_back(ptr_out);
655 for (std::int64_t j = _off_diagonal_offset[i]; j < _row_ptr[i + 1]; ++j)
656 {
657 if (!is_zero_block(j))
658 {
659 _cols[ptr_out] = _cols[j];
660 std::copy_n(std::next(_data.begin(), j * bs2), bs2,
661 std::next(_data.begin(), ptr_out * bs2));
662 ++ptr_out;
663 }
664 }
665 new_row_ptr.push_back(ptr_out);
666 }
667 _data.resize(ptr_out * bs2);
668 _cols.resize(ptr_out);
669 _row_ptr = new_row_ptr;
670 _off_diagonal_offset = new_off_diagonal_offset;
671 _finalized = true;
672 }

674private:
675 // Parallel distribution of the rows and columns
676 std::array<std::shared_ptr<const common::IndexMap>, 2> _index_maps;
677
678 // Block mode (compact or expanded)
679 BlockMode _block_mode;
680
681 // Block sizes
682 std::array<int, 2> _bs;
683
684 // Matrix data
685 container_type _data;
686 column_container_type _cols;
687 rowptr_container_type _row_ptr;
688
689 // Start of off-diagonal (unowned columns) on each row
690 rowptr_container_type _off_diagonal_offset;
691
692 // Communicator with neighborhood (ghost->owner communicator for rows)
693 dolfinx::MPI::Comm _comm;
694
695 // -- Precomputed data for scatter_rev/update
696
697 // Request in non-blocking communication
698 MPI_Request _request;
699
700 // Position in _data to add received data
701 std::vector<std::size_t> _unpack_pos;
702
703 // Displacements for alltoall for each neighbor when sending and
704 // receiving
705 std::vector<int> _val_send_disp, _val_recv_disp;
706
707 // Ownership of each row, by neighbor (for the neighbourhood defined
708 // on _comm)
709 std::vector<int> _ghost_row_to_rank;
710
711 // Temporary stores for data during non-blocking communication
712 container_type _ghost_value_data;
713 container_type _ghost_value_data_in;
714
715 // Set by eliminate_zeros(). Once true, the sparsity may have been
716 // reduced and the precomputed scatter_rev communication pattern
717 // (_unpack_pos, _val_send_disp, _val_recv_disp) is no longer valid,
718 // so further modification of the matrix is disallowed.
719 bool _finalized = false;
720
721 // Throw if the matrix has been finalized by eliminate_zeros().
722 void check_not_finalized() const
723 {
724 if (_finalized)
725 {
726 throw std::runtime_error(
727 "MatrixCSR has been finalized by eliminate_zeros() and can no "
728 "longer be modified or scattered.");
729 }
730 }

735template <typename U, typename V, typename W, typename X>
736template <SparsityImplementation SparsityType>
737MatrixCSR<U, V, W, X>::MatrixCSR(const SparsityType& p, BlockMode mode)
738 : _index_maps({p.index_map(0), p.index_map(1)}), _block_mode(mode),
739 _bs({p.block_size(0), p.block_size(1)}),
740 _data(p.graph().first.size() * _bs[0] * _bs[1], 0),
741 _cols(p.graph().first.begin(), p.graph().first.end()),
742 _row_ptr(p.graph().second.begin(), p.graph().second.end()),
743 _comm(MPI_COMM_NULL)
744{
745 if (_block_mode == BlockMode::expanded)
746 {
747 // Rebuild IndexMaps
748 for (int i = 0; i < 2; ++i)
749 {
750 auto im = _index_maps[i];
751 std::int32_t size_local = im->size_local() * _bs[i];
752 std::span ghost_i = im->ghosts();
753 std::vector<std::int64_t> ghosts;
754 const std::vector<int> ghost_owner_i(im->owners().begin(),
755 im->owners().end());
756 std::vector<int> src_rank;
757 for (std::size_t j = 0; j < ghost_i.size(); ++j)
758 {
759 for (int k = 0; k < _bs[i]; ++k)
760 {
761 ghosts.push_back(ghost_i[j] * _bs[i] + k);
762 src_rank.push_back(ghost_owner_i[j]);
763 }
764 }
765
766 std::array<std::vector<int>, 2> src_dest0
767 = {std::vector(_index_maps[i]->src().begin(),
768 _index_maps[i]->src().end()),
769 std::vector(_index_maps[i]->dest().begin(),
770 _index_maps[i]->dest().end())};
771 _index_maps[i] = std::make_shared<common::IndexMap>(
772 _index_maps[i]->comm(), size_local, src_dest0, ghosts, src_rank);
773 }
774
775 // Convert sparsity pattern and set _bs to 1
776
777 column_container_type new_cols;
778 new_cols.reserve(_data.size());
779 rowptr_container_type new_row_ptr{0};
780 new_row_ptr.reserve(_row_ptr.size() * _bs[0]);
781 std::span<const std::int32_t> num_diag_nnz = p.off_diagonal_offsets();
782 for (std::size_t i = 0; i < _row_ptr.size() - 1; ++i)
783 {
784 // Repeat row _bs[0] times
785 for (int q0 = 0; q0 < _bs[0]; ++q0)
786 {
787 _off_diagonal_offset.push_back(new_row_ptr.back()
788 + num_diag_nnz[i] * _bs[1]);
789 for (auto j = _row_ptr[i]; j < _row_ptr[i + 1]; ++j)
790 {
791 for (int q1 = 0; q1 < _bs[1]; ++q1)
792 new_cols.push_back(_cols[j] * _bs[1] + q1);
793 }
794 new_row_ptr.push_back(new_cols.size());
795 }
796 }
797 _cols = new_cols;
798 _row_ptr = new_row_ptr;
799 _bs[0] = 1;
800 _bs[1] = 1;
801 }
802 else
803 {
804 // Compute off-diagonal offset for each row (compact)
805 std::span<const std::int32_t> num_diag_nnz = p.off_diagonal_offsets();
806 _off_diagonal_offset.reserve(num_diag_nnz.size());
807 std::ranges::transform(num_diag_nnz, _row_ptr,
808 std::back_inserter(_off_diagonal_offset),
809 std::plus{});
810 }
811
812 // Some short-hand
813 std::array local_size
814 = {_index_maps[0]->size_local(), _index_maps[1]->size_local()};
815 std::array local_range
816 = {_index_maps[0]->local_range(), _index_maps[1]->local_range()};
817 std::span ghosts1 = _index_maps[1]->ghosts();
818
819 std::span ghosts0 = _index_maps[0]->ghosts();
820 std::span src_ranks = _index_maps[0]->src();
821 std::span dest_ranks = _index_maps[0]->dest();
822
823 // Create neighbourhood communicator (owner <- ghost)
824 MPI_Comm comm;
825 MPI_Dist_graph_create_adjacent(_index_maps[0]->comm(), dest_ranks.size(),
826 dest_ranks.data(), MPI_UNWEIGHTED,
827 src_ranks.size(), src_ranks.data(),
828 MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm);
829 _comm = dolfinx::MPI::Comm(comm, false);
830
831 // Build map from ghost row index position to owning (neighborhood)
832 // rank
833 _ghost_row_to_rank.reserve(_index_maps[0]->owners().size());
834 for (int r : _index_maps[0]->owners())
835 {
836 auto it = std::ranges::lower_bound(src_ranks, r);
837 assert(it != src_ranks.end() and *it == r);
838 std::size_t pos = std::ranges::distance(src_ranks.begin(), it);
839 _ghost_row_to_rank.push_back(pos);
840 }
841
842 // Compute size of data to send to each neighbor
843 std::vector<std::int32_t> data_per_proc(src_ranks.size(), 0);
844 for (std::size_t i = 0; i < _ghost_row_to_rank.size(); ++i)
845 {
846 assert(_ghost_row_to_rank[i] < (int)data_per_proc.size());
847 std::size_t pos = local_size[0] + i;
848 data_per_proc[_ghost_row_to_rank[i]] += _row_ptr[pos + 1] - _row_ptr[pos];
849 }
850
851 // Compute send displacements
852 _val_send_disp.resize(src_ranks.size() + 1, 0);
853 std::partial_sum(data_per_proc.begin(), data_per_proc.end(),
854 std::next(_val_send_disp.begin()));
855
856 // For each ghost row, pack and send indices to neighborhood
857 std::vector<std::int64_t> ghost_index_data(2 * _val_send_disp.back());
858 {
859 std::vector<int> insert_pos = _val_send_disp;
860 for (std::size_t i = 0; i < _ghost_row_to_rank.size(); ++i)
861 {
862 int rank = _ghost_row_to_rank[i];
863 std::int32_t row_id = local_size[0] + i;
864 for (int j = _row_ptr[row_id]; j < _row_ptr[row_id + 1]; ++j)
865 {
866 // Get position in send buffer
867 std::int32_t idx_pos = 2 * insert_pos[rank];
868
869 // Pack send data (row, col) as global indices
870 ghost_index_data[idx_pos] = ghosts0[i];
871 if (std::int32_t col_local = _cols[j]; col_local < local_size[1])
872 ghost_index_data[idx_pos + 1] = col_local + local_range[1][0];
873 else
874 ghost_index_data[idx_pos + 1] = ghosts1[col_local - local_size[1]];
875
876 insert_pos[rank] += 1;
877 }
878 }
879 }
880
881 // Communicate data with neighborhood
882 std::vector<std::int64_t> ghost_index_array;
883 std::vector<int> recv_disp;
884 {
885 std::vector<int> send_sizes;
886 std::ranges::transform(data_per_proc, std::back_inserter(send_sizes),
887 [](auto x) { return 2 * x; });
888
889 std::vector<int> recv_sizes(dest_ranks.size());
890 send_sizes.reserve(1);
891 recv_sizes.reserve(1);
892 MPI_Neighbor_alltoall(send_sizes.data(), 1, MPI_INT, recv_sizes.data(), 1,
893 MPI_INT, _comm.comm());
894
895 // Build send/recv displacement
896 std::vector<int> send_disp{0};
897 std::partial_sum(send_sizes.begin(), send_sizes.end(),
898 std::back_inserter(send_disp));
899 recv_disp = {0};
900 std::partial_sum(recv_sizes.begin(), recv_sizes.end(),
901 std::back_inserter(recv_disp));
902
903 ghost_index_array.resize(recv_disp.back());
904 MPI_Neighbor_alltoallv(ghost_index_data.data(), send_sizes.data(),
905 send_disp.data(), MPI_INT64_T,
906 ghost_index_array.data(), recv_sizes.data(),
907 recv_disp.data(), MPI_INT64_T, _comm.comm());
908 }
909
910 // Store receive displacements for future use, when transferring
911 // data values
912 _val_recv_disp.resize(recv_disp.size());
913 int bs2 = _bs[0] * _bs[1];
914 std::ranges::transform(recv_disp, _val_recv_disp.begin(),
915 [&bs2](auto d) { return bs2 * d / 2; });
916 std::ranges::transform(_val_send_disp, _val_send_disp.begin(),
917 [&bs2](auto d) { return d * bs2; });
918
919 // Global-to-local map for ghost columns
920 std::vector<std::pair<std::int64_t, std::int32_t>> global_to_local;
921 global_to_local.reserve(ghosts1.size());
922 for (std::int64_t idx : ghosts1)
923 global_to_local.push_back({idx, global_to_local.size() + local_size[1]});
924 std::ranges::sort(global_to_local);
925
926 // Compute location in which data for each index should be stored
927 // when received
928 for (std::size_t i = 0; i < ghost_index_array.size(); i += 2)
929 {
930 // Row must be on this process
931 std::int32_t local_row = ghost_index_array[i] - local_range[0][0];
932 assert(local_row >= 0 and local_row < local_size[0]);
933
934 // Column may be owned or unowned
935 std::int32_t local_col = ghost_index_array[i + 1] - local_range[1][0];
936 if (local_col < 0 or local_col >= local_size[1])
937 {
938 auto it = std::ranges::lower_bound(
939 global_to_local, std::pair(ghost_index_array[i + 1], -1),
940 [](auto a, auto b) { return a.first < b.first; });
941 assert(it != global_to_local.end()
942 and it->first == ghost_index_array[i + 1]);
943 local_col = it->second;
944 }
945 auto cit0 = std::next(_cols.begin(), _row_ptr[local_row]);
946 auto cit1 = std::next(_cols.begin(), _row_ptr[local_row + 1]);
947
948 // Find position of column index and insert data
949 auto cit = std::lower_bound(cit0, cit1, local_col);
950 assert(cit != cit1);
951 assert(*cit == local_col);
952 std::size_t d = std::ranges::distance(_cols.begin(), cit);
953 _unpack_pos.push_back(d);
954 }
955
956 _unpack_pos.shrink_to_fit();
957}

959
960// The matrix A is distributed across P processes by blocks of rows:
961// A = | A_0 |
962// | A_1 |
963// | ... |
964// | A_P-1 |
965//
966// Each submatrix A_i is owned by a single process "i" and can be further
967// decomposed into diagonal (Ai[0]) and off diagonal (Ai[1]) blocks:
968// Ai = |Ai[0] Ai[1]|
969//
970// If A is square, the diagonal block Ai[0] is also square and contains
971// only owned columns and rows. The block Ai[1] contains ghost columns
972// (unowned dofs).
973
974// Likewise, a local vector x can be decomposed into owned and ghost blocks:
975// xi = | x[0] |
976// | x[1] |
977//
978// So the product y = Ax can be computed into two separate steps:
979// y[0] = |Ai[0] Ai[1]| | x[0] | = Ai[0] x[0] + Ai[1] x[1]
980// | x[1] |
981//
984template <typename Scalar, typename V, typename W, typename X>
985void MatrixCSR<Scalar, V, W, X>::mult(la::Vector<Scalar>& x,
986 la::Vector<Scalar>& y) const
987{
988 // start communication (update ghosts)
989 x.scatter_fwd_begin();
990
991 std::int32_t nrowslocal = num_owned_rows();
992 std::span<const std::int64_t> Arow_ptr(row_ptr().data(), nrowslocal + 1);
993 std::span<const std::int32_t> Acols(cols().data(), Arow_ptr[nrowslocal]);
994 std::span<const std::int64_t> Aoff_diag_offset(off_diag_offset().data(),
995 nrowslocal);
996 std::span<const Scalar> Avalues(values().data(),
997 Arow_ptr[nrowslocal] * _bs[0] * _bs[1]);
998
999 std::span<const Scalar> _x = x.array();
1000 std::span<Scalar> _y = y.array();
1001
1002 std::span<const std::int64_t> Arow_begin(Arow_ptr.data(), nrowslocal);
1003 std::span<const std::int64_t> Arow_end(Arow_ptr.data() + 1, nrowslocal);
1004
1005 // First stage: spmv - diagonal
1006 // yi[0] += Ai[0] * xi[0]
1007 if (_bs[1] == 1)
1008 {
1009 impl::spmv<Scalar>(Avalues, Arow_begin, Aoff_diag_offset, Acols, _x, _y,
1010 _bs[0], std::integral_constant<int, 1>{});
1011 }
1012 else if (_bs[1] == 2)
1013 {
1014 impl::spmv<Scalar>(Avalues, Arow_begin, Aoff_diag_offset, Acols, _x, _y,
1015 _bs[0], std::integral_constant<int, 2>{});
1016 }
1017 else if (_bs[1] == 3)
1018 {
1019 impl::spmv<Scalar>(Avalues, Arow_begin, Aoff_diag_offset, Acols, _x, _y,
1020 _bs[0], std::integral_constant<int, 3>{});
1021 }
1022 else
1023 {
1024 impl::spmv<Scalar>(Avalues, Arow_begin, Aoff_diag_offset, Acols, _x, _y,
1025 _bs[0], _bs[1]);
1026 }
1027
1028 // finalize ghost update
1029 x.scatter_fwd_end();
1030
1031 // Second stage: spmv - off-diagonal
1032 // yi[0] += Ai[1] * xi[1]
1033 if (_bs[1] == 1)
1034 {
1035 impl::spmv<Scalar>(Avalues, Aoff_diag_offset, Arow_end, Acols, _x, _y,
1036 _bs[0], std::integral_constant<int, 1>{});
1037 }
1038 else if (_bs[1] == 2)
1039 {
1040 impl::spmv<Scalar>(Avalues, Aoff_diag_offset, Arow_end, Acols, _x, _y,
1041 _bs[0], std::integral_constant<int, 2>{});
1042 }
1043 else if (_bs[1] == 3)
1044 {
1045 impl::spmv<Scalar>(Avalues, Aoff_diag_offset, Arow_end, Acols, _x, _y,
1046 _bs[0], std::integral_constant<int, 3>{});
1047 }
1048 else
1049 {
1050 impl::spmv<Scalar>(Avalues, Aoff_diag_offset, Arow_end, Acols, _x, _y,
1051 _bs[0], _bs[1]);
1052 }
1053}
