33template <typename F>
34constexpr bool is_transform_set(const F& fn)
35{
36 // A reference to a function is never null, and converting one to
37 // bool warns under -Waddress rather than answering the question.
38 if constexpr (std::is_function_v<F>)

299template <class XD, std::floating_point U>
300void gather_cell_coordinates(GeometryPack<XD, U> geometry, std::int32_t cell,
301 U* cdofs)
302{
303 const auto x_dofmap = geometry.dofmap;
304 const std::size_t ndofs_x = x_dofmap.extent(1);
305 const std::int32_t* xdofs
306 = x_dofmap.data_handle() + static_cast<std::ptrdiff_t>(cell) * ndofs_x;
307 const U* x = geometry.x.data_handle();
308 for (std::size_t i = 0; i < ndofs_x; ++i)
309 {
310 const U* src = x + static_cast<std::ptrdiff_t>(xdofs[i]) * 3;
311 for (std::size_t k = 0; k < 3; ++k)
312 cdofs[3 * i + k] = src[k];
313 }
314}

326template <class F>
327void dispatch_bs(int bs, F&& f)
328{
329 switch (bs)
330 {
331 case 1:
332 return f(std::integral_constant<int, 1>{});
333 case 3:
334 return f(std::integral_constant<int, 3>{});
335 default:
336 return f(bs);
337 }
338}

351template <class F>
352void dispatch_bs(int bs0, int bs1, F&& f)
353{
354 if (bs0 == bs1)
355 {
356 switch (bs0)
357 {
358 case 1:
359 return f(std::integral_constant<int, 1>{},
360 std::integral_constant<int, 1>{});
361 case 3:
362 return f(std::integral_constant<int, 3>{},
363 std::integral_constant<int, 3>{});
364 }
365 }
366
367 return f(bs0, bs1);
368}
