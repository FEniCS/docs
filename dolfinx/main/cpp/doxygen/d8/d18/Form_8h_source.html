57constexpr int integral_entity_dim(IntegralType type, int tdim)
58{
59 switch (type)
60 {
61 case IntegralType::exterior_facet:
62 case IntegralType::interior_facet:
63 return tdim - 1;
64 case IntegralType::ridge:
65 return tdim - 2;
66 case IntegralType::vertex:
67 return 0;
68 case IntegralType::cell:
69 return tdim;
70 }
71
72 throw std::invalid_argument("Unknown integral type.");
73}

90inline md::mdspan<const std::uint8_t, md::dextents<std::size_t, 2>>
91entity_permutations(mesh::Topology& topology, IntegralType type,
92 mesh::CellType cell_type)
93{
94 if (type == IntegralType::cell)
95 return {};
96
97 const int tdim = topology.dim();
98 const int dim = integral_entity_dim(type, tdim);
99
100 topology.create_entity_permutations(dim);
101 const std::vector<std::uint8_t>& p = topology.get_entity_permutations(dim);
102 const int num_entities_per_cell = mesh::cell_num_entities(cell_type, dim);
103 return md::mdspan(p.data(), p.size() / num_entities_per_cell,
104 num_entities_per_cell);
105}

210 template <typename X>
211 requires std::is_convertible_v<
212 std::remove_cvref_t<X>,
213 std::map<std::tuple<IntegralType, int, int>,
214 integral_data<scalar_type, geometry_type>>>
215 Form(
216 const std::vector<std::shared_ptr<const FunctionSpace<geometry_type>>>& V,
217 X&& integrals, std::shared_ptr<const mesh::Mesh<geometry_type>> mesh,
218 const std::vector<
219 std::shared_ptr<const Function<scalar_type, geometry_type>>>&
220 coefficients,
221 const std::vector<std::shared_ptr<const Constant<scalar_type>>>&
222 constants,
223 bool needs_facet_permutations,
224 const std::vector<std::reference_wrapper<const mesh::EntityMap>>&
225 entity_maps)
226 : _function_spaces(V), _integrals(std::forward<X>(integrals)),
227 _mesh(mesh), _coefficients(coefficients), _constants(constants),
228 _needs_facet_permutations(needs_facet_permutations)
229 {
230 if (!_mesh)
231 throw std::invalid_argument("Form Mesh is null.");
232
233 // `_mesh` is fixed for the remainder of construction, so its
234 // topology and dimension are fetched once and reused below rather
235 // than being re-fetched for every integral/coefficient.
236 const mesh::Topology& topology = *_mesh->topology();
237 const int tdim = topology.dim();
238
239 // A helper function to find the correct entity map for a given mesh
240 auto get_entity_map
241 = [mesh, &entity_maps](auto& mesh0) -> const mesh::EntityMap&
242 {
243 auto it = std::ranges::find_if(
244 entity_maps,
245 [mesh, mesh0](const mesh::EntityMap& em)
246 {
247 return ((em.topology() == mesh0->topology()
248 and em.sub_topology() == mesh->topology()))
249 or ((em.sub_topology() == mesh0->topology()
250 and em.topology() == mesh->topology()));
251 });
252
253 if (it == entity_maps.end())
254 {
255 throw std::invalid_argument(
256 "Incompatible mesh. argument entity_maps must be provided.");
257 }
258 return *it;
259 };
260
261 // A helper function to compute the (cell, local_entity) pairs in the
262 // argument/coefficient domain from the (cell, local_entity) pairs in
263 // `this->mesh()`.
264 auto compute_entity_domains
265 = [](const auto& int_ents_mesh, int codim, const auto& c_to_e,
266 const auto& emap, bool inverse)
267 {
268 // TODO: This function would be much neater using
269 // `std::views::stride(2)` from C++ 23
270
271 // Get a list of entities to map to the argument/coefficient
272 // domain
273 std::vector<std::int32_t> entities;
274 entities.reserve(int_ents_mesh.size() / 2);
275 if (codim == 0)
276 {
277 // In the codim 0 case, we need to map from cells in
278 // `this->mesh()` to cells in the argument/coefficient mesh, so
279 // here we extract the cells.
280 for (std::size_t i = 0; i < int_ents_mesh.size(); i += 2)
281 entities.push_back(int_ents_mesh[i]);
282 }
283 else
284 {
285 // Otherwise the integration entities are sub-entities of the
286 // cells of `this->mesh()` and are themselves cells of the
287 // argument/coefficient mesh, so here we extract the entity
288 // index using the cell-to-entity connectivity.
289 for (std::size_t i = 0; i < int_ents_mesh.size(); i += 2)
290 {
291 entities.push_back(
292 c_to_e->links(int_ents_mesh[i])[int_ents_mesh[i + 1]]);
293 }
294 }
295
296 // Map from entity indices in `this->mesh()` to the corresponding
297 // cell indices in the argument/coefficient mesh
298 std::vector<std::int32_t> cells_mesh0
299 = emap.sub_topology_to_topology(entities, inverse);
300
301 // Create a list of (cell, local_entity_index) pairs in the
302 // argument/coefficient domain. Only the cell column is meaningful.
303 // For codim > 0 the entity is itself the cell, so it has
304 // no local index. The colummn is never used later on, but written
305 // like this for consistency for packing/assembly.
306 std::vector<std::int32_t> e = int_ents_mesh;
307 for (std::size_t i = 0; i < cells_mesh0.size(); ++i)
308 e[2 * i] = cells_mesh0[i];
309
310 return e;
311 };
312
313 // Map the integration entities of one integral to the
314 // argument/coefficient domain, checking first that the mapping is
315 // expressible: `compute_entity_domains` maps an integration entity
316 // to a *cell* of that mesh, so unless the meshes have equal
317 // dimension the integral's entity dimension must be that mesh's.
318 auto map_entities = [tdim, &topology, &compute_entity_domains](
319 IntegralType type, const auto& entities,
320 const mesh::Mesh<geometry_type>& mesh0,
321 const mesh::EntityMap& emap,
322 bool inverse) -> std::vector<std::int32_t>
323 {
324 if (type == IntegralType::cell)
325 return emap.sub_topology_to_topology(entities, inverse);
326
327 if (type == IntegralType::vertex)
328 {
329 throw std::invalid_argument(
330 "Vertex integrals are not supported for a form with an argument "
331 "or coefficient on another mesh. Supported types are cell, "
332 "exterior facet, interior facet and ridge.");
333 }
334
335 const int dim0 = mesh0.topology()->dim();
336 const int codim = tdim - dim0;
337 const int edim = integral_entity_dim(type, tdim);
338 assert(codim >= 0);
339 if (codim > 0 and edim != dim0)
340 {
341 throw std::invalid_argument(std::format(
342 "Cannot map integration entities of dimension {} to cells of a "
343 "mesh of dimension {}. An argument or coefficient on another mesh "
344 "must live on the entities being integrated over.",
345 edim, dim0));
346 }
347
348 std::shared_ptr<const graph::AdjacencyList<std::int32_t>> c_to_e
349 = topology.connectivity(tdim, edim);
350 assert(codim == 0 or c_to_e);
351 return compute_entity_domains(entities, codim, c_to_e, emap, inverse);
352 };
353
354 _edata.reserve(_function_spaces.size());
355 for (auto& space : _function_spaces)
356 {
357 // Working map: [integral type, integral_idx, kernel_idx]->entities
358 std::map<std::tuple<IntegralType, int, int>,
359 std::variant<std::vector<std::int32_t>,
360 std::span<const std::int32_t>>>
361 vdata;
362
363 if (auto mesh0 = space->mesh(); mesh0 == _mesh)
364 {
365 for (auto& [key, integral] : _integrals)
366 vdata.insert({key, std::span(integral.entities)});
367 }
368 else
369 {
370 // Find correct entity map
371 const mesh::EntityMap& emap = get_entity_map(mesh0);
372
373 // Determine direction of the map. We need to map from
374 // `this->mesh()` to `mesh0`, so if `emap->sub_topology()` isn't
375 // the source topology, we need the inverse map
376 bool inverse = emap.sub_topology() == mesh0->topology();
377 for (auto& [key, itg] : _integrals)
378 {
379 auto [type, idx, kernel_idx] = key;
380 std::vector<std::int32_t> e;
381 assert(mesh0);
382 e = map_entities(type, itg.entities, *mesh0, emap, inverse);
383
384 vdata.insert({key, std::move(e)});
385 }
386 }
387
388 _edata.push_back(std::move(vdata));
389 }
390
391 for (auto& [key, integral] : _integrals)
392 {
393 auto [type, idx, kernel_idx] = key;
394 for (int c : integral.coeffs)
395 {
396 if (auto mesh0 = coefficients.at(c)->function_space()->mesh();
397 mesh0 == _mesh)
398 {
399 _cdata.insert({{type, idx, c}, std::span(integral.entities)});
400 }
401 else
402 {
403 // Find correct entity map and determine direction of the map
404 const mesh::EntityMap& emap = get_entity_map(mesh0);
405 bool inverse = emap.sub_topology() == mesh0->topology();
406
407 std::vector<std::int32_t> e;
408 assert(mesh0);
409 e = map_entities(type, integral.entities, *mesh0, emap, inverse);
410 _cdata.insert({{type, idx, c}, std::move(e)});
411 }
412 }
413 }
414 }

729 // Mapped domain index data for argument functions.
730 //
731 // Consider:
732 //
733 // entities = this->domain(type, idx, kernel_idx);
734 // entities0 = _edata[0][{type, idx, kernel_idx}];
735 //
736 // Then `entities[i]` is a mesh entity index (e.g., cell index) in
737 // `_mesh`, and `entities0[i]` is the index of the same entity but in
738 // the mesh associated with the argument 0 (test function) space.
739 std::vector<std::map<
740 std::tuple<IntegralType, int, int>,
741 std::variant<std::vector<std::int32_t>, std::span<const std::int32_t>>>>
742 _edata;
743
744 // Mapped domain index data for coefficient functions.
745 //
746 // Consider:
747 //
748 // entities = this->domain(type, idx, kernel_idx);
749 // entities0 = _cdata[{type, idx, c}];
750 //
751 // where `c` is the coefficient index.
752 //
753 // Then `entities[i]` is a mesh entity index (e.g., cell index) in
754 // `_mesh`, and `entities0[i]` is the index of the same entity but in
755 // the mesh associated with the coefficient Function.
756 std::map<
757 std::tuple<IntegralType, int, int>,
758 std::variant<std::vector<std::int32_t>, std::span<const std::int32_t>>>
759 _cdata;
