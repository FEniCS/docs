55constexpr int integral_entity_dim(IntegralType type, int tdim)
56{
57 switch (type)
58 {
59 case IntegralType::exterior_facet:
60 case IntegralType::interior_facet:
61 return tdim - 1;
62 case IntegralType::ridge:
63 return tdim - 2;
64 case IntegralType::vertex:
65 return 0;
66 case IntegralType::cell:
67 return tdim;
68 }
69
70 throw std::invalid_argument("Unknown integral type.");
71}

175 template <typename X>
176 requires std::is_convertible_v<
177 std::remove_cvref_t<X>,
178 std::map<std::tuple<IntegralType, int, int>,
179 integral_data<scalar_type, geometry_type>>>
180 Form(
181 const std::vector<std::shared_ptr<const FunctionSpace<geometry_type>>>& V,
182 X&& integrals, std::shared_ptr<const mesh::Mesh<geometry_type>> mesh,
183 const std::vector<
184 std::shared_ptr<const Function<scalar_type, geometry_type>>>&
185 coefficients,
186 const std::vector<std::shared_ptr<const Constant<scalar_type>>>&
187 constants,
188 bool needs_facet_permutations,
189 const std::vector<std::reference_wrapper<const mesh::EntityMap>>&
190 entity_maps)
191 : _function_spaces(V), _integrals(std::forward<X>(integrals)),
192 _mesh(std::move(mesh)), _coefficients(coefficients),
193 _constants(constants),
194 _needs_facet_permutations(needs_facet_permutations)
195 {
196 if (!_mesh)
197 throw std::invalid_argument("Form Mesh is null.");
198
199 // `_mesh` is fixed for the remainder of construction, so its
200 // topology and dimension are fetched once and reused below rather
201 // than being re-fetched for every integral/coefficient.
202 const mesh::Topology& topology = *_mesh->topology();
203 const int tdim = topology.dim();
204
205 // Map the integration entities of one integral to the
206 // argument/coefficient domain, checking first that the mapping is
207 // expressible
208 auto map_entities
209 = [tdim,
210 &topology](IntegralType type, std::span<const std::int32_t> entities,
211 const mesh::Topology& topology0,
212 const mesh::EntityMap& emap) -> std::vector<std::int32_t>
213 {
214 if (type == IntegralType::cell)
215 {
216 return mesh::extract_cells_from_entities(
217 topology0, topology, md::mdspan(entities.data(), entities.size()),
218 std::cref(emap));
219 }
220
221 if (type == IntegralType::vertex)
222 {
223 throw std::invalid_argument(
224 "Vertex integrals are not supported for a form with an argument "
225 "or coefficient on another mesh. Supported types are cell, "
226 "exterior facet, interior facet and ridge.");
227 }
228
229 mesh::check_entity_cell_mapping(topology0, topology,
230 integral_entity_dim(type, tdim));
231
232 // Map the (cell, local_entity) pairs, flattened (interior facets
233 // hold two pairs per entity), to cells of the argument/coefficient
234 // mesh
235 std::vector<std::int32_t> cells = mesh::extract_cells_from_entities(
236 topology0, topology,
237 md::mdspan<const std::int32_t,
238 md::extents<std::size_t, md::dynamic_extent, 2>>(
239 entities.data(), entities.size() / 2, 2),
240 std::cref(emap));
241
242 // Replace the cell of each pair. Only the cell column is
243 // meaningful: for codim > 0 the entity is itself the cell, so it
244 // has no local index. The local index column is kept for the
245 // layout used in packing/assembly.
246 std::vector<std::int32_t> e(entities.begin(), entities.end());
247 for (std::size_t i = 0; i < cells.size(); ++i)
248 e[2 * i] = cells[i];
249 return e;
250 };
251
252 _edata.reserve(_function_spaces.size());
253 for (auto& space : _function_spaces)
254 {
255 // Working map: [integral type, integral_idx, kernel_idx]->entities
256 std::map<std::tuple<IntegralType, int, int>,
257 std::variant<std::vector<std::int32_t>,
258 std::span<const std::int32_t>>>
259 vdata;
260
261 if (auto mesh0 = space->mesh(); mesh0 == _mesh)
262 {
263 for (auto& [key, integral] : _integrals)
264 vdata.insert({key, std::span(integral.entities)});
265 }
266 else
267 {
268 // Find correct entity map
269 const mesh::Topology& topology0 = *mesh0->topology();
270 const mesh::EntityMap& emap
271 = mesh::find_entity_map(entity_maps, topology, topology0);
272 for (auto& [key, itg] : _integrals)
273 {
274 auto [type, idx, kernel_idx] = key;
275 vdata.insert(
276 {key, map_entities(type, itg.entities, topology0, emap)});
277 }
278 }
279
280 _edata.push_back(std::move(vdata));
281 }
282
283 for (auto& [key, integral] : _integrals)
284 {
285 auto [type, idx, kernel_idx] = key;
286 for (int c : integral.coeffs)
287 {
288 if (auto mesh0 = coefficients.at(c)->function_space()->mesh();
289 mesh0 == _mesh)
290 {
291 _cdata.insert({{type, idx, c}, std::span(integral.entities)});
292 }
293 else
294 {
295 // Find correct entity map
296 const mesh::Topology& topology0 = *mesh0->topology();
297 const mesh::EntityMap& emap
298 = mesh::find_entity_map(entity_maps, topology, topology0);
299 _cdata.insert(
300 {{type, idx, c},
301 map_entities(type, integral.entities, topology0, emap)});
302 }
303 }
304 }
305 }

620 // Mapped domain index data for argument functions.
621 //
622 // Consider:
623 //
624 // entities = this->domain(type, idx, kernel_idx);
625 // entities0 = _edata[0][{type, idx, kernel_idx}];
626 //
627 // Then `entities[i]` is a mesh entity index (e.g., cell index) in
628 // `_mesh`, and `entities0[i]` is the index of the same entity but in
629 // the mesh associated with the argument 0 (test function) space.
630 std::vector<std::map<
631 std::tuple<IntegralType, int, int>,
632 std::variant<std::vector<std::int32_t>, std::span<const std::int32_t>>>>
633 _edata;
634
635 // Mapped domain index data for coefficient functions.
636 //
637 // Consider:
638 //
639 // entities = this->domain(type, idx, kernel_idx);
640 // entities0 = _cdata[{type, idx, c}];
641 //
642 // where `c` is the coefficient index.
643 //
644 // Then `entities[i]` is a mesh entity index (e.g., cell index) in
645 // `_mesh`, and `entities0[i]` is the index of the same entity but in
646 // the mesh associated with the coefficient Function.
647 std::map<
648 std::tuple<IntegralType, int, int>,
649 std::variant<std::vector<std::int32_t>, std::span<const std::int32_t>>>
650 _cdata;
