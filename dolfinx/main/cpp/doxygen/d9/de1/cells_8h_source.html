213vtk_to_dolfinx(std::int8_t vtk_cell_type)
214{
215 {
216 // For a complete overview of VTK cell types, see
217 // https://vtk.org/doc/nightly/html/vtkCellType_8h_source.html
218 switch (vtk_cell_type)
219 {
220 using enum mesh::CellType;
221 case 1:
222 return {point, -1};
223 case 3:
224 return {interval, 1};
225 case 5:
226 return {triangle, 1};
227 case 9:
228 return {quadrilateral, 1};
229 case 10:
230 return {tetrahedron, 1};
231 case 12:
232 return {hexahedron, 1};
233 case 13:
234 return {prism, 1};
235 case 14:
236 return {pyramid, 1};
237 case 21:
238 return {interval, 2};
239 case 22:
240 return {triangle, 2};
241 case 23:
242 return {quadrilateral, 2};
243 case 24:
244 return {tetrahedron, 2};
245 case 25:
246 return {hexahedron, 2};
247 case 26:
248 return {prism, 2};
249 case 27:
250 return {pyramid, 2};
251 case 35:
252 return {interval, 3};
253 case 68:
254 return {interval, -1};
255 case 69:
256 return {triangle, -1};
257 case 70:
258 return {quadrilateral, -1};
259 case 71:
260 return {tetrahedron, -1};
261 case 72:
262 return {hexahedron, -1};
263 case 73:
264 return {prism, -1};
265 case 74:
266 return {pyramid,
267 -1}; // Not implemented in VTK yet, but added as placeholder.
268 default:
269 break;
270 }
271 throw std::invalid_argument("Unknown VTK cell type");
272 }
