128 std::vector<std::int32_t> sub_topology_to_topology(CellRange auto&& entities,
129 bool inverse) const
130 {
131 std::size_t num_entities = std::ranges::size(entities);
132 if (!inverse)
133 {
134 // In this case, we want to map from entity indices in
135 // `_sub_topology` to corresponding entities in `_topology`. Hence,
136 // for each index in `entities`, we get the corresponding index in
137 // `_topology` using `_sub_topology_to_topology`
138 auto mapped
139 = std::forward<decltype(entities)>(entities)
140 | std::views::transform([this](std::int32_t i)
141 { return _sub_topology_to_topology[i]; });
142 std::vector<std::int32_t> mapped_v;
143 mapped_v.reserve(num_entities);
144 std::ranges::copy(mapped, std::back_inserter(mapped_v));
145 return mapped_v;
146 }
147 else
148 {
149 // In this case, we are mapping from entity indices in `_topology`
150 // to entity indices in `_sub_topology`. Hence, we first need to
151 // construct the "inverse" of `_sub_topology_to_topology`
152 std::unordered_map<std::int32_t, std::int32_t> topology_to_sub_topology;
153 topology_to_sub_topology.reserve(_sub_topology_to_topology.size());
154 for (std::size_t i = 0; i < _sub_topology_to_topology.size(); ++i)
155 {
156 topology_to_sub_topology.insert(
157 {_sub_topology_to_topology[i], static_cast<std::int32_t>(i)});
158 }
159
160 // For each entity index in `entities` (which are indices in
161 // `_topology`), get the corresponding entity in `_sub_topology`.
162 // Since `_sub_topology` consists of a subset of entities in
163 // `_topology`, there are entities in topology that may not exist in
164 // `_sub_topology`. If this is the case, mark those entities with
165 // -1.
166
167 auto mapped = std::forward<decltype(entities)>(entities)
168 | std::views::transform(
169 [&topology_to_sub_topology](std::int32_t i)
170 {
171 // Map the entity if it exists. If it doesn't, mark
172 // with -1.
173 auto it = topology_to_sub_topology.find(i);
174 return (it != topology_to_sub_topology.end())
175 ? it->second
176 : -1;
177 });
178 std::vector<std::int32_t> mapped_v;
179 mapped_v.reserve(num_entities);
180 std::ranges::copy(mapped, std::back_inserter(mapped_v));
181 return mapped_v;
182 }
183 }

195 std::vector<std::int32_t> _sub_topology_to_topology;
196
197 // A second topology, consisting of a subset of entities in
198 // `_topology`
199 std::shared_ptr<const Topology> _sub_topology;

248template <typename E>
249 requires(std::remove_cvref_t<E>::rank() == 1
250 or std::remove_cvref_t<E>::rank() == 2)
251 and std::same_as<typename std::remove_cvref_t<E>::value_type,
252 std::int32_t>
253std::vector<std::int32_t> extract_cells_from_entities(
254 const Topology& topology_c, const Topology& topology, E entities,
255 std::optional<std::reference_wrapper<const EntityMap>> entity_map)
256{
257 auto span_to_vector = [](auto entities)
258 {
259 assert(entities.rank() == 1);
260
261 std::vector<std::int32_t> vec;
262 vec.reserve(entities.extent(0));
263 for (std::size_t i = 0; i < entities.extent(0); ++i)
264 vec.push_back(entities[i]);
265 return vec;
266 };
267
268 if (&topology_c == &topology)
269 {
270 // If same topology no mapping is needed
271 if constexpr (entities.rank() == 1)
272 return span_to_vector(entities);
273 else
274 // If (cell, local_index) pairs are given, extract the cells
275 return span_to_vector(md::submdspan(entities, md::full_extent, 0));
276 }
277
278 if (!entity_map)
279 {
280 throw std::invalid_argument(
281 "An entity map is required when topology_c is not topology.");
282 }
283 const EntityMap& emap = entity_map->get();
284 const int tdim = topology.dim();
285 const int codim = tdim - topology_c.dim();
286 const bool inverse = emap.sub_topology().get() == &topology_c;
287 if constexpr (entities.rank() == 1)
288 {
289 // Cells map directly to cells only between equal dimensions
290 if (codim != 0)
291 {
292 throw std::invalid_argument(
293 std::format("Cannot map cells of a topology of dimension {} to "
294 "cells of a topology of dimension {}.",
295 tdim, topology_c.dim()));
296 }
297 return emap.sub_topology_to_topology(span_to_vector(entities), inverse);
298 }
299 else
300 {
301 if (codim == 0)
302 {
303 // If codim is zero we extract the cells and map them
304 auto cells = md::submdspan(entities, md::full_extent, 0);
305 return emap.sub_topology_to_topology(span_to_vector(cells), inverse);
306 }
307 else
308 {
309 // Any other codim needs to map (cell, local index) to entities and
310 // then to cells of `topology_c`
311 if (!inverse)
312 {
313 throw std::invalid_argument(
314 "Unsupported mapping. Can only map from submesh to parent mesh.");
315 }
316 assert(codim > 0);
317 std::shared_ptr<const graph::AdjacencyList<std::int32_t>> c_to_e
318 = topology.connectivity(tdim, tdim - codim);
319 if (!c_to_e)
320 {
321 throw std::runtime_error(
322 std::format("Topology connectivity from dimension {} to {} not "
323 "found.",
324 tdim, tdim - codim));
325 }
326 // Map (cell, local_index) to entity of `topology`
327 std::vector<std::int32_t> sub_entities;
328 sub_entities.reserve(entities.extent(0));
329 for (std::size_t e = 0; e < entities.extent(0); ++e)
330 sub_entities.push_back(c_to_e->links(entities(e, 0))[entities(e, 1)]);
331
332 // Map entity of `topology` to cell of `topology_c`
333 return emap.sub_topology_to_topology(sub_entities, inverse);
334 }
335 }
