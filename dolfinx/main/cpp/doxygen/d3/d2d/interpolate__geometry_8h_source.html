57template <std::floating_point T>
58mesh::Mesh<T> interpolate_geometry(
59 std::shared_ptr<mesh::Mesh<T>> mesh, const CoordinateElement<T>& new_cmap,
60 const std::function<std::vector<int>(
61 const graph::AdjacencyList<std::int32_t>&)>& reorder_fn = nullptr)
62{
63 assert(mesh);
64 const CoordinateElement<T>& old_cmap = mesh->geometry().cmaps().front();
65 if (new_cmap.cell_shape() != old_cmap.cell_shape())
66 {
67 throw std::invalid_argument(
68 "Cell shape of new coordinate element must match input mesh.");
69 }
70
71 const int gdim = mesh->geometry().dim();
72
73 // Build a vector-valued Lagrange FiniteElement from the coordinate
74 // element. It must have the same dof layout as `new_cmap`, i.e. be
75 // discontinuous if and only if `new_cmap` is.
76 basix::FiniteElement<T> b_element = basix::create_element<T>(
77 basix::element::family::P,
78 mesh::cell_type_to_basix_type(new_cmap.cell_shape()), new_cmap.degree(),
79 new_cmap.variant(), basix::element::dpc_variant::unset,
80 new_cmap.is_discontinuous());
81 auto element = std::make_shared<const FiniteElement<T>>(
82 b_element, static_cast<std::size_t>(gdim),
83 std::vector<std::size_t>{static_cast<std::size_t>(gdim)});
84
85 FunctionSpace<T> V = create_functionspace(mesh, element, reorder_fn);
86
87 // Tabulate physical coordinates of the new geometry dofs.
88 std::vector<T> x_new = V.tabulate_dof_coordinates(false);
89
90 // Pull the geometry dofmap and index map from V.
91 std::shared_ptr<const DofMap> dm = V.dofmap();
92 assert(dm);
93 std::shared_ptr<const common::IndexMap> new_imap = dm->index_map;
94 assert(new_imap);
95
96 auto map_view = dm->map();
97 std::vector<std::int32_t> dofmap_flat(
98 map_view.data_handle(), map_view.data_handle() + map_view.size());
99
100 // Build input_global_indices as the local-to-global of the new geometry
101 // dofs.
102 const std::int32_t num_nodes
103 = new_imap->size_local() + new_imap->num_ghosts();
104 std::vector<std::int32_t> local(num_nodes);
105 std::iota(local.begin(), local.end(), 0);
106 std::vector<std::int64_t> igi(num_nodes);
107 new_imap->local_to_global(local, igi);
108
109 mesh::Geometry<T> geometry(
110 new_imap, std::vector<std::vector<std::int32_t>>{std::move(dofmap_flat)},
111 std::vector<CoordinateElement<T>>{new_cmap}, std::move(x_new), gdim,
112 std::move(igi));
113
114 return mesh::Mesh<T>(mesh->comm(), mesh->topology(), std::move(geometry));
115}
