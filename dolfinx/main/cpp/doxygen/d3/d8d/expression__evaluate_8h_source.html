62template <dolfinx::scalar T, std::floating_point U>
63void check_expression_entities(const fem::Expression<T, U>& e,
64 const mesh::Mesh<U>& mesh, std::size_t rank)
65{
66 // An Expression is compiled for one cell type, and cell indices are
67 // local to a cell type, so neither cells nor (cell, local index)
68 // pairs identify entities of a mesh with several cell types
69 if (mesh.topology()->cell_types().size() > 1)
70 {
71 throw std::invalid_argument("Evaluating an Expression on a mesh with "
72 "multiple cell types is not supported.");
73 }
74
75 if (e.coordinate_element_hash() != mesh.geometry().cmaps().front().hash())
76 {
77 throw std::invalid_argument(
78 "Expression was created on a different mesh. Cannot tabulate.");
79 }
80
81 const int tdim = mesh.topology()->dim();
82 if (e.entity_dim() > tdim)
83 {
84 throw std::invalid_argument(
85 std::format("Cannot evaluate an Expression on entities of dimension "
86 "{} on a mesh of dimension {}.",
87 e.entity_dim(), tdim));
88 }
89
90 if (rank == 1 and e.entity_dim() != tdim)
91 {
92 throw std::invalid_argument(
93 "An Expression on facets, ridges or peaks must be evaluated on "
94 "(cell, local entity index) pairs.");
95 }
96
97 // An argument on another mesh must live on the evaluated entities
98 if (std::shared_ptr<const FunctionSpace<U>> V = e.argument_space();
99 V and V->mesh()->topology() != mesh.topology())
100 {
101 mesh::check_entity_cell_mapping(*V->mesh()->topology(), *mesh.topology(),
102 e.entity_dim());
103 }
104}

120template <dolfinx::scalar T, std::floating_point U>
121std::pair<std::vector<T>, int> pack_coefficients(const fem::Expression<T, U>& e,
122 const mesh::Mesh<U>& mesh,
123 fem::MDSpan2 auto entities)
124{
125 // Validate before packing, as packing may compute cell permutations
126 // collectively
127 impl::check_expression_entities(e, mesh, entities.rank());
128
129 std::vector<int> coffsets = e.coefficient_offsets();
130 const int cstride = coffsets.back();
131 std::vector<T> coeffs(entities.extent(0) * cstride);
132 std::vector<std::reference_wrapper<const Function<T, U>>> c;
133 std::ranges::transform(e.coefficients(), std::back_inserter(c),
134 [](const std::shared_ptr<const Function<T, U>>& f)
135 -> const Function<T, U>& { return *f; });
136 fem::pack_coefficients(c, *mesh.topology(), entities, e.entity_dim(),
137 e.entity_maps(), coffsets, std::span(coeffs));
138 return {std::move(coeffs), cstride};
139}

169template <dolfinx::scalar T, std::floating_point U>
170void tabulate_expression(
171 std::span<T> values, const fem::Expression<T, U>& e,
172 md::mdspan<const T, md::dextents<std::size_t, 2>> coeffs,
173 std::span<const T> constants, const mesh::Mesh<U>& mesh,
174 fem::MDSpan2 auto entities,
175 std::optional<
176 std::pair<std::reference_wrapper<const FiniteElement<U>>, std::size_t>>
177 element)
178{
179 impl::check_expression_entities(e, mesh, entities.rank());
180
181 std::shared_ptr<const FunctionSpace<U>> V = e.argument_space();
182
183 // Topology providing the argument's cell orientation data and, when
184 // the argument is on another mesh, its cell for each of `entities`.
185 // `topology_v` and `argument_cells` back the spans passed to
186 // impl::tabulate_expression.
187 std::shared_ptr<mesh::Topology> topology_v = mesh.topology_mutable();
188 std::vector<std::int32_t> argument_cells;
189 std::span<const std::uint32_t> cell_info;
190 if (element and element->first.get().needs_dof_transformations())
191 {
192 if (V and V->mesh()->topology() != mesh.topology())
193 topology_v = V->mesh()->topology_mutable();
194
195 // NOTE: Collective
196 topology_v->create_cell_permutations();
197 cell_info = topology_v->get_cell_permutation_info();
198
199 if (topology_v != mesh.topology_mutable())
200 {
201 const mesh::Topology& topology = *mesh.topology();
202 const mesh::EntityMap& emap
203 = mesh::find_entity_map(e.entity_maps(), topology, *topology_v);
204 argument_cells = mesh::extract_cells_from_entities(
205 *topology_v, topology, entities, std::cref(emap));
206#ifndef NDEBUG
207 if (std::ranges::find(argument_cells, -1) != argument_cells.end())
208 {
209 throw std::invalid_argument(
210 "An entity has no cell in the mesh of the argument.");
211 }
212#endif
213 }
214 }
215
216 impl::tabulate_expression(values, e.kernel(), e.X().second, e.value_size(),
217 coeffs, constants, mesh, entities, element,
218 cell_info, argument_cells);
219}

237template <dolfinx::scalar T, std::floating_point U>
238void tabulate_expression(std::span<T> values, const fem::Expression<T, U>& e,
239 const mesh::Mesh<U>& mesh, fem::MDSpan2 auto entities)
240{
241 // Packing validates `e` against `mesh` and `entities` first
242 auto [coeffs, cstride] = fem::pack_coefficients(e, mesh, entities);
243 std::vector<T> constants = fem::pack_constants(e);
244
245 std::optional<
246 std::pair<std::reference_wrapper<const FiniteElement<U>>, std::size_t>>
247 element = std::nullopt;
248 if (std::shared_ptr<const FunctionSpace<U>> V = e.argument_space(); V)
249 {
250 std::size_t num_argument_dofs
251 = V->dofmap()->element_dof_layout().num_dofs() * V->dofmap()->bs();
252 assert(V->element());
253 element = {std::cref(*V->element()), num_argument_dofs};
254 }
255
256 tabulate_expression(
257 values, e,
258 md::mdspan(std::as_const(coeffs).data(), entities.extent(0), cstride),
259 std::span<const T>(constants), mesh, entities, element);
260}

285template <dolfinx::scalar T, std::floating_point U>
286void interpolate(Function<T, U>& u1, mesh::CellRange auto&& cells1,
287 const Expression<T, U>& e0, mesh::CellRange auto&& cells0)
288{
289 // Extract mesh
290 const mesh::Mesh<U>* mesh0 = nullptr;
291 for (auto& c : e0.coefficients())
292 {
293 assert(c);
294 assert(c->function_space());
295 assert(c->function_space()->mesh());
296 if (auto mesh = c->function_space()->mesh().get(); !mesh0)
297 mesh0 = mesh;
298 else if (mesh != mesh0)
299 {
300 throw std::invalid_argument(
301 "Expression coefficient Functions have different meshes.");
302 }
303 }
304
305 // If Expression has no Function coefficients take mesh from `u1`.
306 auto V1 = u1.function_space();
307 assert(V1);
308 assert(V1->mesh());
309 if (!mesh0)
310 mesh0 = V1->mesh().get();
311
312 if (cells0.size() != cells1.size())
313 throw std::invalid_argument("Cell lists have different lengths.");
314
315 // Check that Function and Expression spaces are compatible
316 assert(V1->element());
317 std::size_t value_size = e0.value_size();
318 if (e0.argument_space())
319 throw std::invalid_argument("Cannot interpolate Expression with Argument.");
320
321 if (value_size != (std::size_t)V1->element()->value_size())
322 {
323 throw std::invalid_argument(
324 "Function value size not equal to Expression value size.");
325 }
326
327 // Compatibility check
328 {
329 const auto& [X0, shape0] = e0.X();
330 auto [X1, shape1] = V1->element()->interpolation_points();
331 if (shape0 != shape1)
332 {
333 throw std::invalid_argument(
334 "Function element interpolation points has different shape to "
335 "Expression interpolation points");
336 }
337
338 for (std::size_t i = 0; i < X0.size(); ++i)
339 {
340 if (std::abs(X0[i] - X1[i]) > 1.0e-10)
341 {
342 throw std::invalid_argument("Function element interpolation points not "
343 "equal to Expression interpolation points");
344 }
345 }
346 }
347
348 // Array to hold evaluated Expression
349 std::size_t num_cells = cells0.size();
350 std::size_t num_points = e0.num_points();
351 std::vector<T> fdata(num_cells * num_points * value_size);
352 md::mdspan<const T, md::dextents<std::size_t, 3>> f(fdata.data(), num_cells,
353 num_points, value_size);
354
355 // Evaluate Expression at points
356 std::vector<std::int32_t> _cells0(cells0.begin(), cells0.end());
357 tabulate_expression(std::span(fdata), e0, *mesh0,
358 md::mdspan(_cells0.data(), _cells0.size()));
359
360 // Reshape evaluated data to fit interpolate.
361 // Expression returns matrix of shape (num_cells, num_points *
362 // value_size), i.e. xyzxyz ordering of dof values per cell per
363 // point. The interpolation uses xxyyzz input, ordered for all
364 // points of each cell, i.e. (value_size, num_cells*num_points).
365 std::vector<T> fdata1(num_cells * num_points * value_size);
366 md::mdspan<T, md::dextents<std::size_t, 3>> f1(fdata1.data(), value_size,
367 num_cells, num_points);
368 for (std::size_t i = 0; i < f.extent(0); ++i)
369 for (std::size_t j = 0; j < f.extent(1); ++j)
370 for (std::size_t k = 0; k < f.extent(2); ++k)
371 f1(k, i, j) = f(i, j, k);
372
373 // Interpolate values into appropriate space
374 fem::interpolate<T>(u1, std::span<const T>(fdata1.data(), fdata1.size()),
375 {value_size, num_cells * num_points}, cells1);
376}

404template <dolfinx::scalar T, std::floating_point U>
405void interpolate(Function<T, U>& u1, const Expression<T, U>& e0)
406{
407 assert(u1.function_space());
408 assert(u1.function_space()->mesh());
409 int tdim = u1.function_space()->mesh()->topology()->dim();
410 auto map = u1.function_space()->mesh()->topology()->index_map(tdim);
411 assert(map);
412 interpolate<T, U>(
413 u1, e0, std::ranges::iota_view(0, map->size_local() + map->num_ghosts()));
414}
