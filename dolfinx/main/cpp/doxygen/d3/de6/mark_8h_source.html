52template <std::floating_point T>
53std::vector<std::int32_t> mark_maximum(std::span<const T> values,
54 const common::IndexMap& index_map,
55 std::type_identity_t<T> theta)
56{
57 if ((theta <= 0) or (theta > 1))
58 {
59 throw std::invalid_argument(
60 std::format("theta must satisfy 0 < theta <= 1, got {}.", theta));
61 }
62
63 std::int32_t n = values.size();
64 std::int32_t size = index_map.size_local() + index_map.num_ghosts();
65 if (n != size)
66 {
67 throw std::invalid_argument(
68 std::format("values must have size index_map.size_local() + "
69 "index_map.num_ghosts() = {}, got {}.",
70 size, n));
71 }
72
73 T local_max = index_map.size_local() == 0
74 ? std::numeric_limits<T>::lowest()
75 : std::ranges::max(values.first(index_map.size_local()));
76
77 T max = 0;
78 MPI_Allreduce(&local_max, &max, 1, dolfinx::MPI::mpi_t<T>, MPI_MAX,
79 index_map.comm());
80
81 T threshold = theta * max;
82
83 auto mark = [threshold](T e) { return e > threshold; };
84
85 std::vector<std::int32_t> indices;
86 indices.reserve(std::ranges::count_if(values, mark));
87 for (std::int32_t i = 0; i < n; ++i)
88 {
89 if (mark(values[i]))
90 indices.push_back(i);
91 }
92
93 spdlog::info("Marking (maximum): marked {} of {} local entries (owned + "
94 "ghost).",
95 indices.size(), n);
96
97 return indices;
98}
