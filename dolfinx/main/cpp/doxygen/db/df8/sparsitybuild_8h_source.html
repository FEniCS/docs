36template <std::ranges::input_range R0, std::ranges::input_range R1>
37void cells(la::SparsityPattern& pattern, const std::pair<R0, R1>& cells,
38 std::array<std::reference_wrapper<const DofMap>, 2> dofmaps)
39{
40 assert(cells.first.size() == cells.second.size());
41 const DofMap& map0 = dofmaps[0].get();
42 const DofMap& map1 = dofmaps[1].get();
43
44 // Reserve entries for all cell-wise outer products.
45 if constexpr (std::ranges::sized_range<R0> and std::ranges::sized_range<R1>)
46 {
47 if (std::size_t num_cells = std::ranges::size(cells.first); num_cells > 0)
48 {
49 std::size_t n0 = map0.cell_dofs(*cells.first.begin()).size();
50 std::size_t n1 = map1.cell_dofs(*cells.second.begin()).size();
51 pattern.reserve_blocks(num_cells, num_cells * n0, num_cells * n1);
52 }
53 }
54
55 for (auto cell0 = cells.first.begin(), cell1 = cells.second.begin();
56 cell0 != cells.first.end() and cell1 != cells.second.end();
57 ++cell0, ++cell1)
58 {
59 pattern.insert(map0.cell_dofs(*cell0), map1.cell_dofs(*cell1));
60 }
61}
