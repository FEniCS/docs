42template <dolfinx::scalar T>
43std::map<std::pair<IntegralType, int>, std::pair<std::span<const T>, int>>
44make_coefficients_span(const std::map<std::pair<IntegralType, int>,
45 std::pair<std::vector<T>, int>>& coeffs)
46{
47 using Key = typename std::remove_reference_t<decltype(coeffs)>::key_type;
48 std::map<Key, std::pair<std::span<const T>, int>> c;
49 std::ranges::transform(
50 coeffs, std::inserter(c, c.end()),
51 [](auto& e) -> typename decltype(c)::value_type
52 { return {e.first, {e.second.first, e.second.second}}; });
53 return c;
54}

69template <dolfinx::scalar T, std::floating_point U>
70T assemble_scalar(
71 const Form<T, U>& M, std::span<const T> constants,
72 const std::map<std::pair<IntegralType, int>,
73 std::pair<std::span<const T>, int>>& coefficients)
74{
75 using mdspanx3_t
76 = md::mdspan<const U, md::extents<std::size_t, md::dynamic_extent, 3>>;
77
78 std::shared_ptr<const mesh::Mesh<U>> mesh = M.mesh();
79 assert(mesh);
80 std::span x = mesh->geometry().x();
81
82 // Accumulate contributions from each cell type
83 const int num_cell_types = mesh->topology()->cell_types().size();
84 T val = 0;
85 for (int cell_type_idx = 0; cell_type_idx < num_cell_types; ++cell_type_idx)
86 {
87 // Geometry dofmap and data
88 md::mdspan<const std::int32_t, md::dextents<std::size_t, 2>> x_dofmap
89 = mesh->geometry().dofmaps().at(cell_type_idx);
90 val += impl::assemble_scalar(M, x_dofmap,
91 mdspanx3_t(x.data(), x.size() / 3, 3),
92 constants, coefficients, cell_type_idx);
93 }
94 return val;
95}

233template <typename V,
234 std::floating_point U
235 = scalar_value_t<typename std::remove_cvref_t<V>::value_type>,
236 dolfinx::scalar T = typename std::remove_cvref_t<V>::value_type>
237 requires std::is_same_v<typename std::remove_cvref_t<V>::value_type, T>
238void apply_lifting(
239 V&& b,
240 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
241 a,
242 const std::vector<std::span<const T>>& constants,
243 const std::vector<std::map<std::pair<IntegralType, int>,
244 std::pair<std::span<const T>, int>>>& coeffs,
245 const std::vector<
246 std::vector<std::reference_wrapper<const DirichletBC<T, U>>>>& bcs1,
247 const std::vector<std::span<const T>>& x0, T alpha)
248{
249 // If all forms are null, there is nothing to do
250 if (std::ranges::all_of(a, [](auto ai) { return !ai; }))
251 return;
252
253 common::Timer t("[Apply lifting]");
254
255 if (!x0.empty() and x0.size() != a.size())
256 {
257 throw std::invalid_argument(
258 "Mismatch in size between x0 and bilinear form in assembler.");
259 }
260
261 if (a.size() != bcs1.size())
262 {
263 throw std::invalid_argument(
264 "Mismatch in size between a and bcs in assembler.");
265 }
266
267 // Reused across iterations so `assign` below can recycle the
268 // existing buffer instead of reallocating for every block.
269 std::vector<std::int8_t> bc_markers1;
270 std::vector<T> bc_values1;
271 for (std::size_t j = 0; j < a.size(); ++j)
272 {
273 if (a[j] and !bcs1[j].empty())
274 {
275 assert(a[j]->get().function_spaces().at(0));
276 auto V1 = a[j]->get().function_spaces()[1];
277 assert(V1);
278
279 const int bs0 = a[j]->get().function_spaces()[0]->dofmaps().front()->bs();
280 const int bs1 = V1->dofmaps().front()->bs();
281
282 std::span<const T> _x0;
283 if (!x0.empty())
284 _x0 = x0[j];
285
286 std::shared_ptr<const DofMap> dofmap = V1->dofmaps().front();
287 auto map1 = dofmap->index_map;
288 const int map_bs1 = dofmap->index_map_bs();
289 assert(map1);
290 const int crange = map_bs1 * (map1->size_local() + map1->num_ghosts());
291 bc_markers1.assign(crange, false);
292 bc_values1.assign(crange, 0);
293 for (auto& bc : bcs1[j])
294 {
295 bc.get().mark_dofs(bc_markers1);
296 bc.get().set(bc_values1, std::nullopt, 1);
297 }
298
299 if (bs0 == 1 and bs1 == 1)
300 {
301 impl::lift_bc(b, a[j]->get(), std::integral_constant<int, 1>{},
302 std::integral_constant<int, 1>{}, constants[j], coeffs[j],
303 std::span<const T>(bc_values1), bc_markers1, _x0, alpha);
304 }
305 else if (bs0 == 3 and bs1 == 3)
306 {
307 impl::lift_bc(b, a[j]->get(), std::integral_constant<int, 3>{},
308 std::integral_constant<int, 3>{}, constants[j], coeffs[j],
309 std::span<const T>(bc_values1), bc_markers1, _x0, alpha);
310 }
311 else
312 {
313 impl::lift_bc(b, a[j]->get(), bs0, bs1, constants[j], coeffs[j],
314 std::span<const T>(bc_values1), bc_markers1, _x0, alpha);
315 }
316 }
317 }
318}

348template <typename V,
349 std::floating_point U
350 = scalar_value_t<typename std::remove_cvref_t<V>::value_type>,
351 dolfinx::scalar T = typename std::remove_cvref_t<V>::value_type>
352 requires std::is_same_v<typename std::remove_cvref_t<V>::value_type, T>
353void apply_lifting(
354 V&& b,
355 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
356 a,
357 const std::vector<
358 std::vector<std::reference_wrapper<const DirichletBC<T, U>>>>& bcs1,
359 const std::vector<std::span<const T>>& x0, T alpha)
360{
361 std::vector<
362 std::map<std::pair<IntegralType, int>, std::pair<std::vector<T>, int>>>
363 coeffs;
364 std::vector<std::vector<T>> constants;
365 for (const auto& _a : a)
366 {
367 if (_a)
368 {
369 auto coefficients = allocate_coefficient_storage(_a->get());
370 pack_coefficients(_a->get(), coefficients);
371 coeffs.push_back(coefficients);
372 constants.push_back(pack_constants(_a->get()));
373 }
374 else
375 {
376 coeffs.emplace_back();
377 constants.emplace_back();
378 }
379 }
380
381 std::vector<std::span<const T>> _constants(constants.begin(),
382 constants.end());
383 std::vector<std::map<std::pair<IntegralType, int>,
384 std::pair<std::span<const T>, int>>>
385 _coeffs;
386 std::ranges::transform(coeffs, std::back_inserter(_coeffs),
387 [](auto& c) { return make_coefficients_span(c); });
388
389 apply_lifting(b, a, _constants, _coeffs, bcs1, x0, alpha);
390}

439template <dolfinx::scalar T, std::floating_point U>
440void assemble_matrix(
441 auto mat_add, const Form<T, U>& a, std::span<const T> constants,
442 const std::map<std::pair<IntegralType, int>,
443 std::pair<std::span<const T>, int>>& coefficients,
444 const std::vector<std::reference_wrapper<const DirichletBC<T, U>>>& bcs)
445{
446 // Index maps for dof ranges
447 // NOTE: For mixed-topology meshes, there will be multiple DOF maps,
448 // but the index maps are the same.
449 auto map0 = a.function_spaces().at(0)->dofmaps().front()->index_map;
450 auto map1 = a.function_spaces().at(1)->dofmaps().front()->index_map;
451 auto bs0 = a.function_spaces().at(0)->dofmaps().front()->index_map_bs();
452 auto bs1 = a.function_spaces().at(1)->dofmaps().front()->index_map_bs();
453
454 // Build dof markers
455 std::vector<std::int8_t> dof_marker0, dof_marker1;
456 assert(map0);
457 std::int32_t dim0 = bs0 * (map0->size_local() + map0->num_ghosts());
458 assert(map1);
459 std::int32_t dim1 = bs1 * (map1->size_local() + map1->num_ghosts());
460 for (std::size_t k = 0; k < bcs.size(); ++k)
461 {
462 assert(bcs[k].get().function_space());
463 if (a.function_spaces().at(0)->contains(*bcs[k].get().function_space()))
464 {
465 dof_marker0.resize(dim0, false);
466 bcs[k].get().mark_dofs(dof_marker0);
467 }
468
469 if (a.function_spaces().at(1)->contains(*bcs[k].get().function_space()))
470 {
471 dof_marker1.resize(dim1, false);
472 bcs[k].get().mark_dofs(dof_marker1);
473 }
474 }
475
476 // Assemble
477 fem::assemble_matrix(mat_add, a, constants, coefficients, dof_marker0,
478 dof_marker1);
479}
