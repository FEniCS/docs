47template <dolfinx::scalar T>
48std::map<std::pair<IntegralType, int>, std::pair<std::span<const T>, int>>
49make_coefficients_span(const std::map<std::pair<IntegralType, int>,
50 std::pair<std::vector<T>, int>>& coeffs)
51{
52 using Key = typename std::remove_reference_t<decltype(coeffs)>::key_type;
53 std::map<Key, std::pair<std::span<const T>, int>> c;
54 std::ranges::transform(
55 coeffs, std::inserter(c, c.end()),
56 [](auto& e) -> typename decltype(c)::value_type
57 { return {e.first, {e.second.first, e.second.second}}; });
58 return c;
59}

74template <dolfinx::scalar T, std::floating_point U>
75T assemble_scalar(
76 const Form<T, U>& M, std::span<const T> constants,
77 const std::map<std::pair<IntegralType, int>,
78 std::pair<std::span<const T>, int>>& coefficients)
79{
80 using mdspanx3_t
81 = md::mdspan<const U, md::extents<std::size_t, md::dynamic_extent, 3>>;
82
83 std::shared_ptr<const mesh::Mesh<U>> mesh = M.mesh();
84 assert(mesh);
85 std::span x = mesh->geometry().x();
86
87 // Accumulate contributions from each cell type
88 const int num_cell_types = mesh->topology()->cell_types().size();
89 T val = 0;
90 for (int cell_type_idx = 0; cell_type_idx < num_cell_types; ++cell_type_idx)
91 {
92 // Geometry dofmap and data
93 md::mdspan<const std::int32_t, md::dextents<std::size_t, 2>> x_dofmap
94 = mesh->geometry().dofmaps().at(cell_type_idx);
95 val += impl::assemble_scalar(M, x_dofmap,
96 mdspanx3_t(x.data(), x.size() / 3, 3),
97 constants, coefficients, cell_type_idx);
98 }
99 return val;
100}

168template <dolfinx::scalar T, std::floating_point U>
169std::vector<std::int8_t> bc_dof_markers(
170 const FunctionSpace<U>& V,
171 const std::vector<std::reference_wrapper<const DirichletBC<T, U>>>& bcs)
172{
173 std::vector<std::int8_t> markers;
174 for (auto& bc : bcs)
175 {
176 assert(bc.get().function_space());
177 if (V.contains(*bc.get().function_space()))
178 {
179 if (markers.empty())
180 {
181 std::shared_ptr<const DofMap> dofmap = V.dofmaps().front();
182 std::shared_ptr<const common::IndexMap> map = dofmap->index_map;
183 assert(map);
184 markers.resize(dofmap->index_map_bs()
185 * (map->size_local() + map->num_ghosts()),
186 0);
187 }
188 bc.get().mark_dofs(markers);
189 }
190 }
191 return markers;
192}

198template <dolfinx::scalar T, std::floating_point U>
199std::pair<std::vector<std::vector<std::int8_t>>, std::vector<std::vector<T>>>
200bc_lifting_data(
201 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
202 a,
203 const std::vector<
204 std::vector<std::reference_wrapper<const DirichletBC<T, U>>>>& bcs1)
205{
206 if (a.size() != bcs1.size())
207 {
208 throw std::invalid_argument(
209 "Mismatch in size between a and bcs in assembler.");
210 }
211
212 std::vector<std::vector<std::int8_t>> markers(a.size());
213 std::vector<std::vector<T>> values(a.size());
214 for (std::size_t j = 0; j < a.size(); ++j)
215 {
216 if (a[j] and !bcs1[j].empty())
217 {
218 std::shared_ptr<const DofMap> dofmap
219 = a[j]->get().function_spaces().at(1)->dofmaps().front();
220 std::shared_ptr<const common::IndexMap> map1 = dofmap->index_map;
221 assert(map1);
222 const std::int32_t crange
223 = dofmap->index_map_bs() * (map1->size_local() + map1->num_ghosts());
224 markers[j].assign(crange, 0);
225 values[j].assign(crange, 0);
226 for (auto& bc : bcs1[j])
227 {
228 bc.get().mark_dofs(markers[j]);
229 bc.get().set(values[j], std::nullopt, 1);
230 }
231 }
232 }
233
234 return {std::move(markers), std::move(values)};
235}

271template <typename V,
272 std::floating_point U
273 = scalar_value_t<typename std::remove_cvref_t<V>::value_type>,
274 dolfinx::scalar T = typename std::remove_cvref_t<V>::value_type>
275 requires std::is_same_v<typename std::remove_cvref_t<V>::value_type, T>
276void apply_lifting(
277 V&& b,
278 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
279 a,
280 const std::vector<std::span<const T>>& constants,
281 const std::vector<std::map<std::pair<IntegralType, int>,
282 std::pair<std::span<const T>, int>>>& coeffs,
283 const std::vector<std::span<const std::int8_t>>& bc_markers1,
284 const std::vector<std::span<const T>>& bc_values1,
285 const std::vector<std::span<const T>>& x0, T alpha)
286{
287 // If all forms are null, there is nothing to do
288 if (std::ranges::all_of(a, [](auto ai) { return !ai; }))
289 return;
290
291 common::Timer t("[Apply lifting]");
292
293 if (!x0.empty() and x0.size() != a.size())
294 {
295 throw std::invalid_argument(
296 "Mismatch in size between x0 and bilinear form in assembler.");
297 }
298
299 if (bc_markers1.size() != a.size() or bc_values1.size() != a.size())
300 {
301 throw std::invalid_argument(
302 "Mismatch in size between a and bc markers/values in assembler.");
303 }
304
305 for (std::size_t j = 0; j < a.size(); ++j)
306 {
307 if (!a[j] or bc_markers1[j].empty())
308 continue;
309
310 assert(a[j]->get().function_spaces().at(0));
311 auto V1 = a[j]->get().function_spaces()[1];
312 assert(V1);
313
314 std::shared_ptr<const DofMap> dofmap = V1->dofmaps().front();
315 auto map1 = dofmap->index_map;
316 assert(map1);
317 const std::size_t crange
318 = dofmap->index_map_bs() * (map1->size_local() + map1->num_ghosts());
319 if (bc_markers1[j].size() != crange or bc_values1[j].size() != crange)
320 {
321 throw std::invalid_argument(std::format(
322 "bc markers/values for block {} have length {}/{}, expected {}.", j,
323 bc_markers1[j].size(), bc_values1[j].size(), crange));
324 }
325
326 const int bs0 = a[j]->get().function_spaces()[0]->dofmaps().front()->bs();
327 const int bs1 = dofmap->bs();
328
329 std::span<const T> _x0;
330 if (!x0.empty())
331 _x0 = x0[j];
332
333 impl::dispatch_bs(bs0, bs1,
334 [&b, &a, j, &constants, &coeffs, &bc_values1,
335 &bc_markers1, &_x0, alpha](auto bs0, auto bs1)
336 {
337 impl::lift_bc(b, a[j]->get(), bs0, bs1, constants[j],
338 coeffs[j], bc_values1[j], bc_markers1[j],
339 _x0, alpha);
340 });
341 }

484template <typename V,
485 std::floating_point U
486 = scalar_value_t<typename std::remove_cvref_t<V>::value_type>,
487 dolfinx::scalar T = typename std::remove_cvref_t<V>::value_type>
488 requires std::is_same_v<typename std::remove_cvref_t<V>::value_type, T>
489void apply_lifting(
490 V&& b,
491 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
492 a,
493 const std::vector<
494 std::vector<std::reference_wrapper<const DirichletBC<T, U>>>>& bcs1,
495 const std::vector<std::span<const T>>& x0, T alpha)
496{
497 std::vector<
498 std::map<std::pair<IntegralType, int>, std::pair<std::vector<T>, int>>>
499 coeffs;
500 std::vector<std::vector<T>> constants;
501 for (const auto& _a : a)
502 {
503 if (_a)
504 {
505 auto coefficients = allocate_coefficient_storage(_a->get());
506 pack_coefficients(_a->get(), coefficients);
507 coeffs.push_back(coefficients);
508 constants.push_back(pack_constants(_a->get()));
509 }
510 else
511 {
512 coeffs.emplace_back();
513 constants.emplace_back();
514 }
515 }
516
517 std::vector<std::span<const T>> _constants(constants.begin(),
518 constants.end());
519 std::vector<std::map<std::pair<IntegralType, int>,
520 std::pair<std::span<const T>, int>>>
521 _coeffs;
522 std::ranges::transform(coeffs, std::back_inserter(_coeffs),
523 [](auto& c) { return make_coefficients_span(c); });
524
525 auto [bc_markers1, bc_values1] = impl::bc_lifting_data(a, bcs1);
526 apply_lifting(
527 b, a, _constants, _coeffs,
528 std::vector<std::span<const std::int8_t>>(bc_markers1.begin(),
529 bc_markers1.end()),
530 std::vector<std::span<const T>>(bc_values1.begin(), bc_values1.end()), x0,
531 alpha);
532}

584template <dolfinx::scalar T, std::floating_point U>
585void assemble_matrix(
586 auto mat_add, const Form<T, U>& a,
587 const std::vector<std::reference_wrapper<const DirichletBC<T, U>>>& bcs)
588{
589 // Prepare constants and coefficients
590 const std::vector<T> constants = pack_constants(a);
591 auto coefficients = allocate_coefficient_storage(a);
592 pack_coefficients(a, coefficients);
593
594 std::vector<std::int8_t> dof_marker0
595 = impl::bc_dof_markers(*a.function_spaces().at(0), bcs);
596 std::vector<std::int8_t> dof_marker1
597 = impl::bc_dof_markers(*a.function_spaces().at(1), bcs);
598
599 // Assemble
600 assemble_matrix(mat_add, a, std::span<const T>(constants),
601 make_coefficients_span(coefficients),
602 std::span<const std::int8_t>(dof_marker0),
603 std::span<const std::int8_t>(dof_marker1));
604}

646template <dolfinx::scalar T>
647void set_diagonal(auto&& set_fn, const common::LocalIndexRange auto& rows,
648 T diagonal = T(1))
649{
650 std::span<const T, 1> diag_span(&diagonal, 1);
651 for (std::size_t i = 0; i < std::ranges::size(rows); ++i)
652 {
653 std::span<const std::int32_t, 1> row(std::ranges::data(rows) + i, 1);
654 set_fn(row, row, diag_span);

669void set_diagonal(auto&& set_fn, const common::LocalIndexRange auto& rows,
