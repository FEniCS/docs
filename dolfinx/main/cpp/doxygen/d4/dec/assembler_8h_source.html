51template <dolfinx::scalar T>
52std::map<std::pair<IntegralType, int>, std::pair<std::span<const T>, int>>
53make_coefficients_span(const std::map<std::pair<IntegralType, int>,
54 std::pair<std::vector<T>, int>>& coeffs)
55{
56 using Key = typename std::remove_reference_t<decltype(coeffs)>::key_type;
57 std::map<Key, std::pair<std::span<const T>, int>> c;
58 std::ranges::transform(
59 coeffs, std::inserter(c, c.end()),
60 [](auto& e) -> typename decltype(c)::value_type
61 { return {e.first, {e.second.first, e.second.second}}; });
62 return c;
63}

78template <dolfinx::scalar T, std::floating_point U>
79T assemble_scalar(
80 const Form<T, U>& M, std::span<const T> constants,
81 const std::map<std::pair<IntegralType, int>,
82 std::pair<std::span<const T>, int>>& coefficients)
83{
84 using mdspanx3_t
85 = md::mdspan<const U, md::extents<std::size_t, md::dynamic_extent, 3>>;
86
87 std::shared_ptr<const mesh::Mesh<U>> mesh = M.mesh();
88 assert(mesh);
89 std::span x = mesh->geometry().x();
90
91 // Accumulate contributions from each cell type
92 const int num_cell_types = mesh->topology()->cell_types().size();
93 T val = 0;
94 for (int cell_type_idx = 0; cell_type_idx < num_cell_types; ++cell_type_idx)
95 {
96 // Geometry dofmap and data
97 md::mdspan<const std::int32_t, md::dextents<std::size_t, 2>> x_dofmap
98 = mesh->geometry().dofmaps().at(cell_type_idx);
99 val += impl::assemble_scalar(M, x_dofmap,
100 mdspanx3_t(x.data(), x.size() / 3, 3),
101 constants, coefficients, cell_type_idx);
102 }
103 return val;
104}

172template <dolfinx::scalar T, std::floating_point U>
173std::vector<std::int8_t> bc_dof_markers(
174 const FunctionSpace<U>& V,
175 const std::vector<std::reference_wrapper<const DirichletBC<T, U>>>& bcs)
176{
177 std::vector<std::int8_t> markers;
178 for (auto& bc : bcs)
179 {
180 assert(bc.get().function_space());
181 if (V.contains(*bc.get().function_space()))
182 {
183 if (markers.empty())
184 {
185 std::shared_ptr<const DofMap> dofmap = V.dofmaps().front();
186 std::shared_ptr<const common::IndexMap> map = dofmap->index_map;
187 assert(map);
188 markers.resize(dofmap->index_map_bs()
189 * (map->size_local() + map->num_ghosts()),
190 0);
191 }
192 bc.get().mark_dofs(markers);
193 }
194 }
195 return markers;
196}

213template <dolfinx::scalar T, std::floating_point U>
214std::array<std::span<const std::int8_t>, 2> bc_dof_markers_pair(
215 const Form<T, U>& a,
216 const std::vector<std::reference_wrapper<const DirichletBC<T, U>>>& bcs,
217 std::array<std::vector<std::int8_t>, 2>& storage)
218{
219 storage[0] = bc_dof_markers(*a.function_spaces().at(0), bcs);
220 std::span<const std::int8_t> marker0(storage[0]);
221 if (a.function_spaces().at(0) == a.function_spaces().at(1))
222 return {marker0, marker0};
223
224 storage[1] = bc_dof_markers(*a.function_spaces().at(1), bcs);
225 return {marker0, std::span<const std::int8_t>(storage[1])};
226}

232template <dolfinx::scalar T, std::floating_point U>
233std::pair<std::vector<std::vector<std::int8_t>>, std::vector<std::vector<T>>>
234bc_lifting_data(
235 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
236 a,
237 const std::vector<
238 std::vector<std::reference_wrapper<const DirichletBC<T, U>>>>& bcs1)
239{
240 if (a.size() != bcs1.size())
241 {
242 throw std::invalid_argument(
243 "Mismatch in size between a and bcs in assembler.");
244 }
245
246 std::vector<std::vector<std::int8_t>> markers(a.size());
247 std::vector<std::vector<T>> values(a.size());
248 for (std::size_t j = 0; j < a.size(); ++j)
249 {
250 if (a[j] and !bcs1[j].empty())
251 {
252 std::shared_ptr<const DofMap> dofmap
253 = a[j]->get().function_spaces().at(1)->dofmaps().front();
254 std::shared_ptr<const common::IndexMap> map1 = dofmap->index_map;
255 assert(map1);
256 const std::int32_t crange
257 = dofmap->index_map_bs() * (map1->size_local() + map1->num_ghosts());
258 markers[j].assign(crange, 0);
259 values[j].assign(crange, 0);
260 for (auto& bc : bcs1[j])
261 {
262 bc.get().mark_dofs(markers[j]);
263 bc.get().set(values[j], std::nullopt, 1);
264 }
265 }
266 }
267
268 return {std::move(markers), std::move(values)};
269}

305template <typename V,
306 std::floating_point U
307 = scalar_value_t<typename std::remove_cvref_t<V>::value_type>,
308 dolfinx::scalar T = typename std::remove_cvref_t<V>::value_type>
309 requires std::is_same_v<typename std::remove_cvref_t<V>::value_type, T>
310void apply_lifting(
311 V&& b,
312 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
313 a,
314 const std::vector<std::span<const T>>& constants,
315 const std::vector<std::map<std::pair<IntegralType, int>,
316 std::pair<std::span<const T>, int>>>& coeffs,
317 const std::vector<std::span<const std::int8_t>>& bc_markers1,
318 const std::vector<std::span<const T>>& bc_values1,
319 const std::vector<std::span<const T>>& x0, T alpha)
320{
321 // If all forms are null, there is nothing to do
322 if (std::ranges::all_of(a, [](auto ai) { return !ai; }))
323 return;
324
325 common::Timer t("[Apply lifting]");
326
327 if (!x0.empty() and x0.size() != a.size())
328 {
329 throw std::invalid_argument(
330 "Mismatch in size between x0 and bilinear form in assembler.");
331 }
332
333 if (bc_markers1.size() != a.size() or bc_values1.size() != a.size())
334 {
335 throw std::invalid_argument(
336 "Mismatch in size between a and bc markers/values in assembler.");
337 }
338
339 for (std::size_t j = 0; j < a.size(); ++j)
340 {
341 if (!a[j] or bc_markers1[j].empty())
342 continue;
343
344 assert(a[j]->get().function_spaces().at(0));
345 auto V1 = a[j]->get().function_spaces()[1];
346 assert(V1);
347
348 std::shared_ptr<const DofMap> dofmap = V1->dofmaps().front();
349 auto map1 = dofmap->index_map;
350 assert(map1);
351 const std::size_t crange
352 = dofmap->index_map_bs() * (map1->size_local() + map1->num_ghosts());
353 if (bc_markers1[j].size() != crange or bc_values1[j].size() != crange)
354 {
355 throw std::invalid_argument(std::format(
356 "bc markers/values for block {} have length {}/{}, expected {}.", j,
357 bc_markers1[j].size(), bc_values1[j].size(), crange));
358 }
359
360 const int bs0 = a[j]->get().function_spaces()[0]->dofmaps().front()->bs();
361 const int bs1 = dofmap->bs();
362
363 std::span<const T> _x0;
364 if (!x0.empty())
365 _x0 = x0[j];
366
367 impl::dispatch_bs(bs0, bs1,
368 [&b, &a, j, &constants, &coeffs, &bc_values1,
369 &bc_markers1, &_x0, alpha](auto bs0, auto bs1)
370 {
371 impl::lift_bc(b, a[j]->get(), bs0, bs1, constants[j],
372 coeffs[j], bc_values1[j], bc_markers1[j],
373 _x0, alpha);
374 });
375 }

518template <typename V,
519 std::floating_point U
520 = scalar_value_t<typename std::remove_cvref_t<V>::value_type>,
521 dolfinx::scalar T = typename std::remove_cvref_t<V>::value_type>
522 requires std::is_same_v<typename std::remove_cvref_t<V>::value_type, T>
523void apply_lifting(
524 V&& b,
525 const std::vector<std::optional<std::reference_wrapper<const Form<T, U>>>>&
526 a,
527 const std::vector<
528 std::vector<std::reference_wrapper<const DirichletBC<T, U>>>>& bcs1,
529 const std::vector<std::span<const T>>& x0, T alpha)
530{
531 std::vector<
532 std::map<std::pair<IntegralType, int>, std::pair<std::vector<T>, int>>>
533 coeffs;
534 std::vector<std::vector<T>> constants;
535 for (const auto& _a : a)
536 {
537 if (_a)
538 {
539 auto coefficients = allocate_coefficient_storage(_a->get());
540 pack_coefficients(_a->get(), coefficients);
541 coeffs.push_back(coefficients);
542 constants.push_back(pack_constants(_a->get()));
543 }
544 else
545 {
546 coeffs.emplace_back();
547 constants.emplace_back();
548 }
549 }
550
551 std::vector<std::span<const T>> _constants(constants.begin(),
552 constants.end());
553 std::vector<std::map<std::pair<IntegralType, int>,
554 std::pair<std::span<const T>, int>>>
555 _coeffs;
556 std::ranges::transform(coeffs, std::back_inserter(_coeffs),
557 [](auto& c) { return make_coefficients_span(c); });
558
559 auto [bc_markers1, bc_values1] = impl::bc_lifting_data(a, bcs1);
560 apply_lifting(
561 b, a, _constants, _coeffs,
562 std::vector<std::span<const std::int8_t>>(bc_markers1.begin(),
563 bc_markers1.end()),
564 std::vector<std::span<const T>>(bc_values1.begin(), bc_values1.end()), x0,
565 alpha);
566}

697template <dolfinx::scalar T, std::floating_point U>
698void set_diagonal(
699 auto set_fn, const FunctionSpace<U>& V,
700 const std::vector<std::reference_wrapper<const DirichletBC<T, U>>>& bcs,
701 T diagonal = T(1))
702{
703 spdlog::debug("Set diagonal");
704
705 std::vector<std::int32_t> rows;
706 for (auto& bc : bcs)
707 {
708 if (V.contains(*bc.get().function_space()))
709 {
710 const auto [dofs, range] = bc.get().dof_indices();
711 std::span<const std::int32_t> owned = dofs.first(range);
712 rows.insert(rows.end(), owned.begin(), owned.end());
713 }
714 }
715
716 // A condition's dofs are strictly increasing (a DirichletBC
717 // precondition), so one condition needs no sort. Several give sorted
718 // runs, which a comparison sort handles poorly and which are often
719 // already in order, hence the check before radix sorting.
720 if (!std::ranges::is_sorted(rows))
721 dolfinx::radix_sort(rows);
722
723 // Overlapping conditions can repeat a row
724 rows.erase(std::ranges::unique(rows).begin(), rows.end());
725 la::set_diagonal(set_fn, rows, diagonal);
726}
