104 explicit Scatterer(const IndexMap& map)
105 : _sizes_remote(map.src().size(), 0),
106 _displs_remote(map.src().size() + 1), _sizes_local(map.dest().size()),
107 _displs_local(map.dest().size() + 1)
108 {
109 if (dolfinx::MPI::size(map.comm()) == 1)
110 return;
111
112 int ierr;
113 const std::span<const int> src = map.src();
114 const std::span<const int> dest = map.dest();
115
116 // Check that src and dest ranks are unique and sorted
117 assert(std::ranges::is_sorted(src));
118 assert(std::ranges::is_sorted(dest));
119
120 // Create communicators with directed edges:
121 // (0) owner -> ghost,
122 // (1) ghost -> owner
123 MPI_Comm comm0;
124 ierr = MPI_Dist_graph_create_adjacent(
125 map.comm(), src.size(), src.data(), MPI_UNWEIGHTED, dest.size(),
126 dest.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm0);
127 _comm0 = dolfinx::MPI::Comm(comm0, false);
128 dolfinx::MPI::check_error(map.comm(), ierr);
129
130 MPI_Comm comm1;
131 ierr = MPI_Dist_graph_create_adjacent(
132 map.comm(), dest.size(), dest.data(), MPI_UNWEIGHTED, src.size(),
133 src.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm1);
134 _comm1 = dolfinx::MPI::Comm(comm1, false);
135 dolfinx::MPI::check_error(map.comm(), ierr);
136
137 // Build permutation array that sorts ghost indices by owning rank
138 std::span owners = map.owners();
139 std::vector<std::int32_t> perm(owners.size());
140 std::iota(perm.begin(), perm.end(), 0);
141 dolfinx::radix_sort(perm, [&owners](auto index) { return owners[index]; });
142
143 // Sort (i) ghost indices and (ii) ghost index owners by rank
144 // (using perm array)
145 std::span ghosts = map.ghosts();
146 std::vector<int> owners_sorted(owners.size());
147 std::vector<std::int64_t> ghosts_sorted(owners.size());
148 std::ranges::transform(perm, owners_sorted.begin(),
149 [&owners](auto idx) { return owners[idx]; });
150 std::ranges::transform(perm, ghosts_sorted.begin(),
151 [&ghosts](auto idx) { return ghosts[idx]; });
152
153 // For data associated with ghost indices, packed by owning
154 // (neighbourhood) rank, compute sizes and displacements. I.e., when
155 // sending ghost index data from this rank to the owning ranks,
156 // disp[i] is the first entry in the buffer sent to neighbourhood
157 // rank i, and disp[i + 1] - disp[i] is the number of values sent to
158 // rank i.
159 assert(_sizes_remote.size() == src.size());
160 assert(_displs_remote.size() == src.size() + 1);
161 auto begin = owners_sorted.begin();
162 for (std::size_t i = 0; i < src.size(); i++)
163 {
164 auto upper = std::ranges::upper_bound(begin, owners_sorted.end(), src[i]);
165 std::size_t num_ind = std::ranges::distance(begin, upper);
166 _displs_remote[i + 1] = _displs_remote[i] + num_ind;
167 _sizes_remote[i] = num_ind;
168 begin = upper;
169 }
170
171 // For data associated with owned indices that are ghosted by other
172 // ranks, compute the size and displacement arrays. When sending
173 // data associated with ghost indices to the owner, these size and
174 // displacement arrays are for the receive buffer.
175
176 // Compute sizes and displacements of local data (how many local
177 // elements to be sent/received grouped by neighbors)
178 assert(_sizes_local.size() == dest.size());
179 assert(_displs_local.size() == dest.size() + 1);
180 _sizes_remote.reserve(1); // ensure data is not a nullptr
181 _sizes_local.reserve(1); // ensure data is not a nullptr
182 ierr
183 = MPI_Neighbor_alltoall(_sizes_remote.data(), 1, MPI_INT,
184 _sizes_local.data(), 1, MPI_INT, _comm1.comm());
185 dolfinx::MPI::check_error(_comm1.comm(), ierr);
186
187 std::inclusive_scan(_sizes_local.begin(), _sizes_local.end(),
188 std::next(_displs_local.begin()));
189
190 assert(static_cast<int>(ghosts_sorted.size()) == _displs_remote.back());
191
192 // Send ghost global indices to owning rank, and receive owned
193 // indices that are ghosts on other ranks
194 std::vector<std::int64_t> recv_buffer(_displs_local.back(), 0);
195 ierr = MPI_Neighbor_alltoallv(
196 ghosts_sorted.data(), _sizes_remote.data(), _displs_remote.data(),
197 MPI_INT64_T, recv_buffer.data(), _sizes_local.data(),
198 _displs_local.data(), MPI_INT64_T, _comm1.comm());
199 dolfinx::MPI::check_error(_comm1.comm(), ierr);
200
201 const std::array<std::int64_t, 2> range = map.local_range();
202#ifndef NDEBUG
203 // Check that all received indices are within the owned range
204 std::ranges::for_each(recv_buffer, [range](auto idx)
205 { assert(idx >= range[0] and idx < range[1]); });
206#endif
207
208 // Sizes, displacements and indices are all in blocks. The block
209 // size enters only through the MPI datatype used to send them, and
210 // through the caller's pack/unpack.
211 {
212 // Convert the received indices from global to local numbering
213 std::vector<typename container_type::value_type> idx(recv_buffer.size());
214 std::ranges::transform(recv_buffer, idx.begin(),
215 [offset = range[0]](auto i) ->
216 typename container_type::value_type
217 { return i - offset; });
218 _local_inds = std::move(idx);
219 }
220
221 _remote_inds = container_type(perm.begin(), perm.end());
222 }

596 static void wait(const dolfinx::MPI::Comm& comm, MPI_Request& request)
597 {
598 int ierr = MPI_Wait(&request, MPI_STATUS_IGNORE);

602 // Communicator where the source ranks own the indices in the callers
603 // halo, and the destination ranks 'ghost' indices owned by the
604 // caller. I.e.,
605 // - in-edges (src) are from ranks that own my ghosts
606 // - out-edges (dest) go to ranks that 'ghost' my owned indices
607 dolfinx::MPI::Comm _comm0{MPI_COMM_NULL};
608
609 // Communicator where the source ranks have ghost indices that are
610 // owned by the caller, and the destination ranks are the owners of
611 // indices in the callers halo region. I.e.,
612 // - in-edges (src) are from ranks that 'ghost' my owned indices
613 // - out-edges (dest) are to the owning ranks of my ghost indices
614 dolfinx::MPI::Comm _comm1{MPI_COMM_NULL};
615
616 // Permutation indices used to pack and unpack ghost data (remote)
617 container_type _remote_inds;
618
619 // Number of remote indices (ghosts) for each neighbor process
620 std::vector<int> _sizes_remote;
621
622 // Displacements of remote data for mpi scatter and gather
623 std::vector<int> _displs_remote;
624
625 // Permutation indices used to pack and unpack local shared data
626 // (owned indices that are shared with other processes). Indices are
627 // grouped by neighbor process.
628 container_type _local_inds;
629
630 // Number of local shared indices per neighbor process
631 std::vector<int> _sizes_local;
632
633 // Displacements of local data for mpi scatter and gather

655template <class Container>
656std::vector<std::int32_t>
657num_sharing_ranks(const IndexMap& map, const Scatterer<Container>& sc,
658 std::span<const std::int32_t> indices, int bs)
659{
660 if (bs < 1)
661 throw std::invalid_argument(
662 std::format("Block size must be positive, not {}.", bs));
663
664 const std::int32_t size_local = map.size_local();
665 const std::int32_t num_blocks = size_local + map.num_ghosts();
666
667 // Throw before the scatters, which all ranks must reach
668 for (std::int32_t i : indices)
669 {
670 if (i < 0 or i / bs >= num_blocks)
671 throw std::out_of_range("Local index out of range.");
672 }
673
674 // Initialised to 1 for this rank. The owner holds one
675 // local_indices_block entry per ghosting rank, so it counts
676 // occurrences rather than accumulating a reverse scatter.
677 std::vector<std::int32_t> count(num_blocks, 1);
678 const Container& local = sc.local_indices_block();
679 const Container& remote = sc.remote_indices_block();
680 for (std::int32_t i : local)
681 ++count[i];
682
683 std::vector<std::int32_t> send(local.size());
684 std::ranges::transform(local, send.begin(),
685 [&count](std::int32_t i) { return count[i]; });
686 std::vector<std::int32_t> recv(remote.size());
687 MPI_Request request = MPI_REQUEST_NULL;
688 sc.scatter_fwd_begin(send.data(), recv.data(), 1, request);
689 sc.scatter_fwd_end(request);
690 for (std::size_t i = 0; i < remote.size(); ++i)
691 count[size_local + remote[i]] = recv[i];
692
693 std::vector<std::int32_t> n(indices.size());
694 std::ranges::transform(indices, n.begin(), [&count, bs](std::int32_t i)
695 { return count[i / bs]; });
696 return n;
697}
