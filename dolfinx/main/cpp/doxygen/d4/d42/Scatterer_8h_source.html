68 explicit Scatterer(const IndexMap& map)
69 : _sizes_remote(map.src().size(), 0),
70 _displs_remote(map.src().size() + 1), _sizes_local(map.dest().size()),
71 _displs_local(map.dest().size() + 1)
72 {
73 if (dolfinx::MPI::size(map.comm()) == 1)
74 return;
75
76 int ierr;
77 const std::span<const int> src = map.src();
78 const std::span<const int> dest = map.dest();
79
80 // Check that src and dest ranks are unique and sorted
81 assert(std::ranges::is_sorted(src));
82 assert(std::ranges::is_sorted(dest));
83
84 // Create communicators with directed edges:
85 // (0) owner -> ghost,
86 // (1) ghost -> owner
87 MPI_Comm comm0;
88 ierr = MPI_Dist_graph_create_adjacent(
89 map.comm(), src.size(), src.data(), MPI_UNWEIGHTED, dest.size(),
90 dest.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm0);
91 _comm0 = dolfinx::MPI::Comm(comm0, false);
92 dolfinx::MPI::check_error(map.comm(), ierr);
93
94 MPI_Comm comm1;
95 ierr = MPI_Dist_graph_create_adjacent(
96 map.comm(), dest.size(), dest.data(), MPI_UNWEIGHTED, src.size(),
97 src.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm1);
98 _comm1 = dolfinx::MPI::Comm(comm1, false);
99 dolfinx::MPI::check_error(map.comm(), ierr);
100
101 // Build permutation array that sorts ghost indices by owning rank
102 std::span owners = map.owners();
103 std::vector<std::int32_t> perm(owners.size());
104 std::iota(perm.begin(), perm.end(), 0);
105 dolfinx::radix_sort(perm, [&owners](auto index) { return owners[index]; });
106
107 // Sort (i) ghost indices and (ii) ghost index owners by rank
108 // (using perm array)
109 std::span ghosts = map.ghosts();
110 std::vector<int> owners_sorted(owners.size());
111 std::vector<std::int64_t> ghosts_sorted(owners.size());
112 std::ranges::transform(perm, owners_sorted.begin(),
113 [&owners](auto idx) { return owners[idx]; });
114 std::ranges::transform(perm, ghosts_sorted.begin(),
115 [&ghosts](auto idx) { return ghosts[idx]; });
116
117 // For data associated with ghost indices, packed by owning
118 // (neighbourhood) rank, compute sizes and displacements. I.e., when
119 // sending ghost index data from this rank to the owning ranks,
120 // disp[i] is the first entry in the buffer sent to neighbourhood
121 // rank i, and disp[i + 1] - disp[i] is the number of values sent to
122 // rank i.
123 assert(_sizes_remote.size() == src.size());
124 assert(_displs_remote.size() == src.size() + 1);
125 auto begin = owners_sorted.begin();
126 for (std::size_t i = 0; i < src.size(); i++)
127 {
128 auto upper = std::ranges::upper_bound(begin, owners_sorted.end(), src[i]);
129 std::size_t num_ind = std::ranges::distance(begin, upper);
130 _displs_remote[i + 1] = _displs_remote[i] + num_ind;
131 _sizes_remote[i] = num_ind;
132 begin = upper;
133 }
134
135 // For data associated with owned indices that are ghosted by other
136 // ranks, compute the size and displacement arrays. When sending
137 // data associated with ghost indices to the owner, these size and
138 // displacement arrays are for the receive buffer.
139
140 // Compute sizes and displacements of local data (how many local
141 // elements to be sent/received grouped by neighbors)
142 assert(_sizes_local.size() == dest.size());
143 assert(_displs_local.size() == dest.size() + 1);
144 _sizes_remote.reserve(1); // ensure data is not a nullptr
145 _sizes_local.reserve(1); // ensure data is not a nullptr
146 ierr
147 = MPI_Neighbor_alltoall(_sizes_remote.data(), 1, MPI_INT,
148 _sizes_local.data(), 1, MPI_INT, _comm1.comm());
149 dolfinx::MPI::check_error(_comm1.comm(), ierr);
150
151 std::inclusive_scan(_sizes_local.begin(), _sizes_local.end(),
152 std::next(_displs_local.begin()));
153
154 assert(static_cast<int>(ghosts_sorted.size()) == _displs_remote.back());
155
156 // Send ghost global indices to owning rank, and receive owned
157 // indices that are ghosts on other ranks
158 std::vector<std::int64_t> recv_buffer(_displs_local.back(), 0);
159 ierr = MPI_Neighbor_alltoallv(
160 ghosts_sorted.data(), _sizes_remote.data(), _displs_remote.data(),
161 MPI_INT64_T, recv_buffer.data(), _sizes_local.data(),
162 _displs_local.data(), MPI_INT64_T, _comm1.comm());
163 dolfinx::MPI::check_error(_comm1.comm(), ierr);
164
165 const std::array<std::int64_t, 2> range = map.local_range();
166#ifndef NDEBUG
167 // Check that all received indices are within the owned range
168 std::ranges::for_each(recv_buffer, [range](auto idx)
169 { assert(idx >= range[0] and idx < range[1]); });
170#endif
171
172 // Sizes, displacements and indices are all in blocks. The block
173 // size enters only through the MPI datatype used to send them, and
174 // through the caller's pack/unpack.
175 {
176 // Convert the received indices from global to local numbering
177 std::vector<typename container_type::value_type> idx(recv_buffer.size());
178 std::ranges::transform(recv_buffer, idx.begin(),
179 [offset = range[0]](auto i) ->
180 typename container_type::value_type
181 { return i - offset; });
182 _local_inds = std::move(idx);
183 }
184
185 _remote_inds = container_type(perm.begin(), perm.end());
186 }

285 void scatter_fwd_begin_dtype(const T* send_buffer, T* recv_buffer,
286 MPI_Datatype type, MPI_Request& request) const
287 {
288 if (!has_neighbours())
289 {
290 request = MPI_REQUEST_NULL;
291 return;
292 }
293
294 int ierr = MPI_Ineighbor_alltoallv(
295 send_buffer, _sizes_local.data(), _displs_local.data(), type,
296 recv_buffer, _sizes_remote.data(), _displs_remote.data(), type,
297 _comm0.comm(), &request);
298 dolfinx::MPI::check_error(_comm0.comm(), ierr);
299 }

560 static void wait(const dolfinx::MPI::Comm& comm, MPI_Request& request)

566 // Communicator where the source ranks own the indices in the callers
567 // halo, and the destination ranks 'ghost' indices owned by the
568 // caller. I.e.,
569 // - in-edges (src) are from ranks that own my ghosts
570 // - out-edges (dest) go to ranks that 'ghost' my owned indices
571 dolfinx::MPI::Comm _comm0{MPI_COMM_NULL};
572
573 // Communicator where the source ranks have ghost indices that are
574 // owned by the caller, and the destination ranks are the owners of
575 // indices in the callers halo region. I.e.,
576 // - in-edges (src) are from ranks that 'ghost' my owned indices
577 // - out-edges (dest) are to the owning ranks of my ghost indices
578 dolfinx::MPI::Comm _comm1{MPI_COMM_NULL};
579
580 // Permutation indices used to pack and unpack ghost data (remote)
581 container_type _remote_inds;
582
583 // Number of remote indices (ghosts) for each neighbor process
584 std::vector<int> _sizes_remote;
585
586 // Displacements of remote data for mpi scatter and gather
587 std::vector<int> _displs_remote;
588
589 // Permutation indices used to pack and unpack local shared data

592 container_type _local_inds;
593
594 // Number of local shared indices per neighbor process
