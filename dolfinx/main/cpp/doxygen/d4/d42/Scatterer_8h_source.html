103 explicit Scatterer(const IndexMap& map)
104 : _sizes_remote(map.src().size(), 0),
105 _displs_remote(map.src().size() + 1), _sizes_local(map.dest().size()),
106 _displs_local(map.dest().size() + 1)
107 {
108 if (dolfinx::MPI::size(map.comm()) == 1)
109 return;
110
111 int ierr;
112 const std::span<const int> src = map.src();
113 const std::span<const int> dest = map.dest();
114
115 // Check that src and dest ranks are unique and sorted
116 assert(std::ranges::is_sorted(src));
117 assert(std::ranges::is_sorted(dest));
118
119 // Create communicators with directed edges:
120 // (0) owner -> ghost,
121 // (1) ghost -> owner
122 MPI_Comm comm0;
123 ierr = MPI_Dist_graph_create_adjacent(
124 map.comm(), src.size(), src.data(), MPI_UNWEIGHTED, dest.size(),
125 dest.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm0);
126 _comm0 = dolfinx::MPI::Comm(comm0, false);
127 dolfinx::MPI::check_error(map.comm(), ierr);
128
129 MPI_Comm comm1;
130 ierr = MPI_Dist_graph_create_adjacent(
131 map.comm(), dest.size(), dest.data(), MPI_UNWEIGHTED, src.size(),
132 src.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm1);
133 _comm1 = dolfinx::MPI::Comm(comm1, false);
134 dolfinx::MPI::check_error(map.comm(), ierr);
135
136 // Build permutation array that sorts ghost indices by owning rank
137 std::span owners = map.owners();
138 std::vector<std::int32_t> perm(owners.size());
139 std::iota(perm.begin(), perm.end(), 0);
140 dolfinx::radix_sort(perm, [&owners](auto index) { return owners[index]; });
141
142 // Sort (i) ghost indices and (ii) ghost index owners by rank
143 // (using perm array)
144 std::span ghosts = map.ghosts();
145 std::vector<int> owners_sorted(owners.size());
146 std::vector<std::int64_t> ghosts_sorted(owners.size());
147 std::ranges::transform(perm, owners_sorted.begin(),
148 [&owners](auto idx) { return owners[idx]; });
149 std::ranges::transform(perm, ghosts_sorted.begin(),
150 [&ghosts](auto idx) { return ghosts[idx]; });
151
152 // For data associated with ghost indices, packed by owning
153 // (neighbourhood) rank, compute sizes and displacements. I.e., when
154 // sending ghost index data from this rank to the owning ranks,
155 // disp[i] is the first entry in the buffer sent to neighbourhood
156 // rank i, and disp[i + 1] - disp[i] is the number of values sent to
157 // rank i.
158 assert(_sizes_remote.size() == src.size());
159 assert(_displs_remote.size() == src.size() + 1);
160 auto begin = owners_sorted.begin();
161 for (std::size_t i = 0; i < src.size(); i++)
162 {
163 auto upper = std::ranges::upper_bound(begin, owners_sorted.end(), src[i]);
164 std::size_t num_ind = std::ranges::distance(begin, upper);
165 _displs_remote[i + 1] = _displs_remote[i] + num_ind;
166 _sizes_remote[i] = num_ind;
167 begin = upper;
168 }
169
170 // For data associated with owned indices that are ghosted by other
171 // ranks, compute the size and displacement arrays. When sending
172 // data associated with ghost indices to the owner, these size and
173 // displacement arrays are for the receive buffer.
174
175 // Compute sizes and displacements of local data (how many local
176 // elements to be sent/received grouped by neighbors)
177 assert(_sizes_local.size() == dest.size());
178 assert(_displs_local.size() == dest.size() + 1);
179 _sizes_remote.reserve(1); // ensure data is not a nullptr
180 _sizes_local.reserve(1); // ensure data is not a nullptr
181 ierr
182 = MPI_Neighbor_alltoall(_sizes_remote.data(), 1, MPI_INT,
183 _sizes_local.data(), 1, MPI_INT, _comm1.comm());
184 dolfinx::MPI::check_error(_comm1.comm(), ierr);
185
186 std::inclusive_scan(_sizes_local.begin(), _sizes_local.end(),
187 std::next(_displs_local.begin()));
188
189 assert(static_cast<int>(ghosts_sorted.size()) == _displs_remote.back());
190
191 // Send ghost global indices to owning rank, and receive owned
192 // indices that are ghosts on other ranks
193 std::vector<std::int64_t> recv_buffer(_displs_local.back(), 0);
194 ierr = MPI_Neighbor_alltoallv(
195 ghosts_sorted.data(), _sizes_remote.data(), _displs_remote.data(),
196 MPI_INT64_T, recv_buffer.data(), _sizes_local.data(),
197 _displs_local.data(), MPI_INT64_T, _comm1.comm());
198 dolfinx::MPI::check_error(_comm1.comm(), ierr);
199
200 const std::array<std::int64_t, 2> range = map.local_range();
201#ifndef NDEBUG
202 // Check that all received indices are within the owned range
203 std::ranges::for_each(recv_buffer, [range](auto idx)
204 { assert(idx >= range[0] and idx < range[1]); });
205#endif
206
207 // Sizes, displacements and indices are all in blocks. The block
208 // size enters only through the MPI datatype used to send them, and
209 // through the caller's pack/unpack.
210 {
211 // Convert the received indices from global to local numbering
212 std::vector<typename container_type::value_type> idx(recv_buffer.size());
213 std::ranges::transform(recv_buffer, idx.begin(),
214 [offset = range[0]](auto i) ->
215 typename container_type::value_type
216 { return i - offset; });
217 _local_inds = std::move(idx);
218 }
219
220 _remote_inds = container_type(perm.begin(), perm.end());
221 }

601 // Communicator where the source ranks own the indices in the callers
602 // halo, and the destination ranks 'ghost' indices owned by the
603 // caller. I.e.,
604 // - in-edges (src) are from ranks that own my ghosts
605 // - out-edges (dest) go to ranks that 'ghost' my owned indices
606 dolfinx::MPI::Comm _comm0{MPI_COMM_NULL};
607
608 // Communicator where the source ranks have ghost indices that are
609 // owned by the caller, and the destination ranks are the owners of
610 // indices in the callers halo region. I.e.,
611 // - in-edges (src) are from ranks that 'ghost' my owned indices
612 // - out-edges (dest) are to the owning ranks of my ghost indices
613 dolfinx::MPI::Comm _comm1{MPI_COMM_NULL};
614
615 // Permutation indices used to pack and unpack ghost data (remote)
616 container_type _remote_inds;
617
618 // Number of remote indices (ghosts) for each neighbor process
619 std::vector<int> _sizes_remote;
620
621 // Displacements of remote data for mpi scatter and gather
622 std::vector<int> _displs_remote;
623
624 // Permutation indices used to pack and unpack local shared data
625 // (owned indices that are shared with other processes). Indices are
626 // grouped by neighbor process.
627 container_type _local_inds;
628
629 // Number of local shared indices per neighbor process
630 std::vector<int> _sizes_local;
631
632 // Displacements of local data for mpi scatter and gather

652template <class Container>
653std::vector<std::int32_t>
654num_sharing_ranks(const IndexMap& map, const Scatterer<Container>& sc,
655 std::span<const std::int32_t> indices, int bs)
656{
657 // One per block, counting this rank. A reverse scatter accumulates
658 // the ghost entries onto the owner, giving the owner the full count,
659 // and a forward scatter returns it to the ghosting ranks.
660 const std::int32_t size_local = map.size_local();
661 std::vector<std::int32_t> count(size_local + map.num_ghosts(), 1);
662
663 const Container& local = sc.local_indices_block();
664 const Container& remote = sc.remote_indices_block();
665 std::vector<std::int32_t> send(std::max(local.size(), remote.size()));
666 std::vector<std::int32_t> recv(std::max(local.size(), remote.size()));
667 MPI_Request request = MPI_REQUEST_NULL;
668
669 for (std::size_t i = 0; i < remote.size(); ++i)
670 send[i] = count[size_local + remote[i]];
671 sc.scatter_rev_begin(send.data(), recv.data(), 1, request);
672 sc.scatter_rev_end(request);
673 for (std::size_t i = 0; i < local.size(); ++i)
674 count[local[i]] += recv[i];
675
676 for (std::size_t i = 0; i < local.size(); ++i)
677 send[i] = count[local[i]];
678 sc.scatter_fwd_begin(send.data(), recv.data(), 1, request);
679 sc.scatter_fwd_end(request);
680 for (std::size_t i = 0; i < remote.size(); ++i)
681 count[size_local + remote[i]] = recv[i];
682
683 std::vector<std::int32_t> n(indices.size());
684 std::ranges::transform(
685 indices, n.begin(),
686 [&count, bs](std::int32_t i)
687 {
688 std::int32_t block = i / bs;
689 if (block < 0 or block >= static_cast<std::int32_t>(count.size()))
690 throw std::out_of_range("Local index out of range.");
691 return count[block];
692 });
693 return n;
694}
