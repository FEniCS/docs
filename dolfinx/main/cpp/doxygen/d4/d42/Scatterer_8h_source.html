101 explicit Scatterer(const IndexMap& map)
102 : _sizes_remote(map.src().size(), 0),
103 _displs_remote(map.src().size() + 1), _sizes_local(map.dest().size()),
104 _displs_local(map.dest().size() + 1)
105 {
106 if (dolfinx::MPI::size(map.comm()) == 1)
107 return;
108
109 int ierr;
110 const std::span<const int> src = map.src();
111 const std::span<const int> dest = map.dest();
112
113 // Check that src and dest ranks are unique and sorted
114 assert(std::ranges::is_sorted(src));
115 assert(std::ranges::is_sorted(dest));
116
117 // Create communicators with directed edges:
118 // (0) owner -> ghost,
119 // (1) ghost -> owner
120 MPI_Comm comm0;
121 ierr = MPI_Dist_graph_create_adjacent(
122 map.comm(), src.size(), src.data(), MPI_UNWEIGHTED, dest.size(),
123 dest.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm0);
124 _comm0 = dolfinx::MPI::Comm(comm0, false);
125 dolfinx::MPI::check_error(map.comm(), ierr);
126
127 MPI_Comm comm1;
128 ierr = MPI_Dist_graph_create_adjacent(
129 map.comm(), dest.size(), dest.data(), MPI_UNWEIGHTED, src.size(),
130 src.data(), MPI_UNWEIGHTED, MPI_INFO_NULL, false, &comm1);
131 _comm1 = dolfinx::MPI::Comm(comm1, false);
132 dolfinx::MPI::check_error(map.comm(), ierr);
133
134 // Build permutation array that sorts ghost indices by owning rank
135 std::span owners = map.owners();
136 std::vector<std::int32_t> perm(owners.size());
137 std::iota(perm.begin(), perm.end(), 0);
138 dolfinx::radix_sort(perm, [&owners](auto index) { return owners[index]; });
139
140 // Sort (i) ghost indices and (ii) ghost index owners by rank
141 // (using perm array)
142 std::span ghosts = map.ghosts();
143 std::vector<int> owners_sorted(owners.size());
144 std::vector<std::int64_t> ghosts_sorted(owners.size());
145 std::ranges::transform(perm, owners_sorted.begin(),
146 [&owners](auto idx) { return owners[idx]; });
147 std::ranges::transform(perm, ghosts_sorted.begin(),
148 [&ghosts](auto idx) { return ghosts[idx]; });
149
150 // For data associated with ghost indices, packed by owning
151 // (neighbourhood) rank, compute sizes and displacements. I.e., when
152 // sending ghost index data from this rank to the owning ranks,
153 // disp[i] is the first entry in the buffer sent to neighbourhood
154 // rank i, and disp[i + 1] - disp[i] is the number of values sent to
155 // rank i.
156 assert(_sizes_remote.size() == src.size());
157 assert(_displs_remote.size() == src.size() + 1);
158 auto begin = owners_sorted.begin();
159 for (std::size_t i = 0; i < src.size(); i++)
160 {
161 auto upper = std::ranges::upper_bound(begin, owners_sorted.end(), src[i]);
162 std::size_t num_ind = std::ranges::distance(begin, upper);
163 _displs_remote[i + 1] = _displs_remote[i] + num_ind;
164 _sizes_remote[i] = num_ind;
165 begin = upper;
166 }
167
168 // For data associated with owned indices that are ghosted by other
169 // ranks, compute the size and displacement arrays. When sending
170 // data associated with ghost indices to the owner, these size and
171 // displacement arrays are for the receive buffer.
172
173 // Compute sizes and displacements of local data (how many local
174 // elements to be sent/received grouped by neighbors)
175 assert(_sizes_local.size() == dest.size());
176 assert(_displs_local.size() == dest.size() + 1);
177 _sizes_remote.reserve(1); // ensure data is not a nullptr
178 _sizes_local.reserve(1); // ensure data is not a nullptr
179 ierr
180 = MPI_Neighbor_alltoall(_sizes_remote.data(), 1, MPI_INT,
181 _sizes_local.data(), 1, MPI_INT, _comm1.comm());
182 dolfinx::MPI::check_error(_comm1.comm(), ierr);
183
184 std::inclusive_scan(_sizes_local.begin(), _sizes_local.end(),
185 std::next(_displs_local.begin()));
186
187 assert(static_cast<int>(ghosts_sorted.size()) == _displs_remote.back());
188
189 // Send ghost global indices to owning rank, and receive owned
190 // indices that are ghosts on other ranks
191 std::vector<std::int64_t> recv_buffer(_displs_local.back(), 0);
192 ierr = MPI_Neighbor_alltoallv(
193 ghosts_sorted.data(), _sizes_remote.data(), _displs_remote.data(),
194 MPI_INT64_T, recv_buffer.data(), _sizes_local.data(),
195 _displs_local.data(), MPI_INT64_T, _comm1.comm());
196 dolfinx::MPI::check_error(_comm1.comm(), ierr);
197
198 const std::array<std::int64_t, 2> range = map.local_range();
199#ifndef NDEBUG
200 // Check that all received indices are within the owned range
201 std::ranges::for_each(recv_buffer, [range](auto idx)
202 { assert(idx >= range[0] and idx < range[1]); });
203#endif
204
205 // Sizes, displacements and indices are all in blocks. The block
206 // size enters only through the MPI datatype used to send them, and
207 // through the caller's pack/unpack.
208 {
209 // Convert the received indices from global to local numbering
210 std::vector<typename container_type::value_type> idx(recv_buffer.size());
211 std::ranges::transform(recv_buffer, idx.begin(),
212 [offset = range[0]](auto i) ->
213 typename container_type::value_type
214 { return i - offset; });
215 _local_inds = std::move(idx);
216 }
217
218 _remote_inds = container_type(perm.begin(), perm.end());
219 }

319 MPI_Datatype type, MPI_Request& request) const
320 {
321 if (!has_neighbours())

323 request = MPI_REQUEST_NULL;

325 }
326

328 send_buffer, _sizes_local.data(), _displs_local.data(), type,

592 // Complete a non-blocking request, checking errors against `comm`
593 static void wait(const dolfinx::MPI::Comm& comm, MPI_Request& request)
594 {
595 int ierr = MPI_Wait(&request, MPI_STATUS_IGNORE);
596 dolfinx::MPI::check_error(comm.comm(), ierr);

599 // Communicator where the source ranks own the indices in the callers
600 // halo, and the destination ranks 'ghost' indices owned by the
601 // caller. I.e.,
602 // - in-edges (src) are from ranks that own my ghosts
603 // - out-edges (dest) go to ranks that 'ghost' my owned indices
604 dolfinx::MPI::Comm _comm0{MPI_COMM_NULL};
605
606 // Communicator where the source ranks have ghost indices that are
607 // owned by the caller, and the destination ranks are the owners of
608 // indices in the callers halo region. I.e.,
609 // - in-edges (src) are from ranks that 'ghost' my owned indices
610 // - out-edges (dest) are to the owning ranks of my ghost indices
611 dolfinx::MPI::Comm _comm1{MPI_COMM_NULL};
612
613 // Permutation indices used to pack and unpack ghost data (remote)
614 container_type _remote_inds;
615
616 // Number of remote indices (ghosts) for each neighbor process
617 std::vector<int> _sizes_remote;
618
619 // Displacements of remote data for mpi scatter and gather
620 std::vector<int> _displs_remote;
621
622 // Permutation indices used to pack and unpack local shared data
623 // (owned indices that are shared with other processes). Indices are
624 // grouped by neighbor process.

626
627 // Number of local shared indices per neighbor process

630 // Displacements of local data for mpi scatter and gather
