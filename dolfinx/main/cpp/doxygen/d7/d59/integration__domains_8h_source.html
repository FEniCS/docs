34template <int num_cells>
35std::array<std::int32_t, 2 * num_cells>
36get_cell_facet_pairs(std::int32_t f, std::span<const std::int32_t> cells,
37 const graph::AdjacencyList<std::int32_t>& c_to_f)
38{
39 // Loop over cells sharing facet
40 assert(cells.size() == num_cells);
41 std::array<std::int32_t, 2 * num_cells> cell_local_facet_pairs;
42 for (int c = 0; c < num_cells; ++c)
43 {
44 // Get local index of facet with respect to the cell
45 std::int32_t cell = cells[c];
46 auto cell_facets = c_to_f.links(cell);
47 auto facet_it = std::find(cell_facets.begin(), cell_facets.end(), f);
48 assert(facet_it != cell_facets.end());
49 int local_f = std::ranges::distance(cell_facets.begin(), facet_it);
50 cell_local_facet_pairs[2 * c] = cell;
51 cell_local_facet_pairs[2 * c + 1] = local_f;
52 }
53
54 return cell_local_facet_pairs;
55}

64template <int num_cells>
65std::array<std::int32_t, 2 * num_cells>
66get_cell_entity_pairs(std::int32_t e, std::span<const std::int32_t> cells,
67 const graph::AdjacencyList<std::int32_t>& c_to_e)
68{
69 static_assert(num_cells == 1); // Patch assembly not supported.
70
71 assert(cells.size() > 0);
72
73 // Use first cell for assembly over by default
74 std::int32_t cell = cells[0];
75
76 // Find local index of entity within cell
77 auto cell_entities = c_to_e.links(cell);
78 auto it = std::ranges::find(cell_entities, e);
79 assert(it != cell_entities.end());
80 std::int32_t local_index = std::ranges::distance(cell_entities.begin(), it);
81
82 return {cell, local_index};
83}
