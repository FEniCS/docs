64 template <class F>
65 static void dispatch_bs(int bs, F&& f)
66 {
67 switch (bs)
68 {
69 case 1:
70 return f(std::integral_constant<int, 1>{});
71 case 2:
72 return f(std::integral_constant<int, 2>{});
73 case 3:
74 return f(std::integral_constant<int, 3>{});
75 default:
76 return f(bs);
77 }
78 }
79
84 auto get_pack()
85 {
86 return [bs = _bs](typename ScatterContainer::const_iterator idx_first,
87 typename ScatterContainer::const_iterator idx_last,
88 const auto in_first, auto out_first)
89 {
90 // out[i * bs + j] = in[idx[i] * bs + j]
91 dispatch_bs(bs,
92 [&](auto B)
93 {
94 auto out = out_first;
95 for (auto idx = idx_first; idx != idx_last; ++idx)
96 {
97 auto in = std::next(in_first, (*idx) * B);
98 for (int j = 0; j < B; ++j, ++in, ++out)
99 *out = *in;
100 }
101 });
102 };
103 }
104
109 auto get_unpack()
110 {
111 // Assignment is accumulation with an operation that keeps the
112 // received value
113 return get_unpack_op([](auto, auto received) { return received; });
114 }
115
121 template <typename BinaryOp>
122 auto get_unpack_op(BinaryOp op)
123 {
124 return [op, bs = _bs](typename ScatterContainer::const_iterator idx_first,
125 typename ScatterContainer::const_iterator idx_last,
126 const auto in_first, auto out_first)
127 {
128 // out[idx[i] * bs + j] = op(out[idx[i] * bs + j], in[i * bs + j])
129 dispatch_bs(bs,
130 [&](auto B)
131 {
132 auto in = in_first;
133 for (auto idx = idx_first; idx != idx_last; ++idx)
134 {
135 auto out = std::next(out_first, (*idx) * B);
136 for (int j = 0; j < B; ++j, ++in, ++out)
137 *out = op(*out, *in);
138 }
139 });
140 };
141 }

477template <class V>
478auto inner_product(const V& a, const V& b)
479{
480 using T = typename V::value_type;
481 const std::int32_t local_size = a.bs() * a.index_map()->size_local();
482 if (local_size != b.bs() * b.index_map()->size_local())
483 throw std::runtime_error("Incompatible vector sizes");
484
485 const T local = std::transform_reduce(
486 a.array().begin(), std::next(a.array().begin(), local_size),
487 b.array().begin(), static_cast<T>(0), std::plus{},
488 [](T a, T b) -> T
489 {
490 if constexpr (std::is_same<T, std::complex<double>>::value
491 or std::is_same<T, std::complex<float>>::value)
492 {
493 return std::conj(a) * b;
494 }
495 else
496 return a * b;
497 });
498
499 T result;
500 MPI_Allreduce(&local, &result, 1, dolfinx::MPI::mpi_t<T>, MPI_SUM,
501 a.index_map()->comm());
502 return result;
503}

508template <class V>
509auto squared_norm(const V& a)
510{
511 using T = typename V::value_type;
512 T result = inner_product(a, a);
513 return std::real(result);
514}

522template <class V>
523auto norm(const V& x, Norm type = Norm::l2)
524{
525 using T = typename V::value_type;
526 switch (type)
527 {
528 case Norm::l1:
529 {
530 std::int32_t size_local = x.bs() * x.index_map()->size_local();
531 using U = typename dolfinx::scalar_value_t<T>;
532 U local_l1 = std::accumulate(
533 x.array().begin(), std::next(x.array().begin(), size_local), U(0),
534 [](auto norm, auto x) { return norm + std::abs(x); });
535 U l1(0);
536 MPI_Allreduce(&local_l1, &l1, 1, MPI::mpi_t<U>, MPI_SUM,
537 x.index_map()->comm());
538 return l1;
539 }
540 case Norm::l2:
541 return std::sqrt(squared_norm(x));
542 case Norm::linf:
543 {
544 std::int32_t size_local = x.bs() * x.index_map()->size_local();
545 auto max_pos = std::max_element(
546 x.array().begin(), std::next(x.array().begin(), size_local),
547 [](T a, T b) { return std::norm(a) < std::norm(b); });
548 auto local_linf = std::abs(*max_pos);
549 decltype(local_linf) linf = 0;
550 MPI_Allreduce(&local_linf, &linf, 1, MPI::mpi_t<decltype(linf)>, MPI_MAX,
551 x.index_map()->comm());
552 return linf;
553 }
554 default:
555 throw std::runtime_error("Norm type not supported");
556 }

565template <class V>
566void orthonormalize(std::vector<std::reference_wrapper<V>> basis)
567{
568 using T = typename V::value_type;
569 using U = typename dolfinx::scalar_value_t<T>;
570
571 // Loop over each vector in basis
572 for (std::size_t i = 0; i < basis.size(); ++i)
573 {
574 // Orthogonalize vector i with respect to previously orthonormalized
575 // vectors
576 V& bi = basis[i].get();
577 for (std::size_t j = 0; j < i; ++j)
578 {
579 const V& bj = basis[j].get();
580
581 // basis_i <- basis_i - dot_ij basis_j
582 auto dot_ij = inner_product(bi, bj);
583 std::ranges::transform(bj.array(), bi.array(), bi.array().begin(),
584 [dot_ij](auto xj, auto xi)
585 { return xi - dot_ij * xj; });
586 }
587
588 // Normalise basis function
589 auto norm = la::norm(bi, la::Norm::l2);
590 if (norm * norm < std::numeric_limits<U>::epsilon())
591 {
592 throw std::runtime_error(
593 "Linear dependency detected. Cannot orthogonalize.");
594 }
595 std::ranges::transform(bi.array(), bi.array().begin(),
596 [norm](auto x) { return x / norm; });
597 }
598}

608template <class V>
609bool is_orthonormal(
610 std::vector<std::reference_wrapper<const V>> basis,
611 dolfinx::scalar_value_t<typename V::value_type> eps = std::numeric_limits<
612 dolfinx::scalar_value_t<typename V::value_type>>::epsilon())
613{
614 using T = typename V::value_type;
615 for (std::size_t i = 0; i < basis.size(); i++)
616 {
617 for (std::size_t j = i; j < basis.size(); ++j)
618 {
619 T delta_ij = (i == j) ? T(1) : T(0);
620 auto dot_ij = inner_product(basis[i].get(), basis[j].get());
621 if (std::norm(delta_ij - dot_ij) > eps)
622 return false;
623 }
624 }
625
626 return true;
627}
