65 template <class F>
66 static void dispatch_bs(int bs, F&& f)
67 {
68 switch (bs)
69 {
70 case 1:
71 return f(std::integral_constant<int, 1>{});
72 case 2:
73 return f(std::integral_constant<int, 2>{});
74 case 3:
75 return f(std::integral_constant<int, 3>{});
76 default:
77 return f(bs);
78 }
79 }

85 auto get_pack()
86 {
87 return [bs = _bs](typename ScatterContainer::const_iterator idx_first,
88 typename ScatterContainer::const_iterator idx_last,
89 const auto in_first, auto out_first)
90 {
91 // out[i * bs + j] = in[idx[i] * bs + j]
92 dispatch_bs(bs,
93 [&](auto B)
94 {
95 auto out = out_first;
96 for (auto idx = idx_first; idx != idx_last; ++idx)
97 {
98 auto in = std::next(in_first, (*idx) * B);
99 for (int j = 0; j < B; ++j, ++in, ++out)
100 *out = *in;
101 }
102 });
103 };
104 }
105
110 auto get_unpack()
111 {
112 // Assignment is accumulation with an operation that keeps the
113 // received value
114 return get_unpack_op([](auto, auto received) { return received; });
115 }
116
122 template <typename BinaryOp>
123 auto get_unpack_op(BinaryOp op)
124 {
125 return [op, bs = _bs](typename ScatterContainer::const_iterator idx_first,
126 typename ScatterContainer::const_iterator idx_last,
127 const auto in_first, auto out_first)
128 {
129 // out[idx[i] * bs + j] = op(out[idx[i] * bs + j], in[i * bs + j])
130 dispatch_bs(bs,
131 [&](auto B)
132 {
133 auto in = in_first;
134 for (auto idx = idx_first; idx != idx_last; ++idx)
135 {
136 auto out = std::next(out_first, (*idx) * B);
137 for (int j = 0; j < B; ++j, ++in, ++out)
138 *out = op(*out, *in);
139 }
140 });
141 };
142 }

517template <class V>
518auto inner_product(const V& a, const V& b)
519{
520 using T = typename V::value_type;
521 const std::int32_t local_size = a.bs() * a.index_map()->size_local();
522 if (local_size != b.bs() * b.index_map()->size_local())
523 throw std::runtime_error("Incompatible vector sizes");
524
525 const T local = std::transform_reduce(
526 a.array().begin(), std::next(a.array().begin(), local_size),
527 b.array().begin(), static_cast<T>(0), std::plus{},
528 [](T a, T b) -> T
529 {
530 if constexpr (std::is_same<T, std::complex<double>>::value
531 or std::is_same<T, std::complex<float>>::value)
532 {
533 return std::conj(a) * b;
534 }
535 else
536 return a * b;
537 });
538
539 T result;
540 MPI_Allreduce(&local, &result, 1, dolfinx::MPI::mpi_t<T>, MPI_SUM,
541 a.index_map()->comm());
542 return result;
543}

548template <class V>
549auto squared_norm(const V& a)
550{
551 using T = typename V::value_type;
552 T result = inner_product(a, a);
553 return std::real(result);
554}

562template <class V>
563auto norm(const V& x, Norm type = Norm::l2)
564{
565 using T = typename V::value_type;
566 switch (type)
567 {
568 case Norm::l1:
569 {
570 std::int32_t size_local = x.bs() * x.index_map()->size_local();
571 using U = typename dolfinx::scalar_value_t<T>;
572 U local_l1 = std::accumulate(
573 x.array().begin(), std::next(x.array().begin(), size_local), U(0),
574 [](auto norm, auto x) { return norm + std::abs(x); });
575 U l1(0);
576 MPI_Allreduce(&local_l1, &l1, 1, MPI::mpi_t<U>, MPI_SUM,
577 x.index_map()->comm());
578 return l1;
579 }
580 case Norm::l2:
581 return std::sqrt(squared_norm(x));
582 case Norm::linf:
583 {
584 std::int32_t size_local = x.bs() * x.index_map()->size_local();
585 auto max_pos = std::max_element(
586 x.array().begin(), std::next(x.array().begin(), size_local),
587 [](T a, T b) { return std::norm(a) < std::norm(b); });
588 auto local_linf = std::abs(*max_pos);
589 decltype(local_linf) linf = 0;
590 MPI_Allreduce(&local_linf, &linf, 1, MPI::mpi_t<decltype(linf)>, MPI_MAX,
591 x.index_map()->comm());
592 return linf;
593 }
594 default:
595 throw std::runtime_error("Norm type not supported");
596 }

605template <class V>
606void orthonormalize(std::vector<std::reference_wrapper<V>> basis)
607{
608 using T = typename V::value_type;
609 using U = typename dolfinx::scalar_value_t<T>;
610
611 // Loop over each vector in basis
612 for (std::size_t i = 0; i < basis.size(); ++i)
613 {
614 // Orthogonalize vector i with respect to previously orthonormalized
615 // vectors
616 V& bi = basis[i].get();
617 for (std::size_t j = 0; j < i; ++j)
618 {
619 const V& bj = basis[j].get();
620
621 // basis_i <- basis_i - dot_ij basis_j
622 auto dot_ij = inner_product(bi, bj);
623 std::ranges::transform(bj.array(), bi.array(), bi.array().begin(),
624 [dot_ij](auto xj, auto xi)
625 { return xi - dot_ij * xj; });
626 }
627
628 // Normalise basis function
629 auto norm = la::norm(bi, la::Norm::l2);
630 if (norm * norm < std::numeric_limits<U>::epsilon())
631 {
632 throw std::runtime_error(
633 "Linear dependency detected. Cannot orthogonalize.");
634 }
635 std::ranges::transform(bi.array(), bi.array().begin(),
636 [norm](auto x) { return x / norm; });
637 }
638}

648template <class V>
649bool is_orthonormal(
650 std::vector<std::reference_wrapper<const V>> basis,
651 dolfinx::scalar_value_t<typename V::value_type> eps = std::numeric_limits<
652 dolfinx::scalar_value_t<typename V::value_type>>::epsilon())
653{
654 using T = typename V::value_type;
655 for (std::size_t i = 0; i < basis.size(); i++)
656 {
657 for (std::size_t j = i; j < basis.size(); ++j)
658 {
659 T delta_ij = (i == j) ? T(1) : T(0);
660 auto dot_ij = inner_product(basis[i].get(), basis[j].get());
661 if (std::norm(delta_ij - dot_ij) > eps)
662 return false;
663 }
664 }
665
666 return true;
667}
