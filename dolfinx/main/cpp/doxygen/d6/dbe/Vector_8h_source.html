66 template <class F>
67 static void dispatch_bs(int bs, F&& f)
68 {
69 switch (bs)
70 {
71 case 1:
72 return f(std::integral_constant<int, 1>{});
73 case 2:
74 return f(std::integral_constant<int, 2>{});
75 case 3:
76 return f(std::integral_constant<int, 3>{});
77 default:
78 return f(bs);
79 }
80 }

86 auto get_pack()
87 {
88 return [bs = _bs](typename ScatterContainer::const_iterator idx_first,
89 typename ScatterContainer::const_iterator idx_last,
90 const auto in_first, auto out_first)
91 {
92 // out[i * bs + j] = in[idx[i] * bs + j]
93 dispatch_bs(bs,
94 [&](auto B)
95 {
96 auto out = out_first;
97 for (auto idx = idx_first; idx != idx_last; ++idx)
98 {
99 auto in = std::next(in_first, (*idx) * B);
100 for (int j = 0; j < B; ++j, ++in, ++out)
101 *out = *in;
102 }
103 });
104 };
105 }
106
111 auto get_unpack()
112 {
113 // Assignment is accumulation with an operation that keeps the
114 // received value
115 return get_unpack_op([](auto, auto received) { return received; });
116 }
117
123 template <typename BinaryOp>
124 auto get_unpack_op(BinaryOp op)
125 {
126 return [op, bs = _bs](typename ScatterContainer::const_iterator idx_first,
127 typename ScatterContainer::const_iterator idx_last,
128 const auto in_first, auto out_first)
129 {
130 // out[idx[i] * bs + j] = op(out[idx[i] * bs + j], in[i * bs + j])
131 dispatch_bs(bs,
132 [&](auto B)
133 {
134 auto in = in_first;
135 for (auto idx = idx_first; idx != idx_last; ++idx)
136 {
137 auto out = std::next(out_first, (*idx) * B);
138 for (int j = 0; j < B; ++j, ++in, ++out)
139 *out = op(*out, *in);
140 }
141 });
142 };
143 }

529template <class V>
530auto inner_product(const V& a, const V& b)
531{
532 using T = typename V::value_type;
533 const std::int32_t local_size = a.bs() * a.index_map()->size_local();
534 if (local_size != b.bs() * b.index_map()->size_local())
535 throw std::runtime_error("Incompatible vector sizes");
536
537 const T local = std::transform_reduce(
538 a.array().begin(), std::next(a.array().begin(), local_size),
539 b.array().begin(), static_cast<T>(0), std::plus{},
540 [](T a, T b) -> T
541 {
542 if constexpr (std::is_same<T, std::complex<double>>::value
543 or std::is_same<T, std::complex<float>>::value)
544 {
545 return std::conj(a) * b;
546 }
547 else
548 return a * b;
549 });
550
551 T result;
552 MPI_Allreduce(&local, &result, 1, dolfinx::MPI::mpi_t<T>, MPI_SUM,
553 a.index_map()->comm());
554 return result;
555}

560template <class V>
561auto squared_norm(const V& a)
562{
563 using T = typename V::value_type;
564 T result = inner_product(a, a);
565 return std::real(result);
566}

574template <class V>
575auto norm(const V& x, Norm type = Norm::l2)
576{
577 using T = typename V::value_type;
578 switch (type)
579 {
580 case Norm::l1:
581 {
582 std::int32_t size_local = x.bs() * x.index_map()->size_local();
583 using U = typename dolfinx::scalar_value_t<T>;
584 U local_l1 = std::accumulate(
585 x.array().begin(), std::next(x.array().begin(), size_local), U(0),
586 [](auto norm, auto x) { return norm + std::abs(x); });
587 U l1(0);
588 MPI_Allreduce(&local_l1, &l1, 1, MPI::mpi_t<U>, MPI_SUM,
589 x.index_map()->comm());
590 return l1;
591 }
592 case Norm::l2:
593 return std::sqrt(squared_norm(x));
594 case Norm::linf:
595 {
596 std::int32_t size_local = x.bs() * x.index_map()->size_local();
597 auto max_pos = std::max_element(
598 x.array().begin(), std::next(x.array().begin(), size_local),
599 [](T a, T b) { return std::norm(a) < std::norm(b); });
600 auto local_linf = std::abs(*max_pos);
601 decltype(local_linf) linf = 0;
602 MPI_Allreduce(&local_linf, &linf, 1, MPI::mpi_t<decltype(linf)>, MPI_MAX,
603 x.index_map()->comm());
604 return linf;
605 }
606 default:
607 throw std::runtime_error("Norm type not supported");
608 }

617template <class V>
618void orthonormalize(std::vector<std::reference_wrapper<V>> basis)
619{
620 using T = typename V::value_type;
621 using U = typename dolfinx::scalar_value_t<T>;
622
623 // Loop over each vector in basis
624 for (std::size_t i = 0; i < basis.size(); ++i)
625 {
626 // Orthogonalize vector i with respect to previously orthonormalized
627 // vectors
628 V& bi = basis[i].get();
629 for (std::size_t j = 0; j < i; ++j)
630 {
631 const V& bj = basis[j].get();
632
633 // basis_i <- basis_i - dot_ij basis_j
634 auto dot_ij = inner_product(bi, bj);
635 std::ranges::transform(bj.array(), bi.array(), bi.array().begin(),
636 [dot_ij](auto xj, auto xi)
637 { return xi - dot_ij * xj; });
638 }
639
640 // Normalise basis function
641 auto norm = la::norm(bi, la::Norm::l2);
642 if (norm * norm < std::numeric_limits<U>::epsilon())
643 {
644 throw std::runtime_error(
645 "Linear dependency detected. Cannot orthogonalize.");
646 }
647 std::ranges::transform(bi.array(), bi.array().begin(),
648 [norm](auto x) { return x / norm; });
649 }
650}

660template <class V>
661bool is_orthonormal(
662 std::vector<std::reference_wrapper<const V>> basis,
663 dolfinx::scalar_value_t<typename V::value_type> eps = std::numeric_limits<
664 dolfinx::scalar_value_t<typename V::value_type>>::epsilon())
665{
666 using T = typename V::value_type;
667 for (std::size_t i = 0; i < basis.size(); i++)
668 {
669 for (std::size_t j = i; j < basis.size(); ++j)
670 {
671 T delta_ij = (i == j) ? T(1) : T(0);
672 auto dot_ij = inner_product(basis[i].get(), basis[j].get());
673 if (std::norm(delta_ij - dot_ij) > eps)
674 return false;
675 }
676 }
677
678 return true;
679}
