62template <typename T>
63void reorder_list(std::span<T> list, std::span<const std::int32_t> nodemap)
64{
65 if (nodemap.empty())
66 return;
67
68 assert(list.size() % nodemap.size() == 0);
69 std::size_t degree = list.size() / nodemap.size();
70 const std::vector<T> orig(list.begin(), list.end());
71 for (std::size_t n = 0; n < nodemap.size(); ++n)
72 {
73 std::span links_old(orig.data() + n * degree, degree);
74 auto links_new = list.subspan(nodemap[n] * degree, degree);
75 std::ranges::copy(links_old, links_new.begin());
76 }
77}

92template <std::floating_point T>
93std::tuple<std::vector<std::int32_t>, std::vector<T>, std::vector<std::int32_t>>
94compute_vertex_coords_boundary(const mesh::Mesh<T>& mesh, int dim,
95 std::span<const std::int32_t> facets)
96{
97 auto topology = mesh.topology();
98 assert(topology);
99 const int tdim = topology->dim();
100 if (dim == tdim)
101 {
102 throw std::runtime_error(
103 "Cannot use mesh::locate_entities_boundary (boundary) for cells.");
104 }
105
106 // Build set of vertices on boundary and set of boundary entities
107 mesh.topology_mutable()->create_connectivity(tdim - 1, 0);
108 mesh.topology_mutable()->create_connectivity(tdim - 1, dim);
109 std::vector<std::int32_t> vertices, entities;
110 {
111 auto f_to_v = topology->connectivity(tdim - 1, 0);
112 assert(f_to_v);
113 auto f_to_e = topology->connectivity(tdim - 1, dim);
114 assert(f_to_e);
115 for (auto f : facets)
116 {
117 auto v = f_to_v->links(f);
118 vertices.insert(vertices.end(), v.begin(), v.end());
119 auto e = f_to_e->links(f);
120 entities.insert(entities.end(), e.begin(), e.end());
121 }
122
123 // Build vector of boundary vertices
124 {
125 std::ranges::sort(vertices);
126 auto [unique_end, range_end] = std::ranges::unique(vertices);
127 vertices.erase(unique_end, range_end);
128 }
129
130 {
131 std::ranges::sort(entities);
132 auto [unique_end, range_end] = std::ranges::unique(entities);
133 entities.erase(unique_end, range_end);
134 }
135 }
136
137 // Get geometry data
138 auto x_dofmap = mesh.geometry().dofmaps().front();
139 std::span<const T> x_nodes = mesh.geometry().x();
140
141 // Get all vertex 'node' indices
142 mesh.topology_mutable()->create_connectivity(0, tdim);
143 mesh.topology_mutable()->create_connectivity(tdim, 0);
144 auto v_to_c = topology->connectivity(0, tdim);
145 assert(v_to_c);
146 auto c_to_v = topology->connectivity(tdim, 0);
147 assert(c_to_v);
148 std::vector<T> x_vertices(3 * vertices.size(), -1.0);
149 std::vector<std::int32_t> vertex_to_pos(v_to_c->num_nodes(), -1);
150 for (std::size_t i = 0; i < vertices.size(); ++i)
151 {
152 const std::int32_t v = vertices[i];
153
154 // Get first cell and find position
155 const std::int32_t c = v_to_c->links(v).front();
156 auto cell_vertices = c_to_v->links(c);
157 auto it = std::ranges::find(cell_vertices, v);
158 assert(it != cell_vertices.end());
159 const std::size_t local_pos
160 = std::ranges::distance(cell_vertices.begin(), it);
161
162 auto dofs = md::submdspan(x_dofmap, c, md::full_extent);
163 for (std::size_t j = 0; j < 3; ++j)
164 x_vertices[j * vertices.size() + i] = x_nodes[3 * dofs[local_pos] + j];
165 vertex_to_pos[v] = i;
166 }
167
168 return {std::move(entities), std::move(x_vertices), std::move(vertex_to_pos)};
169}

283template <std::floating_point T>
284std::vector<T> h(const Mesh<T>& mesh, std::span<const std::int32_t> entities,
285 int dim)
286{
287 if (entities.empty())
288 return std::vector<T>();
289 if (dim == 0)
290 return std::vector<T>(entities.size(), 0);
291
292 // Get the geometry dofs for the vertices of each entity
293 const auto [vertex_xdofs, xdof_shape]
294 = entities_to_geometry(mesh, dim, entities, false);
295
296 // Get the geometry coordinate
297 std::span<const T> x = mesh.geometry().x();
298
299 // Function to compute the length of (p0 - p1)
300 auto delta_norm = [](auto&& p0, auto&& p1)
301 {
302 T norm = 0;
303 for (std::size_t i = 0; i < 3; ++i)
304 norm += (p0[i] - p1[i]) * (p0[i] - p1[i]);
305 return std::sqrt(norm);
306 };
307
308 // Compute greatest distance between any to vertices
309 assert(dim > 0);
310 std::vector<T> h(entities.size(), 0);
311 for (std::size_t e = 0; e < entities.size(); ++e)
312 {
313 // Get geometry 'dof' for each vertex of entity e
314 std::span<const std::int32_t> e_vertices(
315 vertex_xdofs.data() + e * xdof_shape[1], xdof_shape[1]);
316
317 // Compute maximum distance between any two vertices
318 for (std::size_t i = 0; i < e_vertices.size(); ++i)
319 {
320 std::span<const T, 3> p0(x.data() + 3 * e_vertices[i], 3);
321 for (std::size_t j = i + 1; j < e_vertices.size(); ++j)
322 {
323 std::span<const T, 3> p1(x.data() + 3 * e_vertices[j], 3);
324 h[e] = std::max(h[e], delta_norm(p0, p1));
325 }
326 }
327 }
328
329 return h;
330}

335template <std::floating_point T>
336std::vector<T> cell_normals(const Mesh<T>& mesh, int dim,
337 std::span<const std::int32_t> entities)
338{
339 if (entities.empty())
340 return std::vector<T>();
341
342 auto topology = mesh.topology();
343 assert(topology);
344 if (topology->cell_type() == CellType::prism and dim == 2)
345 {
346 throw std::runtime_error(
347 "Cell normal computation for prism cells not yet supported.");
348 }
349
350 const int gdim = mesh.geometry().dim();
351 const CellType type = cell_entity_type(topology->cell_type(), dim, 0);
352
353 // Find geometry nodes for topology entities
354 std::span<const T> x = mesh.geometry().x();
355 const auto [geometry_entities, eshape]
356 = entities_to_geometry(mesh, dim, entities, false);
357
358 std::vector<T> n(entities.size() * 3);
359 switch (type)
360 {
361 case CellType::interval:
362 {
363 if (gdim > 2)
364 throw std::invalid_argument("Interval cell normal undefined in 3D.");
365 for (std::size_t i = 0; i < entities.size(); ++i)
366 {
367 // Get the two vertices as points
368 std::array vertices{geometry_entities[i * eshape[1]],
369 geometry_entities[i * eshape[1] + 1]};
370 std::array p = {std::span<const T, 3>(x.data() + 3 * vertices[0], 3),
371 std::span<const T, 3>(x.data() + 3 * vertices[1], 3)};
372
373 // Define normal by rotating tangent counter-clockwise
374 std::array<T, 3> t;
375 std::ranges::transform(p[1], p[0], t.begin(),
376 [](auto x, auto y) { return x - y; });
377
378 T norm = std::sqrt(t[0] * t[0] + t[1] * t[1]);
379 std::span<T, 3> ni(n.data() + 3 * i, 3);
380 ni[0] = -t[1] / norm;
381 ni[1] = t[0] / norm;
382 ni[2] = 0.0;
383 }
384 return n;
385 }
386 case CellType::triangle:
387 {
388 for (std::size_t i = 0; i < entities.size(); ++i)
389 {
390 // Get the three vertices as points
391 std::array vertices = {geometry_entities[i * eshape[1] + 0],
392 geometry_entities[i * eshape[1] + 1],
393 geometry_entities[i * eshape[1] + 2]};
394 std::array p = {std::span<const T, 3>(x.data() + 3 * vertices[0], 3),
395 std::span<const T, 3>(x.data() + 3 * vertices[1], 3),
396 std::span<const T, 3>(x.data() + 3 * vertices[2], 3)};
397
398 // Compute (p1 - p0) and (p2 - p0)
399 std::array<T, 3> dp1, dp2;
400 std::ranges::transform(p[1], p[0], dp1.begin(),
401 [](auto x, auto y) { return x - y; });
402 std::ranges::transform(p[2], p[0], dp2.begin(),
403 [](auto x, auto y) { return x - y; });
404
405 // Define cell normal via cross product of first two edges
406 std::array<T, 3> ni = math::cross(dp1, dp2);
407 T norm = std::sqrt(ni[0] * ni[0] + ni[1] * ni[1] + ni[2] * ni[2]);
408 std::ranges::transform(ni, std::next(n.begin(), 3 * i),
409 [norm](auto x) { return x / norm; });
410 }
411
412 return n;
413 }
414 case CellType::quadrilateral:
415 {
416 // TODO: check
417 for (std::size_t i = 0; i < entities.size(); ++i)
418 {
419 // Get the three vertices as points
420 std::array vertices = {geometry_entities[i * eshape[1] + 0],
421 geometry_entities[i * eshape[1] + 1],
422 geometry_entities[i * eshape[1] + 2]};
423 std::array p = {std::span<const T, 3>(x.data() + 3 * vertices[0], 3),
424 std::span<const T, 3>(x.data() + 3 * vertices[1], 3),
425 std::span<const T, 3>(x.data() + 3 * vertices[2], 3)};
426
427 // Compute (p1 - p0) and (p2 - p0)
428 std::array<T, 3> dp1, dp2;
429 std::ranges::transform(p[1], p[0], dp1.begin(),
430 [](auto x, auto y) { return x - y; });
431 std::ranges::transform(p[2], p[0], dp2.begin(),
432 [](auto x, auto y) { return x - y; });
433
434 // Define cell normal via cross product of first two edges
435 std::array<T, 3> ni = math::cross(dp1, dp2);
436 T norm = std::sqrt(ni[0] * ni[0] + ni[1] * ni[1] + ni[2] * ni[2]);
437 std::ranges::transform(ni, std::next(n.begin(), 3 * i),
438 [norm](auto x) { return x / norm; });
439 }
440
441 return n;
442 }
443 default:
444 throw std::invalid_argument(
445 "cell_normal not supported for this cell type.");
446 }

452template <std::floating_point T>
453std::vector<T> compute_midpoints(const Mesh<T>& mesh, int dim,
454 std::span<const std::int32_t> entities)
455{
456 if (entities.empty())
457 return std::vector<T>();
458
459 std::span<const T> x = mesh.geometry().x();
460
461 // Build map from entity -> geometry dof
462 const auto [e_to_g, eshape]
463 = entities_to_geometry(mesh, dim, entities, false);
464
465 std::vector<T> x_mid(entities.size() * 3, 0);
466 for (std::size_t e = 0; e < entities.size(); ++e)
467 {
468 std::span<T, 3> p(x_mid.data() + 3 * e, 3);
469 std::span<const std::int32_t> rows(e_to_g.data() + e * eshape[1],
470 eshape[1]);
471 for (auto row : rows)
472 {
473 std::span<const T, 3> xg(x.data() + 3 * row, 3);
474 std::ranges::transform(p, xg, p.begin(),
475 [size = rows.size()](auto x, auto y)
476 { return x + y / size; });
477 }
478 }
479
480 return x_mid;
481}

489template <std::floating_point T>
490std::pair<std::vector<T>, std::array<std::size_t, 2>>
491compute_vertex_coords(const mesh::Mesh<T>& mesh)
492{
493 auto topology = mesh.topology();
494 assert(topology);
495 const int tdim = topology->dim();
496
497 // Create entities and connectivities
498
499 // Get all vertex 'node' indices
500 const std::int32_t num_vertices = topology->index_map(0)->size_local()
501 + topology->index_map(0)->num_ghosts();
502
503 std::vector<std::int32_t> vertex_to_node(num_vertices);
504 for (int cell_type_idx = 0,
505 num_cell_types = topology->entity_types(tdim).size();
506 cell_type_idx < num_cell_types; ++cell_type_idx)
507 {
508 auto x_dofmap = mesh.geometry().dofmaps().at(cell_type_idx);
509 auto c_to_v = topology->connectivity({tdim, cell_type_idx}, {0, 0});
510 assert(c_to_v);
511 for (int c = 0; c < c_to_v->num_nodes(); ++c)
512 {
513 auto x_dofs = md::submdspan(x_dofmap, c, md::full_extent);
514 auto vertices = c_to_v->links(c);
515 for (std::size_t i = 0; i < vertices.size(); ++i)
516 vertex_to_node[vertices[i]] = x_dofs[i];
517 }
518 }
519
520 // Pack coordinates of vertices
521 std::span<const T> x_nodes = mesh.geometry().x();
522 std::vector<T> x_vertices(3 * vertex_to_node.size(), 0.0);
523 for (std::size_t i = 0; i < vertex_to_node.size(); ++i)
524 {
525 std::int32_t pos = 3 * vertex_to_node[i];
526 for (std::size_t j = 0; j < 3; ++j)
527 x_vertices[j * vertex_to_node.size() + i] = x_nodes[pos + j];
528 }
529
530 return {std::move(x_vertices), {3, vertex_to_node.size()}};
531}

557template <std::floating_point T, MarkerFn<T> U>
558std::vector<std::int32_t> locate_entities(const Mesh<T>& mesh, int dim,
559 U marker, int entity_type_idx)
560{
561
562 using cmdspan3x_t
563 = md::mdspan<const T, md::extents<std::size_t, 3, md::dynamic_extent>>;
564
565 // Run marker function on vertex coordinates
566 const auto [xdata, xshape] = impl::compute_vertex_coords(mesh);
567
568 cmdspan3x_t x(xdata.data(), xshape);
569 const std::vector<std::int8_t> marked = marker(x);
570 if (marked.size() != x.extent(1))
571 throw std::runtime_error("Length of array of markers is wrong.");
572
573 auto topology = mesh.topology();
574 assert(topology);
575 const int tdim = topology->dim();
576
577 mesh.topology_mutable()->create_entities(dim);
578 if (dim < tdim)
579 mesh.topology_mutable()->create_connectivity(dim, 0);
580
581 // Iterate over entities of dimension 'dim' to build vector of marked
582 // entities
583 auto e_to_v = topology->connectivity({dim, entity_type_idx}, {0, 0});
584 assert(e_to_v);
585 std::vector<std::int32_t> entities;
586 for (int e = 0; e < e_to_v->num_nodes(); ++e)
587 {
588 // Iterate over entity vertices
589 bool all_vertices_marked = true;
590 for (std::int32_t v : e_to_v->links(e))
591 {
592 if (!marked[v])
593 {
594 all_vertices_marked = false;
595 break;
596 }
597 }
598
599 if (all_vertices_marked)
600 entities.push_back(e);
601 }
602
603 return entities;
604}

730template <std::floating_point T>
731std::pair<std::vector<std::int32_t>, std::array<std::size_t, 2>>
732entities_to_geometry(const Mesh<T>& mesh, int dim,
733 std::span<const std::int32_t> entities,
734 bool permute = false)
735{
736 auto topology = mesh.topology();
737 assert(topology);
738 CellType cell_type = topology->cell_type();
739 if ((cell_type == CellType::prism or cell_type == CellType::pyramid)
740 and dim == 2)
741 {
742 throw std::runtime_error("mesh::entities_to_geometry for prism/pyramid "
743 "cell facets not yet supported.");
744 }
745
746 const int tdim = topology->dim();
747 const Geometry<T>& geometry = mesh.geometry();
748 auto xdofs = geometry.dofmaps().front();
749
750 // Get the DOF layout and the number of DOFs per entity
751 const fem::CoordinateElement<T>& coord_ele = geometry.cmaps().front();
752 const fem::ElementDofLayout layout = coord_ele.create_dof_layout();
753 const std::size_t num_entity_dofs = layout.entity_closure_dofs(dim, 0).size();
754 std::vector<std::int32_t> entity_xdofs;
755 entity_xdofs.reserve(entities.size() * num_entity_dofs);
756 std::array<std::size_t, 2> eshape{entities.size(), num_entity_dofs};
757
758 // Get the element's closure DOFs
759 const std::vector<std::vector<std::vector<int>>>& closure_dofs_all
760 = layout.entity_closure_dofs_all();
761
762 // Special case when dim == tdim (cells)
763 if (dim == tdim)
764 {
765 for (std::int32_t c : entities)
766 {
767 // Extract degrees of freedom
768 auto x_c = md::submdspan(xdofs, c, md::full_extent);
769 for (std::int32_t entity_dof : closure_dofs_all[tdim][0])
770 entity_xdofs.push_back(x_c[entity_dof]);
771 }
772
773 return {std::move(entity_xdofs), eshape};
774 }
775
776 assert(dim != tdim);
777
778 auto e_to_c = topology->connectivity(dim, tdim);
779 if (!e_to_c)
780 {
781 throw std::runtime_error(std::format(
782 "Entity-to-cell connectivity has not been computed. Missing dims "
783 "{}->{}",
784 dim, tdim));
785 }
786
787 auto c_to_e = topology->connectivity(tdim, dim);
788 if (!c_to_e)
789 {
790 throw std::runtime_error(std::format(
791 "Cell-to-entity connectivity has not been computed. Missing dims "
792 "{}->{}",
793 tdim, dim));
794 }
795
796 // Get the cell info, which is needed to permute the closure dofs
797 std::span<const std::uint32_t> cell_info;
798 if (permute)
799 cell_info = std::span(mesh.topology()->get_cell_permutation_info());
800
801 for (std::int32_t e : entities)
802 {
803 // Get a cell connected to the entity
804 assert(!e_to_c->links(e).empty());
805 std::int32_t c = e_to_c->links(e).front();
806
807 // Get the local index of the entity
808 std::span<const std::int32_t> cell_entities = c_to_e->links(c);
809 auto it = std::find(cell_entities.begin(), cell_entities.end(), e);
810 assert(it != cell_entities.end());
811 std::size_t local_entity = std::ranges::distance(cell_entities.begin(), it);
812
813 // Cell sub-entities must be permuted so that their local
814 // orientation agrees with their global orientation
815 std::vector<std::int32_t> closure_dofs(closure_dofs_all[dim][local_entity]);
816 if (permute)
817 {
818 mesh::CellType entity_type
819 = mesh::cell_entity_type(cell_type, dim, local_entity);
820 coord_ele.permute_subentity_closure(closure_dofs, cell_info[c],
821 entity_type, local_entity);
822 }
823
824 // Extract degrees of freedom
825 auto x_c = md::submdspan(xdofs, c, md::full_extent);
826 for (std::int32_t entity_dof : closure_dofs)
827 entity_xdofs.push_back(x_c[entity_dof]);
828 }
829
830 return {std::move(entity_xdofs), eshape};

938template <std::floating_point T>
939std::vector<double> cell_centroids_local(
940 std::span<const int> num_vertices_per_cell,
941 const std::vector<std::span<const std::int64_t>>& cells,
942 const boost::unordered_flat_map<std::int64_t, std::size_t>& node_to_pos,
943 std::span<const T> coords, int gdim)
944{
945 std::size_t num_cells = 0;
946 for (std::size_t i = 0; i < cells.size(); ++i)
947 num_cells += cells[i].size() / num_vertices_per_cell[i];
948 std::vector<double> centroid(gdim * num_cells, 0);
949
950 std::size_t c0 = 0;
951 for (std::size_t i = 0; i < cells.size(); ++i)
952 {
953 const int nv = num_vertices_per_cell[i];
954 const double w = 1.0 / nv;
955 for (std::size_t c = 0; c < cells[i].size() / nv; ++c)
956 {
957 for (int v = 0; v < nv; ++v)
958 {
959 auto it = node_to_pos.find(cells[i][nv * c + v]);
960 assert(it != node_to_pos.end());
961 std::size_t pos = it->second;
962 for (int d = 0; d < gdim; ++d)
963 centroid[gdim * (c0 + c) + d] += w * coords[gdim * pos + d];
964 }
965 }
966
967 c0 += cells[i].size() / nv;
968 }
969
970 return centroid;
971}

996template <std::floating_point T>
997std::vector<double>
998compute_cell_centroids(MPI_Comm comm,
999 std::span<const int> num_vertices_per_cell,
1000 const std::vector<std::span<const std::int64_t>>& cells,
1001 MPI_Comm commg, std::span<const T> x, int gdim)
1002{
1003 // Vertices of the cells on this rank, sorted and with duplicates
1004 // removed, and the coordinates for them
1005 std::vector<std::int64_t> nodes;
1006 {
1007 std::size_t size = 0;
1008 for (std::span<const std::int64_t> c : cells)
1009 size += c.size();
1010 nodes.reserve(size);
1011 for (std::span<const std::int64_t> c : cells)
1012 nodes.insert(nodes.end(), c.begin(), c.end());
1013 dolfinx::radix_sort(nodes);
1014 auto [unique_end, range_end] = std::ranges::unique(nodes);
1015 nodes.erase(unique_end, range_end);
1016 }
1017 const std::vector<T> coords
1018 = dolfinx::MPI::distribute_data(comm, nodes, commg, x, gdim);
1019
1020 // Hash map from global vertex index to its position in `nodes` (and
1021 // so its row in `coords`), turning the many repeated cell-vertex
1022 // lookups below into an O(1) average lookup rather than an
1023 // O(log(nodes.size())) binary search each time -- most vertices are
1024 // shared by several cells, so the same key is looked up repeatedly.
1025 boost::unordered_flat_map<std::int64_t, std::size_t> node_to_pos;
1026 node_to_pos.reserve(nodes.size());
1027 for (std::size_t i = 0; i < nodes.size(); ++i)
1028 node_to_pos.emplace(nodes[i], i);
1029
1030 return cell_centroids_local(num_vertices_per_cell, cells, node_to_pos,
1031 std::span<const T>(coords), gdim);
1032}

1082template <std::floating_point T>
1083std::tuple<std::vector<std::vector<std::int64_t>>,
1084 std::vector<std::vector<std::int64_t>>,
1085 std::vector<std::vector<int>>>
1086partition_cells(MPI_Comm comm, MPI_Comm commt,
1087 const std::vector<std::span<const std::int64_t>>& cells,
1088 const std::vector<CellType>& celltypes,
1089 const std::vector<fem::ElementDofLayout>& doflayouts,
1090 bool p1_geometry, const graph::Partitioner& partitioner,
1091 bool ghosting,
1092 std::optional<std::int32_t> max_facet_to_cell_links,
1093 int num_threads, MPI_Comm commg, std::span<const T> x,
1094 std::array<std::size_t, 2> xshape)
1095{
1096 const std::int32_t num_cell_types = cells.size();
1097 std::vector<std::vector<std::int64_t>> cells1(num_cell_types);
1098 std::vector<std::vector<std::int64_t>> original_idx1(num_cell_types);
1099 std::vector<std::vector<int>> ghost_owners(num_cell_types);
1100 if (graph::has_partitioner(partitioner.fn))
1101 {
1102 spdlog::info("Using partitioner with cell data ({} cell types)",
1103 num_cell_types);
1104 graph::AdjacencyList<std::int32_t> dest(0);
1105 std::exception_ptr error;
1106
1107 // Geometric data can be distributed on ranks that do not participate in
1108 // topology partitioning. Gather cell centroids collectively over `comm`
1109 // so that every rank in `commg` participates in the coordinate exchange.
1110 std::vector<double> centroid;
1111 const bool needs_centroids
1112 = std::holds_alternative<graph::geom_partition_fn>(partitioner.fn)
1113 or std::holds_alternative<graph::hybrid_partition_fn>(partitioner.fn);
1114 std::vector<std::vector<std::int64_t>> topology(num_cell_types);
1115 std::vector<std::span<const std::int64_t>> topology_view(num_cell_types);
1116 if (needs_centroids or commt != MPI_COMM_NULL)
1117 {
1118 for (std::int32_t i = 0; i < num_cell_types; ++i)
1119 {
1120 if (p1_geometry)
1121 topology_view[i] = cells[i];
1122 else
1123 {
1124 topology[i] = extract_topology(celltypes[i], doflayouts[i], cells[i]);
1125 topology_view[i] = topology[i];
1126 }
1127 }
1128 }
1129
1130 if (needs_centroids)
1131 {
1132 std::vector<int> num_vertices_per_cell;
1133 num_vertices_per_cell.reserve(celltypes.size());
1134 std::ranges::transform(celltypes,
1135 std::back_inserter(num_vertices_per_cell),
1136 [](CellType c) { return num_cell_vertices(c); });
1137 centroid = compute_cell_centroids(comm, num_vertices_per_cell,
1138 topology_view, commg, x, xshape[1]);
1139 }
1140
1141 if (std::holds_alternative<graph::geom_partition_fn>(partitioner.fn))
1142 {
1143 try_locally(
1144 [&]
1145 {
1146 int size = dolfinx::MPI::size(comm);
1147 const auto& p = std::get<graph::geom_partition_fn>(partitioner.fn);
1148 dest = graph::regular_adjacency_list(
1149 p(comm, size, std::span<const double>(centroid), xshape[1],
1150 partitioner.node_weights),
1151 1);
1152 },
1153 error);
1154 }
1155
1156 if (commt != MPI_COMM_NULL)
1157 {
1158 // A partitioner such as graph::parmetis::geom_partitioner (which
1159 // requires nparts to equal the number of ranks calling it) can
1160 // throw only on the ranks with commt != MPI_COMM_NULL, which may
1161 // be a strict subset of comm (e.g. cells built on rank 0 only).
1162 // mpi_check below turns that into a comm-wide decision before any
1163 // rank reaches the graph::build::distribute collective, or a throw
1164 // here would leave the rest of comm blocked on it forever.
1165 try_locally(
1166 [&]
1167 {
1168 int size = dolfinx::MPI::size(comm);
1169 // Shared by the graph::partition_fn and
1170 // graph::hybrid_partition_fn alternatives below: neither has any
1171 // other way to obtain the mesh dual graph.
1172 auto dual_graph = [&]() -> graph::AdjacencyList<std::int64_t>
1173 {
1174 return build_dual_graph(commt, celltypes, topology_view,
1175 max_facet_to_cell_links, num_threads);
1176 };
1177
1178 dest = std::visit(
1179 [&](const auto& p) -> graph::AdjacencyList<std::int32_t>
1180 {
1181 using P = std::decay_t<decltype(p)>;
1182 if constexpr (std::is_same_v<P, graph::hybrid_partition_fn>)
1183 {
1184 return p(commt, size, dual_graph(),
1185 std::span<const double>(centroid),
1186 partitioner.node_weights, std::nullopt, ghosting);
1187 }
1188 else if constexpr (std::is_same_v<P, graph::partition_fn>)
1189 {
1190 return p(commt, size, dual_graph(),
1191 partitioner.node_weights, std::nullopt, ghosting);
1192 }
1193 else
1194 return dest;
1195 },
1196 partitioner.fn);
1197 },
1198 error);
1199 }
1200
1201 mpi_check(comm, "Cell partitioning", error);
1202
1203 std::int32_t cell_offset = 0;
1204 for (std::int32_t i = 0; i < num_cell_types; ++i)
1205 {
1206 std::size_t num_cell_nodes = doflayouts[i].num_dofs();
1207 if (cells[i].size() % num_cell_nodes != 0)
1208 {
1209 throw std::runtime_error("Cell array size is not a multiple of the "
1210 "number of nodes per cell.");
1211 }
1212 std::size_t num_cells = cells[i].size() / num_cell_nodes;
1213
1214 // Extract destination AdjacencyList for this cell type
1215 std::vector<std::int32_t> offsets_i(
1216 std::next(dest.offsets().begin(), cell_offset),
1217 std::next(dest.offsets().begin(), cell_offset + num_cells + 1));
1218 std::vector<std::int32_t> data_i(
1219 std::next(dest.array().begin(), offsets_i.front()),
1220 std::next(dest.array().begin(), offsets_i.back()));
1221 const std::int32_t offset_0 = offsets_i.front();
1222 std::ranges::transform(offsets_i, offsets_i.begin(),
1223 [offset_0](std::int32_t j)
1224 { return j - offset_0; });
1225 graph::AdjacencyList<std::int32_t> dest_i(data_i, offsets_i);
1226 cell_offset += num_cells;
1227
1228 // Distribute cells (topology, includes higher-order 'nodes') to
1229 // destination rank
1230 std::vector<int> src_ranks;
1231 std::tie(cells1[i], src_ranks, original_idx1[i], ghost_owners[i])
1232 = graph::build::distribute(comm, cells[i],
1233 {num_cells, num_cell_nodes}, dest_i);
1234 spdlog::debug("Got {} cells from distribution", cells1[i].size());
1235 }
1236 }
1237 else // No partitioning: keep cells on their current rank
1238 {
1239 // Count cells of each type on this rank. Each cell still needs a
1240 // globally unique index (assigned below), even though it is not
1241 // being redistributed, and the counts are needed first to size
1242 // `original_idx1` and to determine this rank's share via the
1243 // exclusive scan that follows.
1244 std::int64_t num_owned = 0;
1245 for (std::int32_t i = 0; i < num_cell_types; ++i)
1246 {
1247 cells1[i] = std::vector<std::int64_t>(cells[i].begin(), cells[i].end());
1248 std::int32_t num_cell_nodes = doflayouts[i].num_dofs();
1249 if (cells1[i].size() % num_cell_nodes != 0)
1250 {
1251 throw std::runtime_error("Cell array size is not a multiple of the "
1252 "number of nodes per cell.");
1253 }
1254 original_idx1[i].resize(cells1[i].size() / num_cell_nodes);
1255 num_owned += original_idx1[i].size();
1256 }
1257
1258 // Assign a globally unique index to each cell. `global_offset`
1259 // starts as the number of cells owned by lower-ranked processes
1260 // (from the exclusive scan), and is advanced by each cell type's
1261 // count in turn so that the numbering is contiguous across cell
1262 // types too.
1263 std::int64_t global_offset = 0;
1264 MPI_Exscan(&num_owned, &global_offset, 1, MPI_INT64_T, MPI_SUM, comm);
1265 for (std::int32_t i = 0; i < num_cell_types; ++i)
1266 {
1267 std::iota(original_idx1[i].begin(), original_idx1[i].end(),
1268 global_offset);
1269 global_offset += original_idx1[i].size();
1270 }
1271 }
1272
1273 return {std::move(cells1), std::move(original_idx1), std::move(ghost_owners)};
1274}

1331template <typename U>
1332Mesh<typename std::remove_reference_t<typename U::value_type>> create_mesh(
1333 MPI_Comm comm, MPI_Comm commt,
1334 std::vector<std::span<const std::int64_t>> cells,
1335 const std::vector<fem::CoordinateElement<
1336 typename std::remove_reference_t<typename U::value_type>>>& elements,
1337 MPI_Comm commg, const U& x, std::array<std::size_t, 2> xshape,
1338 const graph::Partitioner& partitioner, GhostMode ghost_mode,
1339 std::optional<std::int32_t> max_facet_to_cell_links, int num_threads,
1340 const graph::Reorder& reorder_fn = graph::Reorder{})
1341{
1342 using T = typename std::remove_reference_t<typename U::value_type>;
1343
1344 if (cells.size() != elements.size())
1345 throw std::runtime_error("Number of cell arrays and elements must match.");
1346 std::vector<CellType> celltypes;
1347 std::ranges::transform(elements, std::back_inserter(celltypes),
1348 [](auto& e) { return e.cell_shape(); });
1349 std::vector<fem::ElementDofLayout> doflayouts;
1350 std::ranges::transform(elements, std::back_inserter(doflayouts),
1351 [](auto& e) { return e.create_dof_layout(); });
1352
1353 // Note: `extract_topology` extracts topology data, i.e. just the
1354 // vertices. For other elements the filtered lists may have 'gaps',
1355 // i.e. the indices might not be contiguous.
1356 //
1357 // For 'P1 geometry' the extraction is the identity operator, and cell
1358 // node data is used directly as cell topology. This avoids copies of
1359 // the (large) cell array, and lets the geometry node indices be taken
1360 // from the topology vertices rather than re-derived by sorting the
1361 // cell array (see below).
1362 const bool p1_geometry = std::ranges::all_of(
1363 std::views::iota(std::size_t(0), elements.size()),
1364 [&celltypes, &doflayouts](std::size_t i)
1365 { return is_vertex_dof_layout(celltypes[i], doflayouts[i]); });
1366
1367 const std::int32_t num_cell_types = cells.size();
1368
1369 // Partition cells across ranks of `comm` (or, if `partitioner` is not
1370 // callable, keep them on their current rank and just assign each a
1371 // globally unique index)
1372 const bool ghosting = (ghost_mode != GhostMode::none);
1373 auto [cells1, original_idx1, ghost_owners] = impl::partition_cells(
1374 comm, commt, cells, celltypes, doflayouts, p1_geometry, partitioner,
1375 ghosting, max_facet_to_cell_links, num_threads, commg,
1376 std::span<const T>(x), xshape);
1377
1378 // Extract cell 'topology', i.e. extract the vertices for each cell
1379 // and discard any 'higher-order' nodes. `cells1_v_storage` is empty
1380 // for 'P1 geometry', where `cells1_v` views `cells1` directly.
1381 std::vector<std::vector<std::int64_t>> cells1_v_storage(num_cell_types);
1382 std::vector<std::span<std::int64_t>> cells1_v(num_cell_types);
1383 for (std::int32_t i = 0; i < num_cell_types; ++i)
1384 {
1385 if (p1_geometry)
1386 cells1_v[i] = cells1[i];
1387 else
1388 {
1389 cells1_v_storage[i]
1390 = extract_topology(celltypes[i], doflayouts[i], cells1[i]);
1391 cells1_v[i] = cells1_v_storage[i];
1392 }
1393
1394 spdlog::info("Extract basic topology: {}->{}", cells1[i].size(),
1395 cells1_v[i].size());
1396 }
1397
1398 // Re-order cells and get boundary vertices. The re-ordering is done
1399 // on the cell topology, i.e. the vertex indices, and the higher-order
1400 // nodes are re-ordered accordingly.
1401 // Centroids are required only by a callable graph::reorder_geom_fn.
1402 // As elsewhere, `reorder_fn` is assumed to hold the same alternative
1403 // on every rank, so an empty function skips the collective exchange
1404 // everywhere; impl::reorder_cells then throws, and the failure is
1405 // made comm-wide by try_locally/mpi_check below.
1406 std::vector<double> cell_centroids;
1407 if (const graph::reorder_geom_fn* fn
1408 = std::get_if<graph::reorder_geom_fn>(&reorder_fn);
1409 fn and *fn)
1410 {
1411 std::vector<int> num_cell_vertices;
1412 num_cell_vertices.reserve(celltypes.size());
1413 std::ranges::transform(celltypes, std::back_inserter(num_cell_vertices),
1414 [](CellType c)
1415 { return mesh::num_cell_vertices(c); });
1416
1417 std::vector<std::span<const std::int64_t>> cells1_v_owned;
1418 cells1_v_owned.reserve(cells1_v.size());
1419 for (std::size_t i = 0; i < cells1_v.size(); ++i)
1420 {
1421 const int num_cell_vertices_i = num_cell_vertices[i];
1422 const std::size_t num_owned_cells
1423 = cells1_v[i].size() / num_cell_vertices_i - ghost_owners[i].size();
1424 cells1_v_owned.emplace_back(cells1_v[i].data(),
1425 num_owned_cells * num_cell_vertices_i);
1426 }
1427
1428 cell_centroids
1429 = impl::compute_cell_centroids(comm, num_cell_vertices, cells1_v_owned,
1430 commg, std::span<const T>(x), xshape[1]);
1431 }
1432
1433 std::vector<std::int64_t> boundary_v;
1434 std::exception_ptr error;
1435 impl::try_locally(
1436 [&]
1437 {
1438 boundary_v = impl::reorder_cells(reorder_fn, cell_centroids, xshape[1],
1439 max_facet_to_cell_links, celltypes,
1440 doflayouts, ghost_owners, cells1,
1441 cells1_v, original_idx1, num_threads);
1442 },
1443 error);
1444 impl::mpi_check(comm, "Cell reordering", error);
1445
1446 spdlog::debug("Got {} boundary vertices", boundary_v.size());
1447
1448 // Create Topology
1449 std::vector<std::span<const std::int64_t>> cells1_v_span(cells1_v.begin(),
1450 cells1_v.end());
1451 std::vector<std::span<const std::int64_t>> original_idx1_span;
1452 std::ranges::transform(original_idx1, std::back_inserter(original_idx1_span),
1453 [](auto& c) { return std::span(c); });
1454 std::vector<std::span<const int>> ghost_owners_span;
1455 std::ranges::transform(ghost_owners, std::back_inserter(ghost_owners_span),
1456 [](auto& c) { return std::span(c); });
1457
1458 // Note: `vertex_index` holds the sorted input global indices of the
1459 // topology vertices, which for 'P1 geometry' are exactly the geometry
1460 // node indices required below.
1461 auto [topology, vertex_index] = mesh::impl::create_topology(
1462 comm, celltypes, cells1_v_span, original_idx1_span, ghost_owners_span,
1463 boundary_v, num_threads);
1464
1465 // Create connectivities required higher-order geometries for creating
1466 // a Geometry object
1467 for (int i = 0; i < num_cell_types; ++i)
1468 {
1469 const auto& entity_dofs = doflayouts[i].entity_dofs_all();
1470 for (int dim = 1; dim < topology.dim(); ++dim)
1471 {
1472 // Accumulate count of all dofs on this dimension
1473 int dim_sum
1474 = std::accumulate(entity_dofs[dim].begin(), entity_dofs[dim].end(), 0,
1475 [](int c, auto v) { return c + v.size(); });
1476
1477 spdlog::debug("Counting entity dofs, dim={}: {}", dim, dim_sum);
1478 if (dim_sum > 0)
1479 topology.create_entities(dim);
1480 }
1481
1482 if (elements[i].needs_dof_permutations())
1483 topology.create_entity_permutations();
1484 }
1485
1486 // Cell 'node' indices (global), as a single flat array. This is
1487 // `cells1` for a single cell type, and concatenated otherwise.
1488 std::vector<std::int64_t> nodes2_storage;
1489 std::span<const std::int64_t> nodes2;
1490 if (num_cell_types == 1)
1491 nodes2 = cells1.front();
1492 else
1493 {
1494 std::size_t size = 0;
1495 for (const std::vector<std::int64_t>& c : cells1)
1496 size += c.size();
1497 nodes2_storage.reserve(size);
1498 for (const std::vector<std::int64_t>& c : cells1)
1499 nodes2_storage.insert(nodes2_storage.end(), c.begin(), c.end());
1500 nodes2 = nodes2_storage;
1501 }
1502
1503 // Sorted list of unique (global) node indices. For 'P1 geometry' the
1504 // nodes are the vertices, which `create_topology` has already sorted
1505 // and made unique, so re-deriving them from the (much larger) cell
1506 // array is avoided.
1507 std::vector<std::int64_t> nodes1;
1508 if (p1_geometry)
1509 nodes1 = std::move(vertex_index);
1510 else
1511 {
1512 nodes1.assign(nodes2.begin(), nodes2.end());
1513 dolfinx::radix_sort(nodes1);
1514 auto [unique_end, range_end] = std::ranges::unique(nodes1);
1515 nodes1.erase(unique_end, range_end);
1516 }
1517
1518 std::vector coords
1519 = dolfinx::MPI::distribute_data(comm, nodes1, commg, x, xshape[1]);
1520
1521 // Create geometry object
1522 Geometry geometry
1523 = create_geometry(topology, elements, nodes1, nodes2, coords, xshape[1]);
1524
1525 return Mesh(comm, std::make_shared<Topology>(std::move(topology)),
1526 std::move(geometry));
1527}

1645 std::span<const std::int32_t> subentity_to_entity)
1646{

1655 = entities_to_geometry(mesh, dim, subentity_to_entity, true).first;

1658 std::ranges::sort(sub_x_dofs);

1660 sub_x_dofs.erase(unique_end, range_end);
1661
1662 // Get the sub-geometry dofs owned by this process
1663 auto x_index_map = geometry.index_map();
1664 assert(x_index_map);
1665
1666 std::shared_ptr<common::IndexMap> sub_x_dof_index_map;
1667 std::vector<std::int32_t> subx_to_x_dofmap;
1668 {

1670 *x_index_map, sub_x_dofs, common::IndexMapOrder::any, true);

1672 subx_to_x_dofmap = std::move(new_to_old);

1679 for (std::int32_t i = 0; i < sub_num_x_dofs; ++i)
1680 {
1681 std::copy_n(std::next(x.begin(), 3 * subx_to_x_dofmap[i]), 3,
1682 std::next(sub_x.begin(), 3 * i));
1683 }

1687 x_index_map->size_local() + x_index_map->num_ghosts(), -1);
1688 for (std::size_t i = 0; i < subx_to_x_dofmap.size(); ++i)
1689 x_to_subx_dof_map[subx_to_x_dofmap[i]] = i;

1692 std::vector<std::int32_t> sub_x_dofmap;
1693 sub_x_dofmap.reserve(x_indices.size());

1695 [&x_to_subx_dof_map](auto x_dof)
1696 {
1697 assert(x_to_subx_dof_map[x_dof] != -1);
1698 return x_to_subx_dof_map[x_dof];
1699 });

1706 // basis functions

1717 [&igi](auto sub_x_dof) { return igi[sub_x_dof]; });

1719 // Create geometry
1720 return {Geometry(
1721 sub_x_dof_index_map,
1722 std::vector<std::vector<std::int32_t>>{std::move(sub_x_dofmap)},
1723 {sub_cmap}, std::move(sub_x), geometry.dim(), std::move(sub_igi)),

1773template <typename T>
1774MeshTags<T> transfer_meshtags_to_submesh(
1775 const MeshTags<T>& tags,
1776 std::shared_ptr<const dolfinx::mesh::Topology> submesh_topology,
1777 const EntityMap& vertex_map, const EntityMap& cell_map)
1778{
1779 int tag_dim = tags.dim();
1780 int submesh_tdim = submesh_topology->dim();
1781 auto topology = tags.topology();
1782 if (tag_dim > submesh_tdim)
1783 {
1784 throw std::runtime_error("Tag dimension must be less than or equal to "
1785 "submesh dimension");
1786 }
1787 std::shared_ptr<const dolfinx::common::IndexMap> sub_cell_imap
1788 = submesh_topology->index_map(submesh_tdim);
1789 if (!sub_cell_imap)
1790 {
1791 throw std::runtime_error(
1792 std::format("Entities of dimension {} does not exist in mesh topology.",
1793 submesh_tdim));
1794 }
1795
1796 // Create a map from parent entity to submesh cell
1797 std::int32_t submesh_num_cells
1798 = sub_cell_imap->size_local() + sub_cell_imap->num_ghosts();
1799 auto sub_cells = std::ranges::views::iota(0, submesh_num_cells);
1800 std::vector<std::int32_t> sub_cell_to_parent_entity
1801 = cell_map.sub_topology_to_topology(sub_cells, false);
1802
1803 // Create a full lookup for all cells on the parent mesh, as the tag can have
1804 // entities that are not in the submesh
1805 auto parent_entity_imap = topology->index_map(submesh_tdim);
1806 if (!parent_entity_imap)
1807 {
1808 throw std::runtime_error(std::format(
1809 "Entities of dimension {} does not exist in parent mesh topology.",
1810 submesh_tdim));
1811 }
1812 std::size_t num_parent_entities
1813 = parent_entity_imap->size_local() + parent_entity_imap->num_ghosts();
1814 std::vector<std::int32_t> parent_entity_to_sub_cell(num_parent_entities, -1);
1815 for (std::size_t i = 0; i < sub_cell_to_parent_entity.size(); ++i)
1816 parent_entity_to_sub_cell[sub_cell_to_parent_entity[i]]
1817 = static_cast<std::int32_t>(i);
1818
1819 // Get map from submesh vertex to parent vertex
1820 std::vector<std::int32_t> sub_to_parent_vertex;
1821 {
1822 auto sub_vertex_map = submesh_topology->index_map(0);
1823 std::int32_t num_sub_vertices
1824 = sub_vertex_map->size_local() + sub_vertex_map->num_ghosts();
1825 auto sub_vertices = std::ranges::views::iota(0, num_sub_vertices);
1826
1827 sub_to_parent_vertex
1828 = vertex_map.sub_topology_to_topology(sub_vertices, false);
1829 }
1830 // Access various connectivity maps
1831 auto sub_e_to_v = submesh_topology->connectivity(tag_dim, 0);
1832 auto sub_c_to_e = submesh_topology->connectivity(submesh_tdim, tag_dim);
1833 auto sub_entity_imap = submesh_topology->index_map(tag_dim);
1834 auto e_to_v = topology->connectivity(tag_dim, 0);
1835 std::shared_ptr<const dolfinx::graph::AdjacencyList<std::int32_t>>
1836 e_to_sub_cell = nullptr;
1837 if (tag_dim != submesh_tdim)
1838 {
1839 e_to_sub_cell = topology->connectivity(tag_dim, submesh_tdim);
1840 if (!e_to_sub_cell)
1841 {
1842 throw std::runtime_error(
1843 std::format("Missing connectivity between {} and {} in parent mesh",
1844 tag_dim, submesh_tdim));
1845 }
1846 }
1847
1848 if (!sub_e_to_v)
1849 {
1850 throw std::runtime_error(std::format(
1851 "Missing connectivity between {} and {} in submesh", tag_dim, 0));
1852 }
1853 if (!sub_c_to_e)
1854 {
1855 throw std::runtime_error(
1856 std::format("Missing connectivity between {} and {} in submesh",
1857 submesh_tdim, tag_dim));
1858 }
1859 if (!sub_entity_imap)
1860 {
1861 throw std::runtime_error(std::format(
1862 "Entities of dimension {} does not exist in submesh topology.",
1863 tag_dim));
1864 }
1865 if (!e_to_v)
1866 {
1867 throw std::runtime_error(
1868 std::format("Missing connectivity between {} and 0", tag_dim));
1869 }
1870
1871 // Prepare sub entity to parent map
1872 std::size_t num_sub_entities
1873 = sub_entity_imap->size_local() + sub_entity_imap->num_ghosts();
1874 constexpr T max_val = std::numeric_limits<T>::max();
1875 std::vector<T> submesh_values(num_sub_entities, max_val);
1876 std::vector<std::int32_t> submesh_indices(num_sub_entities);
1877 std::iota(submesh_indices.begin(), submesh_indices.end(), 0);
1878
1879 std::span<const std::int32_t> tagged_entities = tags.indices();
1880 std::span<const T> tagged_values = tags.values();
1881 // For each entity in the tag, find all cells of the submesh connected to this
1882 // entity
1883 for (std::size_t i = 0; i < tagged_entities.size(); ++i)
1884 {
1885 auto find_and_map_sub_entity
1886 = [tag_dim, submesh_tdim, &e_to_v, &parent_entity_to_sub_cell,
1887 &sub_to_parent_vertex, &sub_e_to_v, &sub_c_to_e,
1888 &e_to_sub_cell](std::int32_t entity)
1889 {
1890 // Fast exit if the tag dimension is the same as the submesh dimension,
1891 // as we can directly map the parent entity to the submesh cell
1892 if (tag_dim == submesh_tdim)
1893 return parent_entity_to_sub_cell[entity];
1894
1895 // Given an entity in the parent meshtag, find all submesh-cells that are
1896 // entities in parent mesh that contain this entity.
1897 auto entity_vertices = e_to_v->links(entity);
1898 auto parent_sub_cells = e_to_sub_cell->links(entity);
1899 auto submesh_cells
1900 = parent_sub_cells
1901 | std::views::transform([&parent_entity_to_sub_cell](auto c)
1902 { return parent_entity_to_sub_cell[c]; })
1903 | std::views::filter([](auto sub_cell) { return sub_cell != -1; });
1904 for (auto sub_cell : submesh_cells)
1905 {
1906 for (auto sub_entity : sub_c_to_e->links(sub_cell))
1907 {
1908 // Convert submesh entity vertices to parent vertices
1909 auto parent_vertices
1910 = sub_e_to_v->links(sub_entity)
1911 | std::views::transform([&sub_to_parent_vertex](auto v)
1912 { return sub_to_parent_vertex[v]; });
1913
1914 // Check if all parent vertices of the submesh entity are in the
1915 // parent entity
1916 bool entity_matches = std::ranges::all_of(
1917 parent_vertices,
1918 [&entity_vertices](auto p_v)
1919 {
1920 // With C++23 this can use std::ranges::contains
1921 return std::ranges::find(entity_vertices, p_v)
1922 != std::ranges::end(entity_vertices);
1923 });
1924
1925 // If a match is found, apply values and exit the lambda immediately
1926 if (entity_matches)
1927 return sub_entity;
1928 }
1929 }
1930 return -1;
1931 };
1932
1933 // Execute the search for the current entity
1934 std::int32_t sub_entity = find_and_map_sub_entity(tagged_entities[i]);
1935 if (sub_entity != -1)
1936 submesh_values[sub_entity] = tagged_values[i];
1937 }
1938
1939 // Filter out the entities that were never mapped (values still equal max)
1940 std::vector<std::int32_t> filtered_indices;
1941 std::vector<T> filtered_values;
1942 filtered_indices.reserve(num_sub_entities);
1943 filtered_values.reserve(num_sub_entities);
1944 for (std::size_t i = 0; i < submesh_values.size(); ++i)
1945 {
1946 if (submesh_values[i] != max_val)
1947 {
1948 filtered_indices.push_back(submesh_indices[i]);
1949 filtered_values.push_back(submesh_values[i]);
1950 }
1951 }
1952 filtered_indices.shrink_to_fit();
1953 filtered_values.shrink_to_fit();
1954 MeshTags<T> new_meshtag(submesh_topology, tag_dim, filtered_indices,
1955 filtered_values, tags.name());
1956 return new_meshtag;
1957}
