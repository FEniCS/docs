55template <std::floating_point T>
56void check_matis_mesh(const mesh::Mesh<T>& mesh,
57 const std::optional<std::string>& type)
58{
59 if (!type or *type != MATIS)
60 return;
61
62 auto topology = mesh.topology();
63 assert(topology);
64 std::shared_ptr<const common::IndexMap> cells
65 = topology->index_map(topology->dim());
66 assert(cells);
67 if (cells->num_ghosts() > 0)
68 {
69 throw std::invalid_argument(
70 "A MATIS matrix requires one non-overlapping subdomain per process. "
71 "Create the mesh with GhostMode::none.");
72 }
73}

85template <std::floating_point T>
86Mat create_matrix(const Form<PetscScalar, T>& a,
87 std::optional<std::string> type = std::nullopt)
88{
89 impl::check_matis_mesh(*a.mesh(), type);
90 la::SparsityPattern pattern = fem::create_sparsity_pattern(a);
91 pattern.finalize();
92 return la::petsc::create_matrix(a.mesh()->comm(), pattern, type);
93}

107template <std::floating_point T>
108Mat create_matrix_block(
109 const std::vector<std::vector<const Form<PetscScalar, T>*>>& a,
110 std::optional<std::string> type = std::nullopt)
111{
112 // Extract and check row/column ranges
113 std::array<std::vector<std::shared_ptr<const FunctionSpace<T>>>, 2> V
114 = fem::common_function_spaces(extract_function_spaces(a));
115 std::array<std::vector<int>, 2> bs_dofs;
116 for (std::size_t i = 0; i < 2; ++i)
117 {
118 for (auto& _V : V[i])
119 bs_dofs[i].push_back(_V->dofmap()->bs());
120 }
121
122 // Build sparsity pattern for each block
123 std::shared_ptr<const mesh::Mesh<T>> mesh;
124 std::vector<std::vector<std::unique_ptr<la::SparsityPattern>>> patterns(
125 V[0].size());
126 for (std::size_t row = 0; row < V[0].size(); ++row)
127 {
128 for (std::size_t col = 0; col < V[1].size(); ++col)
129 {
130 if (const Form<PetscScalar, T>* form = a[row][col]; form)
131 {
132 patterns[row].push_back(std::make_unique<la::SparsityPattern>(
133 create_sparsity_pattern(*form)));
134 if (!mesh)
135 mesh = form->mesh();
136 }
137 else
138 patterns[row].push_back(nullptr);
139 }
140 }
141
142 if (!mesh)
143 throw std::invalid_argument("Could not find a Mesh.");
144
145 impl::check_matis_mesh(*mesh, type);
146
147 // Compute offsets for the fields
148 std::array<std::vector<std::pair<
149 std::reference_wrapper<const common::IndexMap>, int>>,
150 2>
151 maps;
152 for (std::size_t d = 0; d < 2; ++d)
153 {
154 for (auto& space : V[d])
155 {
156 maps[d].emplace_back(*space->dofmap()->index_map,
157 space->dofmap()->index_map_bs());
158 }
159 }
160
161 // Create merged sparsity pattern
162 std::vector<std::vector<const la::SparsityPattern*>> p(V[0].size());
163 for (std::size_t row = 0; row < V[0].size(); ++row)
164 for (std::size_t col = 0; col < V[1].size(); ++col)
165 p[row].push_back(patterns[row][col].get());
166
167 la::SparsityPattern pattern(mesh->comm(), p, maps, bs_dofs);
168 pattern.finalize();
169
170 // TODO: Index map concatenation has already been computed inside
171 // the SparsityPattern constructor, but we also need it here to
172 // build the PETSc local-to-global map. Compute outside and pass
173 // into SparsityPattern constructor.
174
175 // Create row and column local-to-global maps (field0, field1, field2,
176 // etc), i.e. ghosts of field0 appear before owned indices of field1
177 std::array<std::vector<PetscInt>, 2> _maps;
178 for (int d = 0; d < 2; ++d)
179 {
180 if (d == 1 and V[0] == V[1])
181 {
182 // Row and column spaces are identical, so the concatenated
183 // index map for d=1 is identical to the one already computed
184 // for d=0 -- reuse it rather than paying for a second,
185 // communication-heavy call to stack_index_maps.
186 _maps[1] = _maps[0];
187 continue;
188 }
189
190 const std::vector<
191 std::pair<std::reference_wrapper<const common::IndexMap>, int>>& map
192 = maps[d];
193 std::vector<PetscInt>& _map = _maps[d];
194
195 // Concatenate the block index map in the row and column directions
196 const auto [rank_offset, local_offset, ghosts, _]
197 = common::stack_index_maps(map);
198 const std::size_t num_ghosts
199 = std::accumulate(ghosts.begin(), ghosts.end(), std::size_t(0),
200 [](std::size_t n, auto& g) { return n + g.size(); });
201 _map.reserve(local_offset.back() + num_ghosts);
202 for (std::size_t f = 0; f < map.size(); ++f)
203 {
204 auto offset = local_offset[f];
205 const common::IndexMap& imap = map[f].first.get();
206 int bs = map[f].second;
207 auto owned
208 = std::views::iota(std::int32_t(0), bs * imap.size_local())
209 | std::views::transform([offset, rank_offset](std::int32_t i)
210 { return i + rank_offset + offset; });
211 _map.insert(_map.end(), owned.begin(), owned.end());
212 _map.insert(_map.end(), ghosts[f].begin(), ghosts[f].end());
213 }
214 }
215
216 // Create the local-to-global maps on the mesh communicator. MATIS
217 // requires them to share the matrix communicator
218 ISLocalToGlobalMapping l2g0 = nullptr, l2g1 = nullptr;
219 common::petsc::check(
220 ISLocalToGlobalMappingCreate(mesh->comm(), 1, _maps[0].size(),
221 _maps[0].data(), PETSC_COPY_VALUES, &l2g0),
222 "ISLocalToGlobalMappingCreate");
223 if (V[0] != V[1])
224 {
225 common::petsc::check(
226 ISLocalToGlobalMappingCreate(mesh->comm(), 1, _maps[1].size(),
227 _maps[1].data(), PETSC_COPY_VALUES, &l2g1),
228 "ISLocalToGlobalMappingCreate");
229 }
230
231 // Initialise the matrix. MATIS builds its preallocation from the
232 // maps, so they are passed to the constructor
233 Mat A = la::petsc::create_matrix(mesh->comm(), pattern, type, l2g0,
234 l2g1 ? l2g1 : l2g0);
235 common::petsc::check(ISLocalToGlobalMappingDestroy(&l2g0),
236 "ISLocalToGlobalMappingDestroy");
237 if (l2g1)
238 {
239 common::petsc::check(ISLocalToGlobalMappingDestroy(&l2g1),
240 "ISLocalToGlobalMappingDestroy");
241 }
242
243 return A;
244}

249template <std::floating_point T>
250Mat create_matrix_nest(
251 const std::vector<std::vector<const Form<PetscScalar, T>*>>& a,
252 std::optional<std::vector<std::vector<std::optional<std::string>>>> types)
253{
254 if (a.empty())
255 throw std::invalid_argument(
256 "Rectangular array of forms must be non-empty.");
257
258 // Extract and check row/column ranges
259 auto V = fem::common_function_spaces(extract_function_spaces(a));
260
261 // Loop over each form and create matrix
262 int rows = a.size();
263 int cols = a.front().size();
264 std::vector<Mat> mats(rows * cols, nullptr);
265 std::shared_ptr<const mesh::Mesh<T>> mesh;
266 for (int i = 0; i < rows; ++i)
267 {
268 for (int j = 0; j < cols; ++j)
269 {
270 if (const Form<PetscScalar, T>* form = a[i][j]; form)
271 {
272 if (types)
273 mats[i * cols + j] = create_matrix(*form, types->at(i).at(j));
274 else
275 mats[i * cols + j] = create_matrix(*form, std::nullopt);
276 mesh = form->mesh();
277 }
278 }
279 }
280
281 if (!mesh)
282 throw std::invalid_argument("Could not find a Mesh.");
283
284 // Initialise block (MatNest) matrix. On error, destroy the
285 // already-created sub-matrices in `mats` before propagating, since
286 // the nest (which would otherwise take joint ownership of them) was
287 // never successfully assembled.
288 Mat A;
289 try
290 {
291 common::petsc::check(MatCreate(mesh->comm(), &A), "MatCreate");
292 common::petsc::check(MatSetType(A, MATNEST), "MatSetType");
293 common::petsc::check(
294 MatNestSetSubMats(A, rows, nullptr, cols, nullptr, mats.data()),
295 "MatNestSetSubMats");
296 common::petsc::check(MatSetUp(A), "MatSetUp");
297 }
298 catch (...)
299 {
300 for (Mat& m : mats)
301 if (m)
302 common::petsc::check(MatDestroy(&m), "MatDestroy");
303 throw;
304 }
305
306 // De-reference Mat objects
307 for (Mat& m : mats)
308 if (m)
309 common::petsc::check(MatDestroy(&m), "MatDestroy");
310
311 return A;
312}

423template <std::floating_point T>
424void apply_lifting(
425 Vec b,
426 const std::vector<
427 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>& a,
428 const std::vector<std::span<const PetscScalar>>& constants,
429 const std::vector<std::map<std::pair<IntegralType, int>,
430 std::pair<std::span<const PetscScalar>, int>>>&
431 coeffs,
432 const std::vector<std::span<const std::int8_t>>& bc_markers1,
433 const std::vector<std::span<const PetscScalar>>& bc_values1,
434 const std::vector<Vec>& x0, PetscScalar alpha)
435{
436 if (!x0.empty() and x0.size() != a.size())
437 throw std::invalid_argument("Mismatch between x0 and a in apply_lifting.");
438
439 Vec b_local;
440 common::petsc::check(VecGhostGetLocalForm(b, &b_local),
441 "VecGhostGetLocalForm");
442 PetscInt n = 0;
443 common::petsc::check(VecGetSize(b_local, &n), "VecGetSize");
444 PetscScalar* array = nullptr;
445 common::petsc::check(VecGetArray(b_local, &array), "VecGetArray");
446 std::span<PetscScalar> _b(array, n);
447
448 if (x0.empty())
449 {
450 fem::apply_lifting(_b, a, constants, coeffs, bc_markers1, bc_values1, {},
451 alpha);
452 }
453 else
454 {
455 std::vector<std::span<const PetscScalar>> x0_ref;
456 std::vector<Vec> x0_local(a.size());
457 std::vector<const PetscScalar*> x0_array(a.size());
458 for (std::size_t i = 0; i < a.size(); ++i)
459 {
460 assert(x0[i]);
461 common::petsc::check(VecGhostGetLocalForm(x0[i], &x0_local[i]),
462 "VecGhostGetLocalForm");
463 PetscInt n0 = 0;
464 common::petsc::check(VecGetSize(x0_local[i], &n0), "VecGetSize");
465 common::petsc::check(VecGetArrayRead(x0_local[i], &x0_array[i]),
466 "VecGetArrayRead");
467 x0_ref.emplace_back(x0_array[i], n0);
468 }
469
470 fem::apply_lifting(_b, a, constants, coeffs, bc_markers1, bc_values1,
471 x0_ref, alpha);
472
473 for (std::size_t i = 0; i < x0_local.size(); ++i)
474 {
475 common::petsc::check(VecRestoreArrayRead(x0_local[i], &x0_array[i]),
476 "VecRestoreArrayRead");
477 common::petsc::check(VecGhostRestoreLocalForm(x0[i], &x0_local[i]),
478 "VecGhostRestoreLocalForm");
479 }
480 }
481
482 common::petsc::check(VecRestoreArray(b_local, &array), "VecRestoreArray");
483 common::petsc::check(VecGhostRestoreLocalForm(b, &b_local),
484 "VecGhostRestoreLocalForm");
485}

517template <std::floating_point T>
518void apply_lifting(
519 Vec b,
520 const std::vector<
521 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>& a,
522 const std::vector<
523 std::vector<std::reference_wrapper<const DirichletBC<PetscScalar, T>>>>&
524 bcs1,
525 const std::vector<Vec>& x0, PetscScalar alpha)
526{
527 std::vector<std::map<std::pair<IntegralType, int>,
528 std::pair<std::vector<PetscScalar>, int>>>
529 coeffs;
530 std::vector<std::vector<PetscScalar>> constants;
531 for (const auto& _a : a)
532 {
533 if (_a)
534 {
535 auto coefficients = allocate_coefficient_storage(_a->get());
536 pack_coefficients(_a->get(), coefficients);
537 coeffs.push_back(coefficients);
538 constants.push_back(pack_constants(_a->get()));
539 }
540 else
541 {
542 coeffs.emplace_back();
543 constants.emplace_back();
544 }
545 }
546
547 std::vector<std::span<const PetscScalar>> _constants(constants.begin(),
548 constants.end());
549 std::vector<std::map<std::pair<IntegralType, int>,
550 std::pair<std::span<const PetscScalar>, int>>>
551 _coeffs;
552 std::ranges::transform(coeffs, std::back_inserter(_coeffs),
553 [](auto& c) { return make_coefficients_span(c); });
554
555 auto [bc_markers1, bc_values1] = fem::impl::bc_lifting_data(a, bcs1);
556 apply_lifting(b, a, _constants, _coeffs,
557 std::vector<std::span<const std::int8_t>>(bc_markers1.begin(),
558 bc_markers1.end()),
559 std::vector<std::span<const PetscScalar>>(bc_values1.begin(),
560 bc_values1.end()),
561 x0, alpha);
562}

625template <std::floating_point T>
626void assign(const Vec x, Function<PetscScalar, T>& u)
627{
628 Vec x_local = nullptr;
629 common::petsc::check(VecGhostGetLocalForm(x, &x_local),
630 "VecGhostGetLocalForm");
631 PetscInt n = 0;
632 common::petsc::check(VecGetSize(x_local, &n), "VecGetSize");
633
634 std::span<PetscScalar> _u = u.x()->array();
635 if (static_cast<std::size_t>(n) != _u.size())
636 {
637 throw std::runtime_error(std::format(
638 "Vector has {} local entries, function has {}.", n, _u.size()));
639 }
640
641 const PetscScalar* array = nullptr;
642 common::petsc::check(VecGetArrayRead(x_local, &array), "VecGetArrayRead");
643 std::ranges::copy(std::span<const PetscScalar>(array, n), _u.begin());
644 common::petsc::check(VecRestoreArrayRead(x_local, &array),
645 "VecRestoreArrayRead");
646 common::petsc::check(VecGhostRestoreLocalForm(x, &x_local),
647 "VecGhostRestoreLocalForm");

676template <std::floating_point T>
677void assemble_operator(Mat A, const Form<PetscScalar, T>& a,
678 std::span<const std::int8_t> dof_marker0,
679 std::span<const std::int8_t> dof_marker1)
680{
681 common::petsc::check(MatZeroEntries(A), "MatZeroEntries");
682
683 // Block and dof indices coincide at block size 1. Avoid the generic
684 // blocked PETSc path, which expands these indices before insertion.
685 if (a.function_spaces()[0]->dofmap()->index_map_bs() == 1
686 and a.function_spaces()[1]->dofmap()->index_map_bs() == 1)
687 {
688 fem::assemble_matrix(la::petsc::Matrix::set_fn(A, ADD_VALUES), a,
689 dof_marker0, dof_marker1);
690 }
691 else
692 {
693 fem::assemble_matrix(la::petsc::Matrix::set_block_fn(A, ADD_VALUES), a,
694 dof_marker0, dof_marker1);
695 }
696
697 // The unit diagonal is only meaningful when the rows and columns are
698 // indexed by the same space
699 if (a.function_spaces()[0] == a.function_spaces()[1])
700 {
701 // Locally owned constrained rows
702 std::shared_ptr<const DofMap> dofmap0
703 = a.function_spaces()[0]->dofmaps().front();
704 // dof_marker0 is empty when no rows are constrained
705 const std::int32_t num_owned
706 = dof_marker0.empty()
707 ? 0
708 : dofmap0->index_map_bs() * dofmap0->index_map->size_local();
709 std::vector<std::int32_t> rows;
710 std::ranges::copy_if(
711 std::views::iota(std::int32_t(0), num_owned), std::back_inserter(rows),
712 [&dof_marker0](std::int32_t i) { return dof_marker0[i] != 0; });
713
714 // Assembly zeroed these rows, so adding sets the diagonal. Adding
715 // avoids a flush to switch from ADD_VALUES to INSERT_VALUES.
716 la::set_diagonal<PetscScalar>(la::petsc::Matrix::set_fn(A, ADD_VALUES),
717 rows);
718 }
719
720 common::petsc::check(MatAssemblyBegin(A, MAT_FINAL_ASSEMBLY),
721 "MatAssemblyBegin");
722 common::petsc::check(MatAssemblyEnd(A, MAT_FINAL_ASSEMBLY), "MatAssemblyEnd");
723}

751template <std::floating_point T>
752void assemble_residual(
753 const Vec x, Vec b, const Form<PetscScalar, T>& F,
754 const Form<PetscScalar, T>& J,
755 const std::vector<
756 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>& bcs,
757 Function<PetscScalar, T>& u)
758{
759 common::petsc::check(VecGhostUpdateBegin(x, INSERT_VALUES, SCATTER_FORWARD),
760 "VecGhostUpdateBegin");
761 common::petsc::check(VecGhostUpdateEnd(x, INSERT_VALUES, SCATTER_FORWARD),
762 "VecGhostUpdateEnd");
763 impl::assign(x, u);
764
765 // Zero the local form, as assembly accumulates into ghost entries
766 Vec b_local = nullptr;
767 common::petsc::check(VecGhostGetLocalForm(b, &b_local),
768 "VecGhostGetLocalForm");
769 common::petsc::check(VecZeroEntries(b_local), "VecZeroEntries");
770 common::petsc::check(VecGhostRestoreLocalForm(b, &b_local),
771 "VecGhostRestoreLocalForm");
772
773 assemble_vector(b, F);
774
775 // Constrained dof markers and values g on the trial space of J (owned
776 // and ghost), both empty if there are no bcs
777 const std::vector<
778 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>
779 a{J};
780 auto [bc_markers1, bc_values1] = fem::impl::bc_lifting_data(
781 a, std::vector<std::vector<
782 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>>{bcs});
783
784 // Lifting: b <- b + J (g - x)
785 const std::vector<PetscScalar> constants = pack_constants(J);
786 auto coeffs = allocate_coefficient_storage(J);
787 pack_coefficients(J, coeffs);
788 apply_lifting(b, a, {std::span<const PetscScalar>(constants)},
789 {make_coefficients_span(coeffs)},
790 {std::span<const std::int8_t>(bc_markers1.front())},
791 {std::span<const PetscScalar>(bc_values1.front())}, {x},
792 PetscScalar(-1));
793
794 common::petsc::check(VecGhostUpdateBegin(b, ADD_VALUES, SCATTER_REVERSE),
795 "VecGhostUpdateBegin");
796 common::petsc::check(VecGhostUpdateEnd(b, ADD_VALUES, SCATTER_REVERSE),
797 "VecGhostUpdateEnd");
798
799 // Set b = x - g on owned constrained dofs
800 if (!bcs.empty())
801 {
802 PetscScalar* b_array = nullptr;
803 common::petsc::check(VecGetArray(b, &b_array), "VecGetArray");
804 const PetscScalar* x_array = nullptr;
805 common::petsc::check(VecGetArrayRead(x, &x_array), "VecGetArrayRead");
806 for (auto& bc : bcs)
807 {
808 auto [dofs, owned] = bc.get().dof_indices();

856template <std::floating_point T>
857void assemble_jacobian(
858 const Vec x, Mat Jmat, Mat Pmat, const Form<PetscScalar, T>& J,
859 const std::vector<
860 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>& bcs,
861 Function<PetscScalar, T>& u, const Form<PetscScalar, T>* P = nullptr)
862{
863 // Checked before any collective call so that all ranks throw together
864 if (P and P->function_spaces() != J.function_spaces())
865 {
866 throw std::invalid_argument(
867 "Preconditioner form must have the same function spaces as the "
868 "Jacobian form.");
869 }
870
871 common::petsc::check(VecGhostUpdateBegin(x, INSERT_VALUES, SCATTER_FORWARD),
872 "VecGhostUpdateBegin");
873 common::petsc::check(VecGhostUpdateEnd(x, INSERT_VALUES, SCATTER_FORWARD),
874 "VecGhostUpdateEnd");
875 impl::assign(x, u);
876
877 // Markers depend only on the space, so they are built once here and
878 // the preconditioner, over the same spaces, shares them
879 std::array<std::vector<std::int8_t>, 2> markers;
880 auto [dof_marker0, dof_marker1]
881 = fem::impl::bc_dof_markers_pair(J, bcs, markers);
882
883 impl::assemble_operator(Jmat, J, dof_marker0, dof_marker1);
884 if (P)
885 impl::assemble_operator(Pmat, *P, dof_marker0, dof_marker1);
886}
