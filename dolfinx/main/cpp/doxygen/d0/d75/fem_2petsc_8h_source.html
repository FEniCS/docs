71template <std::floating_point T>
72Mat create_matrix_block(
73 const std::vector<std::vector<const Form<PetscScalar, T>*>>& a,
74 std::optional<std::string> type = std::nullopt)
75{
76 // Extract and check row/column ranges
77 std::array<std::vector<std::shared_ptr<const FunctionSpace<T>>>, 2> V
78 = fem::common_function_spaces(extract_function_spaces(a));
79 std::array<std::vector<int>, 2> bs_dofs;
80 for (std::size_t i = 0; i < 2; ++i)
81 {
82 for (auto& _V : V[i])
83 bs_dofs[i].push_back(_V->dofmap()->bs());
84 }
85
86 // Build sparsity pattern for each block
87 std::shared_ptr<const mesh::Mesh<T>> mesh;
88 std::vector<std::vector<std::unique_ptr<la::SparsityPattern>>> patterns(
89 V[0].size());
90 for (std::size_t row = 0; row < V[0].size(); ++row)
91 {
92 for (std::size_t col = 0; col < V[1].size(); ++col)
93 {
94 if (const Form<PetscScalar, T>* form = a[row][col]; form)
95 {
96 patterns[row].push_back(std::make_unique<la::SparsityPattern>(
97 create_sparsity_pattern(*form)));
98 if (!mesh)
99 mesh = form->mesh();
100 }
101 else
102 patterns[row].push_back(nullptr);
103 }
104 }
105
106 if (!mesh)
107 throw std::runtime_error("Could not find a Mesh.");
108
109 // Compute offsets for the fields
110 std::array<std::vector<std::pair<
111 std::reference_wrapper<const common::IndexMap>, int>>,
112 2>
113 maps;
114 for (std::size_t d = 0; d < 2; ++d)
115 {
116 for (auto& space : V[d])
117 {
118 maps[d].emplace_back(*space->dofmap()->index_map,
119 space->dofmap()->index_map_bs());
120 }
121 }
122
123 // Create merged sparsity pattern
124 std::vector<std::vector<const la::SparsityPattern*>> p(V[0].size());
125 for (std::size_t row = 0; row < V[0].size(); ++row)
126 for (std::size_t col = 0; col < V[1].size(); ++col)
127 p[row].push_back(patterns[row][col].get());
128
129 la::SparsityPattern pattern(mesh->comm(), p, maps, bs_dofs);
130 pattern.finalize();
131
132 // FIXME: Add option to pass customised local-to-global map to PETSc
133 // Mat constructor
134
135 // TODO: Index map concatenation has already been computed inside
136 // the SparsityPattern constructor, but we also need it here to
137 // build the PETSc local-to-global map. Compute outside and pass
138 // into SparsityPattern constructor.
139
140 // Initialise matrix
141 Mat A = la::petsc::create_matrix(mesh->comm(), pattern, type);
142
143 // Create row and column local-to-global maps (field0, field1, field2,
144 // etc), i.e. ghosts of field0 appear before owned indices of field1
145 std::array<std::vector<PetscInt>, 2> _maps;
146 for (int d = 0; d < 2; ++d)
147 {
148 if (d == 1 and V[0] == V[1])
149 {
150 // Row and column spaces are identical, so the concatenated
151 // index map for d=1 is identical to the one already computed
152 // for d=0 -- reuse it rather than paying for a second,
153 // communication-heavy call to stack_index_maps.
154 _maps[1] = _maps[0];
155 continue;
156 }
157
158 const std::vector<
159 std::pair<std::reference_wrapper<const common::IndexMap>, int>>& map
160 = maps[d];
161 std::vector<PetscInt>& _map = _maps[d];
162
163 // Concatenate the block index map in the row and column directions
164 const auto [rank_offset, local_offset, ghosts, _]
165 = common::stack_index_maps(map);
166 const std::size_t num_ghosts
167 = std::accumulate(ghosts.begin(), ghosts.end(), std::size_t(0),
168 [](std::size_t n, auto& g) { return n + g.size(); });
169 _map.reserve(local_offset.back() + num_ghosts);
170 for (std::size_t f = 0; f < map.size(); ++f)
171 {
172 auto offset = local_offset[f];
173 const common::IndexMap& imap = map[f].first.get();
174 int bs = map[f].second;
175 auto owned
176 = std::views::iota(std::int32_t(0), bs * imap.size_local())
177 | std::views::transform([offset, rank_offset](std::int32_t i)
178 { return i + rank_offset + offset; });
179 _map.insert(_map.end(), owned.begin(), owned.end());
180 _map.insert(_map.end(), ghosts[f].begin(), ghosts[f].end());
181 }
182 }
183
184 // Create PETSc local-to-global map/index sets and attach to matrix
185 ISLocalToGlobalMapping petsc_local_to_global0;
186 common::petsc::check(ISLocalToGlobalMappingCreate(
187 MPI_COMM_SELF, 1, _maps[0].size(), _maps[0].data(),
188 PETSC_COPY_VALUES, &petsc_local_to_global0),
189 "ISLocalToGlobalMappingCreate");
190 if (V[0] == V[1])
191 {
192 common::petsc::check(MatSetLocalToGlobalMapping(A, petsc_local_to_global0,
193 petsc_local_to_global0),
194 "MatSetLocalToGlobalMapping");
195 common::petsc::check(ISLocalToGlobalMappingDestroy(&petsc_local_to_global0),
196 "ISLocalToGlobalMappingDestroy");
197 }
198 else
199 {
200 ISLocalToGlobalMapping petsc_local_to_global1;
201 common::petsc::check(ISLocalToGlobalMappingCreate(
202 MPI_COMM_SELF, 1, _maps[1].size(), _maps[1].data(),
203 PETSC_COPY_VALUES, &petsc_local_to_global1),
204 "ISLocalToGlobalMappingCreate");
205 common::petsc::check(MatSetLocalToGlobalMapping(A, petsc_local_to_global0,
206 petsc_local_to_global1),
207 "MatSetLocalToGlobalMapping");
208 common::petsc::check(ISLocalToGlobalMappingDestroy(&petsc_local_to_global0),
209 "ISLocalToGlobalMappingDestroy");
210 common::petsc::check(ISLocalToGlobalMappingDestroy(&petsc_local_to_global1),
211 "ISLocalToGlobalMappingDestroy");
212 }
213
214 return A;
215}

220template <std::floating_point T>
221Mat create_matrix_nest(
222 const std::vector<std::vector<const Form<PetscScalar, T>*>>& a,
223 std::optional<std::vector<std::vector<std::optional<std::string>>>> types)
224{
225 if (a.empty())
226 throw std::runtime_error("Rectangular array of forms must be non-empty.");
227
228 // Extract and check row/column ranges
229 auto V = fem::common_function_spaces(extract_function_spaces(a));
230
231 // Loop over each form and create matrix
232 int rows = a.size();
233 int cols = a.front().size();
234 std::vector<Mat> mats(rows * cols, nullptr);
235 std::shared_ptr<const mesh::Mesh<T>> mesh;
236 for (int i = 0; i < rows; ++i)
237 {
238 for (int j = 0; j < cols; ++j)
239 {
240 if (const Form<PetscScalar, T>* form = a[i][j]; form)
241 {
242 if (types)
243 mats[i * cols + j] = create_matrix(*form, types->at(i).at(j));
244 else
245 mats[i * cols + j] = create_matrix(*form, std::nullopt);
246 mesh = form->mesh();
247 }
248 }
249 }
250
251 if (!mesh)
252 throw std::runtime_error("Could not find a Mesh.");
253
254 // Initialise block (MatNest) matrix. On error, destroy the
255 // already-created sub-matrices in `mats` before propagating, since
256 // the nest (which would otherwise take joint ownership of them) was
257 // never successfully assembled.
258 Mat A;
259 try
260 {
261 common::petsc::check(MatCreate(mesh->comm(), &A), "MatCreate");
262 common::petsc::check(MatSetType(A, MATNEST), "MatSetType");
263 common::petsc::check(
264 MatNestSetSubMats(A, rows, nullptr, cols, nullptr, mats.data()),
265 "MatNestSetSubMats");
266 common::petsc::check(MatSetUp(A), "MatSetUp");
267 }
268 catch (...)
269 {
270 for (Mat& m : mats)
271 if (m)
272 MatDestroy(&m);
273 throw;
274 }
275
276 // De-reference Mat objects
277 for (Mat& m : mats)
278 if (m)
279 MatDestroy(&m);
280
281 return A;
282}

389template <std::floating_point T>
390void apply_lifting(
391 Vec b,
392 std::vector<
393 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>
394 a,
395 const std::vector<std::span<const PetscScalar>>& constants,
396 const std::vector<std::map<std::pair<IntegralType, int>,
397 std::pair<std::span<const PetscScalar>, int>>>&
398 coeffs,
399 const std::vector<
400 std::vector<std::reference_wrapper<const DirichletBC<PetscScalar, T>>>>&
401 bcs1,
402 const std::vector<Vec>& x0, PetscScalar alpha)
403{
404 if (!x0.empty() and x0.size() != a.size())
405 throw std::runtime_error("Mismatch between x0 and a in apply_lifting.");
406
407 Vec b_local;
408 common::petsc::check(VecGhostGetLocalForm(b, &b_local),
409 "VecGhostGetLocalForm");
410 PetscInt n = 0;
411 common::petsc::check(VecGetSize(b_local, &n), "VecGetSize");
412 PetscScalar* array = nullptr;
413 common::petsc::check(VecGetArray(b_local, &array), "VecGetArray");
414 std::span<PetscScalar> _b(array, n);
415
416 if (x0.empty())
417 fem::apply_lifting(_b, a, constants, coeffs, bcs1, {}, alpha);
418 else
419 {
420 std::vector<std::span<const PetscScalar>> x0_ref;
421 std::vector<Vec> x0_local(a.size());
422 std::vector<const PetscScalar*> x0_array(a.size());
423 for (std::size_t i = 0; i < a.size(); ++i)
424 {
425 assert(x0[i]);
426 common::petsc::check(VecGhostGetLocalForm(x0[i], &x0_local[i]),
427 "VecGhostGetLocalForm");
428 PetscInt n0 = 0;
429 common::petsc::check(VecGetSize(x0_local[i], &n0), "VecGetSize");
430 common::petsc::check(VecGetArrayRead(x0_local[i], &x0_array[i]),
431 "VecGetArrayRead");
432 x0_ref.emplace_back(x0_array[i], n0);
433 }
434
435 fem::apply_lifting(_b, a, constants, coeffs, bcs1, x0_ref, alpha);
436
437 for (std::size_t i = 0; i < x0_local.size(); ++i)
438 {
439 common::petsc::check(VecRestoreArrayRead(x0_local[i], &x0_array[i]),
440 "VecRestoreArrayRead");
441 common::petsc::check(VecGhostRestoreLocalForm(x0[i], &x0_local[i]),
442 "VecGhostRestoreLocalForm");
443 }
444 }
445
446 common::petsc::check(VecRestoreArray(b_local, &array), "VecRestoreArray");
447 common::petsc::check(VecGhostRestoreLocalForm(b, &b_local),
448 "VecGhostRestoreLocalForm");
449}

551template <std::floating_point T>
552void set_bc(Vec b,
553 const std::vector<
554 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>& bcs,
555 std::optional<const Vec> x0, PetscScalar alpha = 1)
556{
557 PetscInt n = 0;
558 common::petsc::check(VecGetLocalSize(b, &n), "VecGetLocalSize");
559 PetscScalar* array = nullptr;
560 common::petsc::check(VecGetArray(b, &array), "VecGetArray");
561 std::span<PetscScalar> _b(array, n);
562 if (x0.has_value())
563 {
564 Vec x0_local;
565 common::petsc::check(VecGhostGetLocalForm(x0.value(), &x0_local),
566 "VecGhostGetLocalForm");
567 PetscInt n0 = 0;
568 common::petsc::check(VecGetSize(x0_local, &n0), "VecGetSize");
569 const PetscScalar* x0_array = nullptr;
570 common::petsc::check(VecGetArrayRead(x0_local, &x0_array),
571 "VecGetArrayRead");
572 std::span<const PetscScalar> _x0(x0_array, n0);
573 for (auto& bc : bcs)
574 bc.get().set(_b, _x0, alpha);
575 common::petsc::check(VecRestoreArrayRead(x0_local, &x0_array),
576 "VecRestoreArrayRead");
577 common::petsc::check(VecGhostRestoreLocalForm(x0.value(), &x0_local),
578 "VecGhostRestoreLocalForm");
579 }
580 else
581 {
582 for (auto& bc : bcs)
583 bc.get().set(_b, std::nullopt, alpha);
584 }
585 common::petsc::check(VecRestoreArray(b, &array), "VecRestoreArray");
586}

596template <std::floating_point T>
597void assign(const Vec x, Function<PetscScalar, T>& u)
598{
599 Vec x_local = nullptr;
600 common::petsc::check(VecGhostGetLocalForm(x, &x_local),
601 "VecGhostGetLocalForm");
602 PetscInt n = 0;
603 common::petsc::check(VecGetSize(x_local, &n), "VecGetSize");
604
605 std::span<PetscScalar> _u = u.x()->array();
606 if (static_cast<std::size_t>(n) != _u.size())
607 {
608 throw std::runtime_error(std::format(
609 "Vector has {} local entries, function has {}.", n, _u.size()));
610 }
611
612 const PetscScalar* array = nullptr;
613 common::petsc::check(VecGetArrayRead(x_local, &array), "VecGetArrayRead");
614 std::ranges::copy(std::span<const PetscScalar>(array, n), _u.begin());
615 common::petsc::check(VecRestoreArrayRead(x_local, &array),
616 "VecRestoreArrayRead");
617 common::petsc::check(VecGhostRestoreLocalForm(x, &x_local),
618 "VecGhostRestoreLocalForm");
619}

626template <std::floating_point T>
627void assemble_operator(
628 Mat A, const Form<PetscScalar, T>& a,
629 const std::vector<
630 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>& bcs)
631{
632 common::petsc::check(MatZeroEntries(A), "MatZeroEntries");
633 fem::assemble_matrix(la::petsc::Matrix::set_block_fn(A, ADD_VALUES), a, bcs);
634
635 // The unit diagonal is only meaningful when the rows and columns are
636 // indexed by the same space
637 if (a.function_spaces()[0] == a.function_spaces()[1])
638 {
639 // Flush to switch from adding to inserting
640 common::petsc::check(MatAssemblyBegin(A, MAT_FLUSH_ASSEMBLY),
641 "MatAssemblyBegin");
642 common::petsc::check(MatAssemblyEnd(A, MAT_FLUSH_ASSEMBLY),
643 "MatAssemblyEnd");
644 fem::set_diagonal(la::petsc::Matrix::set_fn(A, INSERT_VALUES),
645 *a.function_spaces()[0], bcs);
646 }
647
648 common::petsc::check(MatAssemblyBegin(A, MAT_FINAL_ASSEMBLY),
649 "MatAssemblyBegin");
650 common::petsc::check(MatAssemblyEnd(A, MAT_FINAL_ASSEMBLY), "MatAssemblyEnd");
651}

679template <std::floating_point T>
680void assemble_residual(
681 const Vec x, Vec b, const Form<PetscScalar, T>& F,
682 const Form<PetscScalar, T>& J,
683 const std::vector<
684 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>& bcs,
685 Function<PetscScalar, T>& u)
686{
687 common::petsc::check(VecGhostUpdateBegin(x, INSERT_VALUES, SCATTER_FORWARD),
688 "VecGhostUpdateBegin");
689 common::petsc::check(VecGhostUpdateEnd(x, INSERT_VALUES, SCATTER_FORWARD),
690 "VecGhostUpdateEnd");
691 impl::assign(x, u);
692
693 // Zero the local form, as assembly accumulates into ghost entries
694 Vec b_local = nullptr;
695 common::petsc::check(VecGhostGetLocalForm(b, &b_local),
696 "VecGhostGetLocalForm");
697 common::petsc::check(VecZeroEntries(b_local), "VecZeroEntries");
698 common::petsc::check(VecGhostRestoreLocalForm(b, &b_local),
699 "VecGhostRestoreLocalForm");
700
701 assemble_vector(b, F);
702
703 std::vector<std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>
704 a{J};
705 std::vector<
706 std::vector<std::reference_wrapper<const DirichletBC<PetscScalar, T>>>>
707 bcs1{bcs};
708 apply_lifting(b, a, bcs1, std::vector<Vec>{x}, -1);
709
710 common::petsc::check(VecGhostUpdateBegin(b, ADD_VALUES, SCATTER_REVERSE),
711 "VecGhostUpdateBegin");
712 common::petsc::check(VecGhostUpdateEnd(b, ADD_VALUES, SCATTER_REVERSE),
713 "VecGhostUpdateEnd");
714
715 set_bc(b, bcs, x, -1);
716
717 common::petsc::check(VecGhostUpdateBegin(b, INSERT_VALUES, SCATTER_FORWARD),
718 "VecGhostUpdateBegin");
719 common::petsc::check(VecGhostUpdateEnd(b, INSERT_VALUES, SCATTER_FORWARD),
720 "VecGhostUpdateEnd");
721}
