72template <std::floating_point T>
73Mat create_matrix_block(
74 const std::vector<std::vector<const Form<PetscScalar, T>*>>& a,
75 std::optional<std::string> type = std::nullopt)
76{
77 // Extract and check row/column ranges
78 std::array<std::vector<std::shared_ptr<const FunctionSpace<T>>>, 2> V
79 = fem::common_function_spaces(extract_function_spaces(a));
80 std::array<std::vector<int>, 2> bs_dofs;
81 for (std::size_t i = 0; i < 2; ++i)
82 {
83 for (auto& _V : V[i])
84 bs_dofs[i].push_back(_V->dofmap()->bs());
85 }
86
87 // Build sparsity pattern for each block
88 std::shared_ptr<const mesh::Mesh<T>> mesh;
89 std::vector<std::vector<std::unique_ptr<la::SparsityPattern>>> patterns(
90 V[0].size());
91 for (std::size_t row = 0; row < V[0].size(); ++row)
92 {
93 for (std::size_t col = 0; col < V[1].size(); ++col)
94 {
95 if (const Form<PetscScalar, T>* form = a[row][col]; form)
96 {
97 patterns[row].push_back(std::make_unique<la::SparsityPattern>(
98 create_sparsity_pattern(*form)));
99 if (!mesh)
100 mesh = form->mesh();
101 }
102 else
103 patterns[row].push_back(nullptr);
104 }
105 }
106
107 if (!mesh)
108 throw std::invalid_argument("Could not find a Mesh.");
109
110 // Compute offsets for the fields
111 std::array<std::vector<std::pair<
112 std::reference_wrapper<const common::IndexMap>, int>>,
113 2>
114 maps;
115 for (std::size_t d = 0; d < 2; ++d)
116 {
117 for (auto& space : V[d])
118 {
119 maps[d].emplace_back(*space->dofmap()->index_map,
120 space->dofmap()->index_map_bs());
121 }
122 }
123
124 // Create merged sparsity pattern
125 std::vector<std::vector<const la::SparsityPattern*>> p(V[0].size());
126 for (std::size_t row = 0; row < V[0].size(); ++row)
127 for (std::size_t col = 0; col < V[1].size(); ++col)
128 p[row].push_back(patterns[row][col].get());
129
130 la::SparsityPattern pattern(mesh->comm(), p, maps, bs_dofs);
131 pattern.finalize();
132
133 // TODO: Index map concatenation has already been computed inside
134 // the SparsityPattern constructor, but we also need it here to
135 // build the PETSc local-to-global map. Compute outside and pass
136 // into SparsityPattern constructor.
137
138 // Create row and column local-to-global maps (field0, field1, field2,
139 // etc), i.e. ghosts of field0 appear before owned indices of field1
140 std::array<std::vector<PetscInt>, 2> _maps;
141 for (int d = 0; d < 2; ++d)
142 {
143 if (d == 1 and V[0] == V[1])
144 {
145 // Row and column spaces are identical, so the concatenated
146 // index map for d=1 is identical to the one already computed
147 // for d=0 -- reuse it rather than paying for a second,
148 // communication-heavy call to stack_index_maps.
149 _maps[1] = _maps[0];
150 continue;
151 }
152
153 const std::vector<
154 std::pair<std::reference_wrapper<const common::IndexMap>, int>>& map
155 = maps[d];
156 std::vector<PetscInt>& _map = _maps[d];
157
158 // Concatenate the block index map in the row and column directions
159 const auto [rank_offset, local_offset, ghosts, _]
160 = common::stack_index_maps(map);
161 const std::size_t num_ghosts
162 = std::accumulate(ghosts.begin(), ghosts.end(), std::size_t(0),
163 [](std::size_t n, auto& g) { return n + g.size(); });
164 _map.reserve(local_offset.back() + num_ghosts);
165 for (std::size_t f = 0; f < map.size(); ++f)
166 {
167 auto offset = local_offset[f];
168 const common::IndexMap& imap = map[f].first.get();
169 int bs = map[f].second;
170 auto owned
171 = std::views::iota(std::int32_t(0), bs * imap.size_local())
172 | std::views::transform([offset, rank_offset](std::int32_t i)
173 { return i + rank_offset + offset; });
174 _map.insert(_map.end(), owned.begin(), owned.end());
175 _map.insert(_map.end(), ghosts[f].begin(), ghosts[f].end());
176 }
177 }
178
179 // Create the local-to-global maps on the mesh communicator. MATIS
180 // requires them to share the matrix communicator
181 ISLocalToGlobalMapping l2g0 = nullptr, l2g1 = nullptr;
182 common::petsc::check(
183 ISLocalToGlobalMappingCreate(mesh->comm(), 1, _maps[0].size(),
184 _maps[0].data(), PETSC_COPY_VALUES, &l2g0),
185 "ISLocalToGlobalMappingCreate");
186 if (V[0] != V[1])
187 {
188 common::petsc::check(
189 ISLocalToGlobalMappingCreate(mesh->comm(), 1, _maps[1].size(),
190 _maps[1].data(), PETSC_COPY_VALUES, &l2g1),
191 "ISLocalToGlobalMappingCreate");
192 }
193
194 // Initialise the matrix. MATIS builds its preallocation from the
195 // maps, so they are passed to the constructor
196 Mat A = la::petsc::create_matrix(mesh->comm(), pattern, type, l2g0,
197 l2g1 ? l2g1 : l2g0);
198 common::petsc::check(ISLocalToGlobalMappingDestroy(&l2g0),
199 "ISLocalToGlobalMappingDestroy");
200 if (l2g1)
201 {
202 common::petsc::check(ISLocalToGlobalMappingDestroy(&l2g1),
203 "ISLocalToGlobalMappingDestroy");
204 }
205
206 return A;
207}

212template <std::floating_point T>
213Mat create_matrix_nest(
214 const std::vector<std::vector<const Form<PetscScalar, T>*>>& a,
215 std::optional<std::vector<std::vector<std::optional<std::string>>>> types)
216{
217 if (a.empty())
218 throw std::invalid_argument(
219 "Rectangular array of forms must be non-empty.");
220
221 // Extract and check row/column ranges
222 auto V = fem::common_function_spaces(extract_function_spaces(a));
223
224 // Loop over each form and create matrix
225 int rows = a.size();
226 int cols = a.front().size();
227 std::vector<Mat> mats(rows * cols, nullptr);
228 std::shared_ptr<const mesh::Mesh<T>> mesh;
229 for (int i = 0; i < rows; ++i)
230 {
231 for (int j = 0; j < cols; ++j)
232 {
233 if (const Form<PetscScalar, T>* form = a[i][j]; form)
234 {
235 if (types)
236 mats[i * cols + j] = create_matrix(*form, types->at(i).at(j));
237 else
238 mats[i * cols + j] = create_matrix(*form, std::nullopt);
239 mesh = form->mesh();
240 }
241 }
242 }
243
244 if (!mesh)
245 throw std::invalid_argument("Could not find a Mesh.");
246
247 // Initialise block (MatNest) matrix. On error, destroy the
248 // already-created sub-matrices in `mats` before propagating, since
249 // the nest (which would otherwise take joint ownership of them) was
250 // never successfully assembled.
251 Mat A;
252 try
253 {
254 common::petsc::check(MatCreate(mesh->comm(), &A), "MatCreate");
255 common::petsc::check(MatSetType(A, MATNEST), "MatSetType");
256 common::petsc::check(
257 MatNestSetSubMats(A, rows, nullptr, cols, nullptr, mats.data()),
258 "MatNestSetSubMats");
259 common::petsc::check(MatSetUp(A), "MatSetUp");
260 }
261 catch (...)
262 {
263 for (Mat& m : mats)
264 if (m)
265 common::petsc::check(MatDestroy(&m), "MatDestroy");
266 throw;
267 }
268
269 // De-reference Mat objects
270 for (Mat& m : mats)
271 if (m)
272 common::petsc::check(MatDestroy(&m), "MatDestroy");
273
274 return A;
275}

386template <std::floating_point T>
387void apply_lifting(
388 Vec b,
389 const std::vector<
390 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>& a,
391 const std::vector<std::span<const PetscScalar>>& constants,
392 const std::vector<std::map<std::pair<IntegralType, int>,
393 std::pair<std::span<const PetscScalar>, int>>>&
394 coeffs,
395 const std::vector<std::span<const std::int8_t>>& bc_markers1,
396 const std::vector<std::span<const PetscScalar>>& bc_values1,
397 const std::vector<Vec>& x0, PetscScalar alpha)
398{
399 if (!x0.empty() and x0.size() != a.size())
400 throw std::invalid_argument("Mismatch between x0 and a in apply_lifting.");
401
402 Vec b_local;
403 common::petsc::check(VecGhostGetLocalForm(b, &b_local),
404 "VecGhostGetLocalForm");
405 PetscInt n = 0;
406 common::petsc::check(VecGetSize(b_local, &n), "VecGetSize");
407 PetscScalar* array = nullptr;
408 common::petsc::check(VecGetArray(b_local, &array), "VecGetArray");
409 std::span<PetscScalar> _b(array, n);
410
411 if (x0.empty())
412 {
413 fem::apply_lifting(_b, a, constants, coeffs, bc_markers1, bc_values1, {},
414 alpha);
415 }
416 else
417 {
418 std::vector<std::span<const PetscScalar>> x0_ref;
419 std::vector<Vec> x0_local(a.size());
420 std::vector<const PetscScalar*> x0_array(a.size());
421 for (std::size_t i = 0; i < a.size(); ++i)
422 {
423 assert(x0[i]);
424 common::petsc::check(VecGhostGetLocalForm(x0[i], &x0_local[i]),
425 "VecGhostGetLocalForm");
426 PetscInt n0 = 0;
427 common::petsc::check(VecGetSize(x0_local[i], &n0), "VecGetSize");
428 common::petsc::check(VecGetArrayRead(x0_local[i], &x0_array[i]),
429 "VecGetArrayRead");
430 x0_ref.emplace_back(x0_array[i], n0);
431 }
432
433 fem::apply_lifting(_b, a, constants, coeffs, bc_markers1, bc_values1,
434 x0_ref, alpha);
435
436 for (std::size_t i = 0; i < x0_local.size(); ++i)
437 {
438 common::petsc::check(VecRestoreArrayRead(x0_local[i], &x0_array[i]),
439 "VecRestoreArrayRead");
440 common::petsc::check(VecGhostRestoreLocalForm(x0[i], &x0_local[i]),
441 "VecGhostRestoreLocalForm");
442 }
443 }
444
445 common::petsc::check(VecRestoreArray(b_local, &array), "VecRestoreArray");
446 common::petsc::check(VecGhostRestoreLocalForm(b, &b_local),
447 "VecGhostRestoreLocalForm");
448}

480template <std::floating_point T>
481void apply_lifting(
482 Vec b,
483 const std::vector<
484 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>& a,
485 const std::vector<
486 std::vector<std::reference_wrapper<const DirichletBC<PetscScalar, T>>>>&
487 bcs1,
488 const std::vector<Vec>& x0, PetscScalar alpha)
489{
490 std::vector<std::map<std::pair<IntegralType, int>,
491 std::pair<std::vector<PetscScalar>, int>>>
492 coeffs;
493 std::vector<std::vector<PetscScalar>> constants;
494 for (const auto& _a : a)
495 {
496 if (_a)
497 {
498 auto coefficients = allocate_coefficient_storage(_a->get());
499 pack_coefficients(_a->get(), coefficients);
500 coeffs.push_back(coefficients);
501 constants.push_back(pack_constants(_a->get()));
502 }
503 else
504 {
505 coeffs.emplace_back();
506 constants.emplace_back();
507 }
508 }
509
510 std::vector<std::span<const PetscScalar>> _constants(constants.begin(),
511 constants.end());
512 std::vector<std::map<std::pair<IntegralType, int>,
513 std::pair<std::span<const PetscScalar>, int>>>
514 _coeffs;
515 std::ranges::transform(coeffs, std::back_inserter(_coeffs),
516 [](auto& c) { return make_coefficients_span(c); });
517
518 auto [bc_markers1, bc_values1] = fem::impl::bc_lifting_data(a, bcs1);
519 apply_lifting(b, a, _constants, _coeffs,
520 std::vector<std::span<const std::int8_t>>(bc_markers1.begin(),
521 bc_markers1.end()),
522 std::vector<std::span<const PetscScalar>>(bc_values1.begin(),
523 bc_values1.end()),
524 x0, alpha);
525}

573 {
574 for (auto& bc : bcs)
575 bc.get().set(_b, std::nullopt, alpha);
576 }
577 common::petsc::check(VecRestoreArray(b, &array), "VecRestoreArray");
578}

588template <std::floating_point T>
589void assign(const Vec x, Function<PetscScalar, T>& u)
590{
591 Vec x_local = nullptr;
592 common::petsc::check(VecGhostGetLocalForm(x, &x_local),
593 "VecGhostGetLocalForm");
594 PetscInt n = 0;
595 common::petsc::check(VecGetSize(x_local, &n), "VecGetSize");
596
597 std::span<PetscScalar> _u = u.x()->array();
598 if (static_cast<std::size_t>(n) != _u.size())
599 {
600 throw std::runtime_error(std::format(
601 "Vector has {} local entries, function has {}.", n, _u.size()));
602 }
603
604 const PetscScalar* array = nullptr;
605 common::petsc::check(VecGetArrayRead(x_local, &array), "VecGetArrayRead");
606 std::ranges::copy(std::span<const PetscScalar>(array, n), _u.begin());
607 common::petsc::check(VecRestoreArrayRead(x_local, &array),
608 "VecRestoreArrayRead");
609 common::petsc::check(VecGhostRestoreLocalForm(x, &x_local),
610 "VecGhostRestoreLocalForm");
611}

639template <std::floating_point T>
640void assemble_operator(Mat A, const Form<PetscScalar, T>& a,
641 std::span<const std::int8_t> dof_marker0,
642 std::span<const std::int8_t> dof_marker1)
643{
644 common::petsc::check(MatZeroEntries(A), "MatZeroEntries");
645
646 // Block and dof indices coincide at block size 1. Avoid the generic
647 // blocked PETSc path, which expands these indices before insertion.
648 if (a.function_spaces()[0]->dofmap()->index_map_bs() == 1
649 and a.function_spaces()[1]->dofmap()->index_map_bs() == 1)
650 {
651 fem::assemble_matrix(la::petsc::Matrix::set_fn(A, ADD_VALUES), a,
652 dof_marker0, dof_marker1);
653 }
654 else
655 {
656 fem::assemble_matrix(la::petsc::Matrix::set_block_fn(A, ADD_VALUES), a,
657 dof_marker0, dof_marker1);
658 }
659
660 // The unit diagonal is only meaningful when the rows and columns are
661 // indexed by the same space
662 if (a.function_spaces()[0] == a.function_spaces()[1])
663 {
664 // Locally owned constrained rows
665 std::shared_ptr<const DofMap> dofmap0
666 = a.function_spaces()[0]->dofmaps().front();
667 // dof_marker0 is empty when no rows are constrained
668 const std::int32_t num_owned
669 = dof_marker0.empty()
670 ? 0
671 : dofmap0->index_map_bs() * dofmap0->index_map->size_local();
672 std::vector<std::int32_t> rows;
673 for (std::int32_t i = 0; i < num_owned; ++i)
674 {
675 if (dof_marker0[i])
676 rows.push_back(i);
677 }
678 // Assembly zeroed these rows, so adding sets the diagonal. Adding
679 // avoids a flush to switch from ADD_VALUES to INSERT_VALUES.
680 fem::set_diagonal<PetscScalar>(la::petsc::Matrix::set_fn(A, ADD_VALUES),
681 rows);
682 }
683
684 common::petsc::check(MatAssemblyBegin(A, MAT_FINAL_ASSEMBLY),
685 "MatAssemblyBegin");
686 common::petsc::check(MatAssemblyEnd(A, MAT_FINAL_ASSEMBLY), "MatAssemblyEnd");
687}

715template <std::floating_point T>
716void assemble_residual(
717 const Vec x, Vec b, const Form<PetscScalar, T>& F,
718 const Form<PetscScalar, T>& J,
719 const std::vector<
720 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>& bcs,
721 Function<PetscScalar, T>& u)
722{
723 common::petsc::check(VecGhostUpdateBegin(x, INSERT_VALUES, SCATTER_FORWARD),
724 "VecGhostUpdateBegin");
725 common::petsc::check(VecGhostUpdateEnd(x, INSERT_VALUES, SCATTER_FORWARD),
726 "VecGhostUpdateEnd");
727 impl::assign(x, u);
728
729 // Zero the local form, as assembly accumulates into ghost entries
730 Vec b_local = nullptr;
731 common::petsc::check(VecGhostGetLocalForm(b, &b_local),
732 "VecGhostGetLocalForm");
733 common::petsc::check(VecZeroEntries(b_local), "VecZeroEntries");
734 common::petsc::check(VecGhostRestoreLocalForm(b, &b_local),
735 "VecGhostRestoreLocalForm");
736
737 assemble_vector(b, F);
738
739 // Constrained dof markers and values g on the trial space of J (owned
740 // and ghost), both empty if there are no bcs
741 const std::vector<
742 std::optional<std::reference_wrapper<const Form<PetscScalar, T>>>>
743 a{J};
744 auto [bc_markers1, bc_values1] = fem::impl::bc_lifting_data(
745 a, std::vector<std::vector<
746 std::reference_wrapper<const DirichletBC<PetscScalar, T>>>>{bcs});
747
748 // Lifting: b <- b + J (g - x)
749 const std::vector<PetscScalar> constants = pack_constants(J);
750 auto coeffs = allocate_coefficient_storage(J);
751 pack_coefficients(J, coeffs);
752 apply_lifting(b, a, {std::span<const PetscScalar>(constants)},
753 {make_coefficients_span(coeffs)},
754 {std::span<const std::int8_t>(bc_markers1.front())},
755 {std::span<const PetscScalar>(bc_values1.front())}, {x},
756 PetscScalar(-1));
757
758 common::petsc::check(VecGhostUpdateBegin(b, ADD_VALUES, SCATTER_REVERSE),
759 "VecGhostUpdateBegin");
760 common::petsc::check(VecGhostUpdateEnd(b, ADD_VALUES, SCATTER_REVERSE),
761 "VecGhostUpdateEnd");
762
763 // Set b = x - g on owned constrained dofs
764 if (!bcs.empty())
765 {
766 PetscScalar* b_array = nullptr;
767 common::petsc::check(VecGetArray(b, &b_array), "VecGetArray");
768 const PetscScalar* x_array = nullptr;
769 common::petsc::check(VecGetArrayRead(x, &x_array), "VecGetArrayRead");
770 for (auto& bc : bcs)
771 {
772 auto [dofs, owned] = bc.get().dof_indices();
773 for (std::int32_t dof : dofs.first(owned))
774 b_array[dof] = x_array[dof] - bc_values1.front()[dof];
775 }
776 common::petsc::check(VecRestoreArrayRead(x, &x_array),
777 "VecRestoreArrayRead");
778 common::petsc::check(VecRestoreArray(b, &b_array), "VecRestoreArray");
779 }
780
781 common::petsc::check(VecGhostUpdateBegin(b, INSERT_VALUES, SCATTER_FORWARD),
782 "VecGhostUpdateBegin");
783 common::petsc::check(VecGhostUpdateEnd(b, INSERT_VALUES, SCATTER_FORWARD),
784 "VecGhostUpdateEnd");
785}
