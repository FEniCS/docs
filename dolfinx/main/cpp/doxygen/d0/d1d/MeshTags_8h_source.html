157template <typename T>
158MeshTags<T> create_meshtags(std::shared_ptr<const Topology> topology, int dim,
159 const graph::AdjacencyList<std::int32_t>& entities,
160 std::span<const T> values,
161 std::string name = "mesh_tags")
162{
163 spdlog::info(
164 "Building MeshTags object from tagged entities (defined by vertices).");
165
166 // Compute the indices of the topology entities (index is set to -1 if
167 // it can't be found)
168 if (!topology)
169 throw std::invalid_argument("topology must not be null.");
170 const std::vector<std::int32_t> indices
171 = entities_to_index(*topology, dim, entities.array());
172 if (indices.size() != values.size())
173 {
174 throw std::invalid_argument(
175 "Duplicate mesh entities when building MeshTags object.");
176 }
177
178 // Sort the indices and values by indices
179 auto [indices_sorted, values_sorted] = common::sort_unique(indices, values);
180
181 // Remove any entities that were not found (these have an index of -1)
182 auto it0 = std::ranges::lower_bound(indices_sorted, 0);
183 std::size_t pos0 = std::ranges::distance(indices_sorted.begin(), it0);
184 indices_sorted.erase(indices_sorted.begin(), it0);
185 values_sorted.erase(values_sorted.begin(),
186 std::next(values_sorted.begin(), pos0));
187
188 return MeshTags<T>(topology, dim, std::move(indices_sorted),
189 std::move(values_sorted), std::move(name));
190}
