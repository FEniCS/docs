634 template <typename U>
635 std::function<void(std::span<U>, std::span<const std::uint32_t>, std::int32_t,
636 int)>
637 dof_transformation_fn(doftransform ttype, bool scalar_element = false) const
638 {
639 if (!needs_dof_transformations())
640 return nullptr;
641
642 if (!_sub_elements.empty())
643 {
644 if (!_reference_value_shape) // Mixed element
645 {
646 std::vector<std::function<void(
647 std::span<U>, std::span<const std::uint32_t>, std::int32_t, int)>>
648 sub_element_fns;
649 std::vector<int> dims;
650 sub_element_fns.reserve(_sub_elements.size());
651 dims.reserve(_sub_elements.size());
652 for (std::size_t i = 0; i < _sub_elements.size(); ++i)
653 {
654 sub_element_fns.push_back(
655 _sub_elements[i]->template dof_transformation_fn<U>(ttype));
656 dims.push_back(_sub_elements[i]->space_dimension());
657 }
658
659 return [dims = std::move(dims),
660 sub_element_fns = std::move(sub_element_fns)](
661 std::span<U> data, std::span<const std::uint32_t> cell_info,
662 std::int32_t cell, int block_size)
663 {
664 std::size_t offset = 0;
665 for (std::size_t e = 0; e < sub_element_fns.size(); ++e)
666 {
667 const std::size_t width = dims[e] * block_size;
668 if (sub_element_fns[e])
669 {
670 sub_element_fns[e](data.subspan(offset, width), cell_info, cell,
671 block_size);
672 }
673 offset += width;
674 }
675 };
676 }
677 else if (!scalar_element)
678 {
679 // Blocked element
680 std::function<void(std::span<U>, std::span<const std::uint32_t>,
681 std::int32_t, int)>
682 sub_fn
683 = _sub_elements.front()->template dof_transformation_fn<U>(ttype);
684 assert(sub_fn); // Consistent with needs_dof_transformations() above
685 const int ebs = _bs;
686 return [ebs, sub_fn = std::move(sub_fn)](
687 std::span<U> data, std::span<const std::uint32_t> cell_info,
688 std::int32_t cell, int data_block_size)
689 { sub_fn(data, cell_info, cell, ebs * data_block_size); };
690 }
691 }
692
693 switch (ttype)
694 {
695 case doftransform::inverse_transpose:
696 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
697 std::int32_t cell, int block_size)
698 { Tt_inv_apply(data, cell_info[cell], block_size); };
699 case doftransform::transpose:
700 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
701 std::int32_t cell, int block_size)
702 { Tt_apply(data, cell_info[cell], block_size); };
703 case doftransform::inverse:
704 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
705 std::int32_t cell, int block_size)
706 { Tinv_apply(data, cell_info[cell], block_size); };
707 case doftransform::standard:
708 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
709 std::int32_t cell, int block_size)
710 { T_apply(data, cell_info[cell], block_size); };
711 default:
712 throw std::invalid_argument("Unknown transformation type");
713 }
714 }

739 template <typename U>
740 std::function<void(std::span<U>, std::span<const std::uint32_t>, std::int32_t,
741 int)>
742 dof_transformation_right_fn(doftransform ttype,
743 bool scalar_element = false) const
744 {
745 if (!needs_dof_transformations())
746 return nullptr;
747 else if (!_sub_elements.empty())
748 {
749 if (!_reference_value_shape) // Mixed element
750 {
751 std::vector<std::function<void(
752 std::span<U>, std::span<const std::uint32_t>, std::int32_t, int)>>
753 sub_element_fns;
754 std::vector<int> dims;
755 sub_element_fns.reserve(_sub_elements.size());
756 dims.reserve(_sub_elements.size());
757 for (std::size_t i = 0; i < _sub_elements.size(); ++i)
758 {
759 sub_element_fns.push_back(
760 _sub_elements[i]->template dof_transformation_right_fn<U>(ttype));
761 dims.push_back(_sub_elements[i]->space_dimension());
762 }
763
764 return [dims = std::move(dims),
765 sub_element_fns = std::move(sub_element_fns)](
766 std::span<U> data, std::span<const std::uint32_t> cell_info,
767 std::int32_t cell, int block_size)
768 {
769 // `data` is (block_size, ndofs), row-major. Sub-element `e`
770 // owns columns [offset, offset + dims[e]) of every row, so the
771 // rows are transformed one at a time.
772 const std::size_t ndofs = data.size() / block_size;
773 std::size_t offset = 0;
774 for (std::size_t e = 0; e < sub_element_fns.size(); ++e)
775 {
776 if (sub_element_fns[e])
777 {
778 for (int i = 0; i < block_size; ++i)
779 {
780 sub_element_fns[e](data.subspan(i * ndofs + offset, dims[e]),
781 cell_info, cell, 1);
782 }
783 }
784 offset += dims[e];
785 }
786 };
787 }
788 else if (!scalar_element)
789 {
790 // Blocked element
791 // The transformation from the left can be used here as blocked
792 // elements use xyzxyzxyz ordering, and so applying the DOF
793 // transformation from the right is equivalent to applying the DOF
794 // transformation from the left to data using xxxyyyzzz ordering
795 std::function<void(std::span<U>, std::span<const std::uint32_t>,
796 std::int32_t, int)>
797 sub_fn
798 = _sub_elements.front()->template dof_transformation_fn<U>(ttype);
799 assert(sub_fn); // Consistent with needs_dof_transformations() above
800 return [this, sub_fn = std::move(sub_fn)](
801 std::span<U> data, std::span<const std::uint32_t> cell_info,
802 std::int32_t cell, int data_block_size)
803 {
804 const int ebs = block_size();
805 const std::size_t dof_count = data.size() / data_block_size;
806 for (int block = 0; block < data_block_size; ++block)
807 {
808 sub_fn(data.subspan(block * dof_count, dof_count), cell_info, cell,
809 ebs);
810 }
811 };
812 }
813 }
814
815 switch (ttype)
816 {
817 case doftransform::inverse_transpose:
818 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
819 std::int32_t cell, int n)
820 { Tt_inv_apply_right(data, cell_info[cell], n); };
821 case doftransform::transpose:
822 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
823 std::int32_t cell, int n)
824 { Tt_apply_right(data, cell_info[cell], n); };
825 case doftransform::inverse:
826 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
827 std::int32_t cell, int n)
828 { Tinv_apply_right(data, cell_info[cell], n); };
829 case doftransform::standard:
830 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
831 std::int32_t cell, int n)
832 { T_apply_right(data, cell_info[cell], n); };
833 default:
834 throw std::invalid_argument("Unknown transformation type");
835 }
836 }

1066 // Negate data on a cell marked with mesh::reversed_cell_bit if the
1067 // basis depends on the cell orientation. A sign is its own inverse and
1068 // transpose, so every DOF transformation applies it the same way.
1069 template <typename U>
1070 void flip_reversed_cell(std::span<U> data,
1071 std::uint32_t cell_permutation) const
1072 {
1073 if (_depends_on_cell_orientation
1074 and (cell_permutation & mesh::reversed_cell_bit))
1075 {
1076 std::ranges::transform(data, data.begin(), std::negate{});
