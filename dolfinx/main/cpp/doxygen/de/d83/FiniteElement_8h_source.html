436 template <typename U>
437 std::function<void(std::span<U>, std::span<const std::uint32_t>, std::int32_t,
438 int)>
439 dof_transformation_fn(doftransform ttype, bool scalar_element = false) const
440 {
441 if (!needs_dof_transformations())
442 return nullptr;
443
444 if (!_sub_elements.empty())
445 {
446 if (!_reference_value_shape) // Mixed element
447 {
448 std::vector<std::function<void(
449 std::span<U>, std::span<const std::uint32_t>, std::int32_t, int)>>
450 sub_element_fns;
451 std::vector<int> dims;
452 sub_element_fns.reserve(_sub_elements.size());
453 dims.reserve(_sub_elements.size());
454 for (std::size_t i = 0; i < _sub_elements.size(); ++i)
455 {
456 sub_element_fns.push_back(
457 _sub_elements[i]->template dof_transformation_fn<U>(ttype));
458 dims.push_back(_sub_elements[i]->space_dimension());
459 }
460
461 return [dims = std::move(dims),
462 sub_element_fns = std::move(sub_element_fns)](
463 std::span<U> data, std::span<const std::uint32_t> cell_info,
464 std::int32_t cell, int block_size)
465 {
466 std::size_t offset = 0;
467 for (std::size_t e = 0; e < sub_element_fns.size(); ++e)
468 {
469 const std::size_t width = dims[e] * block_size;
470 if (sub_element_fns[e])
471 {
472 sub_element_fns[e](data.subspan(offset, width), cell_info, cell,
473 block_size);
474 }
475 offset += width;
476 }
477 };
478 }
479 else if (!scalar_element)
480 {
481 // Blocked element
482 std::function<void(std::span<U>, std::span<const std::uint32_t>,
483 std::int32_t, int)>
484 sub_fn
485 = _sub_elements.front()->template dof_transformation_fn<U>(ttype);
486 assert(sub_fn); // Consistent with needs_dof_transformations() above
487 const int ebs = _bs;
488 return [ebs, sub_fn = std::move(sub_fn)](
489 std::span<U> data, std::span<const std::uint32_t> cell_info,
490 std::int32_t cell, int data_block_size)
491 { sub_fn(data, cell_info, cell, ebs * data_block_size); };
492 }
493 }
494
495 switch (ttype)
496 {
497 case doftransform::inverse_transpose:
498 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
499 std::int32_t cell, int block_size)
500 { Tt_inv_apply(data, cell_info[cell], block_size); };
501 case doftransform::transpose:
502 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
503 std::int32_t cell, int block_size)
504 { Tt_apply(data, cell_info[cell], block_size); };
505 case doftransform::inverse:
506 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
507 std::int32_t cell, int block_size)
508 { Tinv_apply(data, cell_info[cell], block_size); };
509 case doftransform::standard:
510 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
511 std::int32_t cell, int block_size)
512 { T_apply(data, cell_info[cell], block_size); };
513 default:
514 throw std::runtime_error("Unknown transformation type");
515 }
516 }

541 template <typename U>
542 std::function<void(std::span<U>, std::span<const std::uint32_t>, std::int32_t,
543 int)>
544 dof_transformation_right_fn(doftransform ttype,
545 bool scalar_element = false) const
546 {
547 if (!needs_dof_transformations())
548 return nullptr;
549 else if (!_sub_elements.empty())
550 {
551 if (!_reference_value_shape) // Mixed element
552 {
553 std::vector<std::function<void(
554 std::span<U>, std::span<const std::uint32_t>, std::int32_t, int)>>
555 sub_element_fns;
556 std::vector<int> dims;
557 sub_element_fns.reserve(_sub_elements.size());
558 dims.reserve(_sub_elements.size());
559 for (std::size_t i = 0; i < _sub_elements.size(); ++i)
560 {
561 sub_element_fns.push_back(
562 _sub_elements[i]->template dof_transformation_right_fn<U>(ttype));
563 dims.push_back(_sub_elements[i]->space_dimension());
564 }
565
566 return [dims = std::move(dims),
567 sub_element_fns = std::move(sub_element_fns)](
568 std::span<U> data, std::span<const std::uint32_t> cell_info,
569 std::int32_t cell, int block_size)
570 {
571 std::size_t offset = 0;
572 for (std::size_t e = 0; e < sub_element_fns.size(); ++e)
573 {
574 if (sub_element_fns[e])
575 {
576 sub_element_fns[e](data.subspan(offset, data.size() - offset),
577 cell_info, cell, block_size);
578 }
579 offset += dims[e];
580 }
581 };
582 }
583 else if (!scalar_element)
584 {
585 // Blocked element
586 // The transformation from the left can be used here as blocked
587 // elements use xyzxyzxyz ordering, and so applying the DOF
588 // transformation from the right is equivalent to applying the DOF
589 // transformation from the left to data using xxxyyyzzz ordering
590 std::function<void(std::span<U>, std::span<const std::uint32_t>,
591 std::int32_t, int)>
592 sub_fn
593 = _sub_elements.front()->template dof_transformation_fn<U>(ttype);
594 assert(sub_fn); // Consistent with needs_dof_transformations() above
595 return [this, sub_fn = std::move(sub_fn)](
596 std::span<U> data, std::span<const std::uint32_t> cell_info,
597 std::int32_t cell, int data_block_size)
598 {
599 const int ebs = block_size();
600 const std::size_t dof_count = data.size() / data_block_size;
601 for (int block = 0; block < data_block_size; ++block)
602 {
603 sub_fn(data.subspan(block * dof_count, dof_count), cell_info, cell,
604 ebs);
605 }
606 };
607 }
608 }
609
610 switch (ttype)
611 {
612 case doftransform::inverse_transpose:
613 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
614 std::int32_t cell, int n)
615 { Tt_inv_apply_right(data, cell_info[cell], n); };
616 case doftransform::transpose:
617 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
618 std::int32_t cell, int n)
619 { Tt_apply_right(data, cell_info[cell], n); };
620 case doftransform::inverse:
621 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
622 std::int32_t cell, int n)
623 { Tinv_apply_right(data, cell_info[cell], n); };
624 case doftransform::standard:
625 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
626 std::int32_t cell, int n)
627 { T_apply_right(data, cell_info[cell], n); };
628 default:
629 throw std::runtime_error("Unknown transformation type");
630 }
631 }
