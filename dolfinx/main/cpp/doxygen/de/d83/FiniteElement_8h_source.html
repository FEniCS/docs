639 template <typename U>
640 std::function<void(std::span<U>, std::span<const std::uint32_t>, std::int32_t,
641 int)>
642 dof_transformation_fn(doftransform ttype, bool scalar_element = false) const
643 {
644 if (!needs_dof_transformations())
645 return nullptr;
646
647 if (!_sub_elements.empty())
648 {
649 if (!_reference_value_shape) // Mixed element
650 {
651 std::vector<std::function<void(
652 std::span<U>, std::span<const std::uint32_t>, std::int32_t, int)>>
653 sub_element_fns;
654 std::vector<int> dims;
655 sub_element_fns.reserve(_sub_elements.size());
656 dims.reserve(_sub_elements.size());
657 for (std::size_t i = 0; i < _sub_elements.size(); ++i)
658 {
659 sub_element_fns.push_back(
660 _sub_elements[i]->template dof_transformation_fn<U>(ttype));
661 dims.push_back(_sub_elements[i]->space_dimension());
662 }
663
664 return [dims = std::move(dims),
665 sub_element_fns = std::move(sub_element_fns)](
666 std::span<U> data, std::span<const std::uint32_t> cell_info,
667 std::int32_t cell, int block_size)
668 {
669 std::size_t offset = 0;
670 for (std::size_t e = 0; e < sub_element_fns.size(); ++e)
671 {
672 const std::size_t width = dims[e] * block_size;
673 if (sub_element_fns[e])
674 {
675 sub_element_fns[e](data.subspan(offset, width), cell_info, cell,
676 block_size);
677 }
678 offset += width;
679 }
680 };
681 }
682 else if (!scalar_element)
683 {
684 // Blocked element
685 std::function<void(std::span<U>, std::span<const std::uint32_t>,
686 std::int32_t, int)>
687 sub_fn
688 = _sub_elements.front()->template dof_transformation_fn<U>(ttype);
689 assert(sub_fn); // Consistent with needs_dof_transformations() above
690 const int ebs = _bs;
691 return [ebs, sub_fn = std::move(sub_fn)](
692 std::span<U> data, std::span<const std::uint32_t> cell_info,
693 std::int32_t cell, int data_block_size)
694 { sub_fn(data, cell_info, cell, ebs * data_block_size); };
695 }
696 }
697
698 switch (ttype)
699 {
700 case doftransform::inverse_transpose:
701 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
702 std::int32_t cell, int block_size)
703 { Tt_inv_apply(data, cell_info[cell], block_size); };
704 case doftransform::transpose:
705 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
706 std::int32_t cell, int block_size)
707 { Tt_apply(data, cell_info[cell], block_size); };
708 case doftransform::inverse:
709 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
710 std::int32_t cell, int block_size)
711 { Tinv_apply(data, cell_info[cell], block_size); };
712 case doftransform::standard:
713 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
714 std::int32_t cell, int block_size)
715 { T_apply(data, cell_info[cell], block_size); };
716 default:
717 throw std::invalid_argument("Unknown transformation type");
718 }
719 }

744 template <typename U>
745 std::function<void(std::span<U>, std::span<const std::uint32_t>, std::int32_t,
746 int)>
747 dof_transformation_right_fn(doftransform ttype,
748 bool scalar_element = false) const
749 {
750 if (!needs_dof_transformations())
751 return nullptr;
752 else if (!_sub_elements.empty())
753 {
754 if (!_reference_value_shape) // Mixed element
755 {
756 std::vector<std::function<void(
757 std::span<U>, std::span<const std::uint32_t>, std::int32_t, int)>>
758 sub_element_fns;
759 std::vector<int> dims;
760 sub_element_fns.reserve(_sub_elements.size());
761 dims.reserve(_sub_elements.size());
762 for (std::size_t i = 0; i < _sub_elements.size(); ++i)
763 {
764 sub_element_fns.push_back(
765 _sub_elements[i]->template dof_transformation_right_fn<U>(ttype));
766 dims.push_back(_sub_elements[i]->space_dimension());
767 }
768
769 return [dims = std::move(dims),
770 sub_element_fns = std::move(sub_element_fns)](
771 std::span<U> data, std::span<const std::uint32_t> cell_info,
772 std::int32_t cell, int block_size)
773 {
774 // `data` is (block_size, ndofs), row-major. Sub-element `e`
775 // owns columns [offset, offset + dims[e]) of every row, so the
776 // rows are transformed one at a time.
777 const std::size_t ndofs = data.size() / block_size;
778 std::size_t offset = 0;
779 for (std::size_t e = 0; e < sub_element_fns.size(); ++e)
780 {
781 if (sub_element_fns[e])
782 {
783 for (int i = 0; i < block_size; ++i)
784 {
785 sub_element_fns[e](data.subspan(i * ndofs + offset, dims[e]),
786 cell_info, cell, 1);
787 }
788 }
789 offset += dims[e];
790 }
791 };
792 }
793 else if (!scalar_element)
794 {
795 // Blocked element
796 // The transformation from the left can be used here as blocked
797 // elements use xyzxyzxyz ordering, and so applying the DOF
798 // transformation from the right is equivalent to applying the DOF
799 // transformation from the left to data using xxxyyyzzz ordering
800 std::function<void(std::span<U>, std::span<const std::uint32_t>,
801 std::int32_t, int)>
802 sub_fn
803 = _sub_elements.front()->template dof_transformation_fn<U>(ttype);
804 assert(sub_fn); // Consistent with needs_dof_transformations() above
805 return [this, sub_fn = std::move(sub_fn)](
806 std::span<U> data, std::span<const std::uint32_t> cell_info,
807 std::int32_t cell, int data_block_size)
808 {
809 const int ebs = block_size();
810 const std::size_t dof_count = data.size() / data_block_size;
811 for (int block = 0; block < data_block_size; ++block)
812 {
813 sub_fn(data.subspan(block * dof_count, dof_count), cell_info, cell,
814 ebs);
815 }
816 };
817 }
818 }
819
820 switch (ttype)
821 {
822 case doftransform::inverse_transpose:
823 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
824 std::int32_t cell, int n)
825 { Tt_inv_apply_right(data, cell_info[cell], n); };
826 case doftransform::transpose:
827 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
828 std::int32_t cell, int n)
829 { Tt_apply_right(data, cell_info[cell], n); };
830 case doftransform::inverse:
831 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
832 std::int32_t cell, int n)
833 { Tinv_apply_right(data, cell_info[cell], n); };
834 case doftransform::standard:
835 return [this](std::span<U> data, std::span<const std::uint32_t> cell_info,
836 std::int32_t cell, int n)
837 { T_apply_right(data, cell_info[cell], n); };
838 default:
839 throw std::invalid_argument("Unknown transformation type");
840 }
841 }

1066private:
1067 // Negate data on each facet marked with mesh::reversed_facet_bit, if
1068 // the basis depends on the facet signs (see
1069 // mesh::Topology::create_facet_orientations). A sign is its own
1070 // inverse and transpose, so every DOF transformation applies it the
1071 // same way.
1072 //
1073 // `post` selects the layout: the right-applied transformations store
1074 // their data as `(n, num_dofs)`, the others as `(num_dofs, n)`.
1075 template <bool post, typename U>
1076 void flip_reversed(std::span<U> data, std::uint32_t cell_permutation,
1077 int n) const
1078 {
1079 if (!_depends_on_facet_orientations
1080 or !(cell_permutation & _orientation_mask)) [[likely]]
1081 {
1082 return;
1083 }
1084
1085 // The mask is non-zero only on a cell of dimension 2, whose facets
1086 // are its edges
1087 assert(mesh::cell_dim(_cell_type) == 2 and _entity_dofs.size() > 1);
1088 const std::vector<std::vector<int>>& facet_dofs = _entity_dofs[1];
1089
1090 assert(n > 0 and data.size() % n == 0);
1091 const std::size_t num_dofs = data.size() / n;
1092 for (std::size_t f = 0; f < facet_dofs.size(); ++f)
1093 {
1094 if (!(cell_permutation & mesh::reversed_facet_bit(f)))
1095 continue;
1096 for (int dof : facet_dofs[f])
1097 {
1098 for (int j = 0; j < n; ++j)
1099 {
1100 U& value = post ? data[j * num_dofs + dof] : data[dof * n + j];
1101 value = -value;
1102 }
1103 }
1104 }
1105 }
